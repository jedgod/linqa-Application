 destination_countries TEXT[] NOT NULL DEFAULT '{}',

 effective_to TIMESTAMPTZ,

export class FinancialResponsibilityRepository {
 constructor(private readonly db: any) {}

 async findActive(a: any) {
  const r=await this.db.query(`
   SELECT * FROM linqasend_financial_responsibility_maps
   WHERE source_country=$1 AND destination_country=$2
    AND transfer_class=$3 AND source_currency=$4 AND destination_currency=$5
    AND funding_mode=$6 AND receive_mode=$7 AND status='ACTIVE'
    AND effective_from <= $8 AND (effective_to IS NULL OR effective_to > $8)
   ORDER BY policy_version DESC LIMIT 1`,
   [a.sourceCountry,a.destinationCountry,a.transferClass,a.sourceCurrency,
    a.destinationCurrency,a.fundingMode,a.receiveMode,a.at]);
  return r.rows[0] ?? null;
 }
}

   `SELECT * FROM linqasend_financial_entities WHERE id=$1 LIMIT 1`,[id]);
  return r.rows[0] ?? null;
 }
}

export class PartnerContractRepository {
 constructor(private readonly db:any) {}
 async findEffective(a:any) {
  const r=await this.db.query(`
   SELECT * FROM linqasend_partner_contracts
   WHERE regulated_entity_id=$1 AND status='ACTIVE'
    AND effective_from <= $2 AND (effective_to IS NULL OR effective_to > $2)
    AND (cardinality(source_countries)=0 OR $3=ANY(source_countries))
    AND (cardinality(destination_countries)=0 OR $4=ANY(destination_countries))
    AND (cardinality(source_currencies)=0 OR $5=ANY(source_currencies))
    AND (cardinality(destination_currencies)=0 OR $6=ANY(destination_currencies))
    AND $7=ANY(allowed_responsibilities)
   ORDER BY effective_from DESC LIMIT 1`,
   [a.entityId,a.at,a.sourceCountry,a.destinationCountry,
    a.sourceCurrency,a.destinationCurrency,a.responsibility]);
  return r.rows[0] ?? null;
 }
}

function requiredResponsibilities(a:any): FinancialResponsibility[] {
 const s=new Set<FinancialResponsibility>([
  "KYC_CIP","AML_MONITORING","SANCTIONS_SCREENING",
  "REFUNDS","SETTLEMENT","RECONCILIATION"
 ]);
 const cross=a.transferClass==="INTERNAL_CROSS_BORDER" ||
             a.transferClass==="EXTERNAL_CROSS_BORDER";
 s.add(cross?"CROSS_BORDER_TRANSMISSION":"DOMESTIC_TRANSMISSION");
 if(a.sourceCurrency!==a.destinationCurrency) s.add("FX_EXECUTION");

 if(["DEBIT_CARD","CREDIT_CARD"].includes(a.fundingMode)) {
  s.add("CARD_ACQUIRING"); s.add("ACCEPT_CUSTOMER_FUNDS"); s.add("CHARGEBACKS");
 } else if(a.fundingMode==="BANK") {
  s.add("BANK_DEBIT"); s.add("ACCEPT_CUSTOMER_FUNDS");
 } else if(a.fundingMode==="LINQA_BALANCE") {
  s.add("CUSTOMER_FUNDS_CUSTODY");
 }

 if(a.receiveMode==="LINQA_BALANCE") {
  s.add("CUSTOMER_ACCOUNT_ISSUER"); s.add("STORED_VALUE_ISSUER");
 } else if(a.receiveMode==="BANK") s.add("LOCAL_BANK_PAYOUT");
 else if(a.receiveMode==="MOBILE_MONEY") s.add("MOBILE_MONEY_PAYOUT");
 else if(a.receiveMode==="CARD") s.add("CARD_PUSH_PAYOUT");

 if(["PARTNER_CUSTODY","FBO_OR_EQUIVALENT_PARTNER_ACCOUNT"].includes(a.custodyModel)){
  s.add("CUSTOMER_FUNDS_CUSTODY"); s.add("SAFEGUARDING");
 }
 if(a.custodyModel==="PARTNER_ISSUED_STORED_VALUE"){
  s.add("STORED_VALUE_ISSUER"); s.add("SAFEGUARDING");
 }
 return [...s];
}

export class RegulatedEntityResolverService {
 constructor(private maps:any,private entities:any,private contracts:any){}

 async resolve(a:any):Promise<FinancialResponsibilityResolution>{
  const at=a.at??new Date();
  const map=await this.maps.findActive({...a,at});
  if(!map) return {ready:false,mapId:"",corridorId:"",policyVersion:0,
   custodyModel:"NO_CUSTOMER_FUNDS",legalApprovalReference:"",
   partnerApprovalReference:"",authorities:[],missing:[],
   blockedReasons:["FINANCIAL_RESPONSIBILITY_MAP_NOT_CONFIGURED"]};

  const required=requiredResponsibilities({...a,custodyModel:map.custody_model});
  const authorities:ResolvedAuthority[]=[]; const missing:any[]=[]; const blocked:string[]=[];

  for(const responsibility of required){
   const entityId=map.responsibility_entities?.[responsibility];
   if(!entityId){ missing.push(responsibility); continue; }

   const e=await this.entities.findById(entityId);
   if(!e){blocked.push(`ENTITY_NOT_FOUND:${responsibility}`);continue;}
   if(e.status!=="ACTIVE"){blocked.push(`ENTITY_NOT_ACTIVE:${responsibility}`);continue;}
   if(e.due_diligence_status!=="APPROVED"){blocked.push(`DUE_DILIGENCE_INVALID:${entityId}`);continue;}
   if(["DOWN","UNKNOWN"].includes(e.operational_status)){
    blocked.push(`ENTITY_UNAVAILABLE:${responsibility}`);continue;
   }

   const c=await this.contracts.findEffective({...a,entityId,responsibility,at});
   if(!c){blocked.push(`ACTIVE_CONTRACT_NOT_FOUND:${responsibility}`);continue;}

   authorities.push({responsibility,entityId:e.id,legalName:e.legal_name,
    entityType:e.entity_type,contractId:c.id,
    providerAdapterKey:c.provider_adapter_key,
    regulatoryReference:e.regulatory_reference});
  }

  return {ready:missing.length===0&&blocked.length===0,
   mapId:map.id,corridorId:map.corridor_id,policyVersion:map.policy_version,
   custodyModel:map.custody_model,
   legalApprovalReference:map.legal_approval_reference,
   partnerApprovalReference:map.partner_approval_reference,
   authorities,missing,blockedReasons:blocked};
 }
}

export class RealMoneyReadinessError extends Error {
 code="LINQASEND_REAL_MONEY_NOT_READY";
 constructor(public details:any){super("Real-money responsibility resolution failed");}
}
export class RealMoneyReadinessService {
 constructor(private resolver:RegulatedEntityResolverService){}
 async requireReady(a:any){
  const r=await this.resolver.resolve(a);
  if(!r.ready) throw new RealMoneyReadinessError({
   mapId:r.mapId,missing:r.missing,blockedReasons:r.blockedReasons});
  return r;
 }
}
