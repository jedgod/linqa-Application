# Integration patch guide

## 1. Keep rejected features rejected
Do not add `onlineOnly`, last-seen exposure, presence ranking, or stranger presence APIs.
Do not create another moderation service. Call the existing block and restricted-user checks.

## 2. Replace only the cache layer
Locate the current in-process CountryLinqa cache in the live discovery service.
Replace storage with `CountryLinqaDistributedCache`.
Keep the existing post-fetch authorization loop and make it explicit:
cache hit -> block check -> restricted-user check -> mutual discovery/privacy check -> return.

If Redis is unavailable, execute uncached. Never fail open on privacy.

## 3. Preserve bounded ranking but make it operationally tunable
Replace literal `300` candidate fetch limits with:
`countryLinqaCandidatePoolLimit()`.
Default stays 300. Hard maximum is 1000.
This closes the hard-coded scaling gap without allowing an unbounded query.

## 4. Database tests
The 41 skipped tests are not a product failure when DATABASE_URL is absent locally.
For certification/CI, skipping is unacceptable.
Use `LINQA_REQUIRE_DB_TESTS=1` + DATABASE_URL and run both the existing chat integration
suite and the 13 CountryLinqa integration tests through the repository's real test scripts.

Do not run an unreviewed destructive suite against a shared dev database.
Use a dedicated CI/test PostgreSQL database.

## 5. Commit isolation
Because unrelated Copilot edits are present, do NOT `git add -A`.
Use `scripts/commit-countrylinqa-only.ps1`, inspect the staged list, then commit.
If a shared file contains both Copilot and CountryLinqa hunks, use `git add -p <file>` manually.

## 6. Physical certification
CI can verify manifests and evidence shape, but it cannot create physical evidence.
Use two physical devices/accounts whose verified countries differ.
Bind evidence to exact Git SHA + server/mobile SHA-256.
Keep iOS blocked until a real iOS target/artifact exists.

# CountryLinqa™ Repo-Native Gap Closure v4.1

This patch closes the software/CI gaps that remain after the repo-native integration.

Decisions intentionally preserved:
- NO "online now" stranger filter. Presence stays private.
- NO duplicate moderation service. Reuse existing block/restricted-user enforcement.
- Ranking remains bounded, but the candidate window is configurable and hard-capped.
- Physical two-country certification remains an execution gate, not something CI can fabricate.
- iOS remains blocked until a real iOS project/build target exists.

What this patch changes:
1. Replaces process-local discovery cache with a shared Redis-compatible cache adapter.
2. Keeps privacy independent from cache: every cached page is re-authorized before response.
3. Adds policy-versioned keys and invalidation.
4. Makes candidate pool limit configurable (default 300, hard maximum 1000).
5. Adds DB-required test gate so CI cannot silently skip PostgreSQL integration tests.
6. Adds skipped-test audit/reporting.
7. Adds CountryLinqa exact-SHA certification manifest and physical evidence verifier.
8. Adds an isolated Git commit helper so CountryLinqa changes can be committed without
   sweeping unrelated Copilot edits into the same commit.

Adapt imports to the live Linqa Redis/config/logger primitives if equivalent helpers already exist.

import fs from "node:fs";
import path from "node:path";

const roots=["server/src/tests","clients/mobile"];
const offenders=[];
function walk(p){
  if(!fs.existsSync(p)) return;
  for(const e of fs.readdirSync(p,{withFileTypes:true})){
    const f=path.join(p,e.name);
    if(e.isDirectory()) walk(f);
    else if(/\.(test|spec)\.(ts|tsx|js|mjs)$/.test(e.name)){
      const s=fs.readFileSync(f,"utf8");
      const lines=s.split(/\r?\n/);
      lines.forEach((line,i)=>{
        if(/\b(it|test|describe)\.skip\s*\(/.test(line))
          offenders.push(`${f}:${i+1}: explicit skip`);
        if(/\b(it|test|describe)\.only\s*\(/.test(line))
          offenders.push(`${f}:${i+1}: .only`);
      });
    }
  }
}
roots.forEach(walk);
if(offenders.length){
  console.error(offenders.join("\n"));
  process.exit(3);
}
console.log("No explicit .skip/.only found.");

const DEFAULT_POOL=300;
const HARD_MAX=1000;

export function countryLinqaCandidatePoolLimit(env=process.env){
  const raw=Number(env.COUNTRYLINQA_CANDIDATE_POOL_LIMIT ?? DEFAULT_POOL);
  if(!Number.isFinite(raw)) return DEFAULT_POOL;
  return Math.max(50,Math.min(HARD_MAX,Math.trunc(raw)));
}

export const COUNTRYLINQA_CANDIDATE_POOL_HARD_MAX=HARD_MAX;

import crypto from "node:crypto";

export interface CountryLinqaRedis {
  get(key:string):Promise<string|null>;
  set(key:string,value:string,...args:any[]):Promise<unknown>;
  del(...keys:string[]):Promise<unknown>;
}

export type CacheEnvelope<T>={
  policyVersion:number;
  generatedAt:string;
  value:T;
};

const stable=(x:unknown)=>JSON.stringify(x,Object.keys((x ?? {}) as any).sort());

export function discoveryCacheKey(input:{
  userId:string;
  policyVersion:number;
  surface:string;
  filters:unknown;
  cursor?:string|null;
}){
  const digest=crypto.createHash("sha256")
    .update(stable({filters:input.filters,cursor:input.cursor ?? null}))
    .digest("hex").slice(0,24);
  return `countrylinqa:v4:${input.userId}:p${input.policyVersion}:${input.surface}:${digest}`;
}

export class CountryLinqaDistributedCache {
  constructor(
    private redis:CountryLinqaRedis|null,
    private ttlSeconds=45
  ){}

  async get<T>(key:string,policyVersion:number):Promise<T|null>{
    if(!this.redis) return null; // fail open for performance, never for authorization
    const raw=await this.redis.get(key);
    if(!raw) return null;
    try{
      const env=JSON.parse(raw) as CacheEnvelope<T>;
      if(env.policyVersion!==policyVersion) return null;
      return env.value;
    }catch{
      return null;
    }
  }

  async set<T>(key:string,policyVersion:number,value:T){
    if(!this.redis) return;
    const env:CacheEnvelope<T>={
      policyVersion,
      generatedAt:new Date().toISOString(),
      value
    };
    await this.redis.set(key,JSON.stringify(env),"EX",this.ttlSeconds);
  }
}

 * The caller MUST run the existing block/restricted-user/privacy eligibility

/**
 * Wire these callbacks to the existing repo-native eligibility functions.
 * This file intentionally does NOT create a second moderation implementation.
 */
export async function recheckCachedCandidates<T extends {userId:string}>(
  rows:T[],
  checks:{
    blocked:(candidateUserId:string)=>Promise<boolean>;
    restricted:(candidateUserId:string)=>Promise<boolean>;
    mutuallyEligible:(candidateUserId:string)=>Promise<boolean>;
  }
){
  const out:T[]=[];
  for(const row of rows){
    if(await checks.blocked(row.userId)) continue;
    if(await checks.restricted(row.userId)) continue;
    if(!(await checks.mutuallyEligible(row.userId))) continue;
    out.push(row);
  }
  return out;
}

import {describe,it,expect} from "vitest";
import {
  CountryLinqaDistributedCache,
  discoveryCacheKey
} from "../services/countrylinqa/distributed-cache";
import {recheckCachedCandidates} from "../services/countrylinqa/privacy-recheck";
import {countryLinqaCandidatePoolLimit} from "../services/countrylinqa/candidate-window";

class FakeRedis{
  m=new Map<string,string>();
  async get(k:string){return this.m.get(k) ?? null;}
  async set(k:string,v:string){this.m.set(k,v);return "OK";}
  async del(...ks:string[]){for(const k of ks)this.m.delete(k);return ks.length;}
}

describe("CountryLinqa shared cache contract",()=>{
  it("versions cache by policy",async()=>{
    const r=new FakeRedis();
    const c=new CountryLinqaDistributedCache(r as any,30);
    const k=discoveryCacheKey({
      userId:"u1",policyVersion:7,surface:"PEOPLE",filters:{language:"en"}
    });
    await c.set(k,7,[{userId:"u2"}]);
    expect(await c.get(k,7)).toEqual([{userId:"u2"}]);
    expect(await c.get(k,8)).toBeNull();
  });

  it("rechecks privacy after a cache hit",async()=>{
    const rows=[{userId:"ok"},{userId:"blocked"},{userId:"restricted"}];
    const out=await recheckCachedCandidates(rows,{
      blocked:async id=>id==="blocked",
      restricted:async id=>id==="restricted",
      mutuallyEligible:async()=>true
    });
    expect(out.map(x=>x.userId)).toEqual(["ok"]);
  });

  it("hard caps candidate pool",()=>{
    expect(countryLinqaCandidatePoolLimit({COUNTRYLINQA_CANDIDATE_POOL_LIMIT:"5000"} as any))
      .toBe(1000);
  });
});
