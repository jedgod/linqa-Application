# CountryLinqa™ Discovery Engine Integration v1

This is the integration layer after the Discovery Settings Production Finish pack.

It connects the user's server-authoritative CountryLinqa policy to:
- People
- Country feeds/posts
- Events
- Businesses
- Communities
- Unified search
- Recommendation candidate generation

Core rule:
A client never decides whether a candidate is geographically eligible.
The server resolves the viewer's effective discovery policy and applies it before
ranking, pagination, profile visibility, or recommendation delivery.

Important:
- Verified identity country remains separate from discovery preferences.
- Block/mute/safety rules override CountryLinqa discovery.
- Inbound policy and outbound policy must BOTH permit a new-person exposure.
- Existing contacts may be handled separately from new discovery.
- Precise location is not required by this engine.

import { db } from "../../db/pool.js";
import { CountryLinqaPolicyResolver } from "./policyResolver.js";
import { appendCountryPredicate } from "./sqlScope.js";

export class CountryContentDiscoveryService {
  private resolver=new CountryLinqaPolicyResolver();

  async posts(userId:string,limit=30) {
    const p=await this.resolver.resolve(userId);
    const scope=this.resolver.outboundCountries(p);
    let params:any[]=[userId];
    const country=appendCountryPredicate(params,"posts.country_code",scope);
    params=country.params;
    params.push(limit);

    return db.query(`
      SELECT posts.*
      FROM country_posts posts
      WHERE posts.status='PUBLISHED'
        ${country.sql}
        AND NOT EXISTS(
          SELECT 1 FROM user_blocks b
          WHERE b.blocker_user_id=$1 AND b.blocked_user_id=posts.author_user_id
        )
      ORDER BY posts.created_at DESC
      LIMIT $${params.length}
    `,params);
  }

  async events(userId:string,limit=30) {
    const p=await this.resolver.resolve(userId);
    if (!p.allowEvents) return {rows:[]};
    const scope=this.resolver.outboundCountries(p);
    let params:any[]=[];
    const country=appendCountryPredicate(params,"e.country_code",scope);
    params=country.params;
    params.push(limit);
    return db.query(`
      SELECT e.* FROM events e
      WHERE e.visibility='PUBLIC' ${country.sql}
      ORDER BY e.starts_at ASC
      LIMIT $${params.length}
    `,params);
  }

  async businesses(userId:string,limit=30) {
    const p=await this.resolver.resolve(userId);
    if (!p.allowBusinesses) return {rows:[]};
    const scope=this.resolver.outboundCountries(p);
    let params:any[]=[];
    const country=appendCountryPredicate(params,"b.country_code",scope);
    params=country.params;
    params.push(limit);
    return db.query(`
      SELECT b.* FROM businesses b
      WHERE b.status='ACTIVE' ${country.sql}
      ORDER BY b.discovery_score DESC
      LIMIT $${params.length}
    `,params);
  }

  async communities(userId:string,limit=30) {
    const p=await this.resolver.resolve(userId);
    if (!p.allowCommunities) return {rows:[]};
    const scope=this.resolver.outboundCountries(p);
    let params:any[]=[];
    const country=appendCountryPredicate(params,"c.country_code",scope);
    params=country.params;
    params.push(limit);
    return db.query(`
      SELECT c.* FROM communities c
      WHERE c.status='ACTIVE' ${country.sql}
      ORDER BY c.member_count DESC
      LIMIT $${params.length}
    `,params);
  }
}

import { countriesForRegion } from "./regionCatalog.js";
import type { EffectiveDiscoveryPolicy } from "./types.js";

export type Candidate = {
  userId:string;
  verifiedCountryCode:string;
  verified:boolean;
  inboundMode:"MY_COUNTRY"|"SELECTED_COUNTRIES"|"REGION"|"GLOBAL"|"CONTACTS_ONLY";
  inboundCountryCodes:string[];
  inboundRegionCode:string|null;
  discoveryPaused:boolean;
};

function outboundAllows(viewer:EffectiveDiscoveryPolicy,candidateCountry:string) {
  switch(viewer.outboundMode) {
    case "MY_COUNTRY": return candidateCountry===viewer.viewerCountryCode;
    case "SELECTED_COUNTRIES": return viewer.outboundCountryCodes.includes(candidateCountry);
    case "REGION": return countriesForRegion(viewer.outboundRegionCode).includes(candidateCountry);
    case "GLOBAL": return true;
  }
}

function inboundAllows(candidate:Candidate,viewerCountry:string,isExistingContact:boolean) {
  if (candidate.discoveryPaused && !isExistingContact) return false;

  switch(candidate.inboundMode) {
    case "MY_COUNTRY": return candidate.verifiedCountryCode===viewerCountry;
    case "SELECTED_COUNTRIES": return candidate.inboundCountryCodes.includes(viewerCountry);
    case "REGION": return countriesForRegion(candidate.inboundRegionCode).includes(viewerCountry);
    case "GLOBAL": return true;
    case "CONTACTS_ONLY": return isExistingContact;
  }
}

export function canExposePerson(input:{
  viewer:EffectiveDiscoveryPolicy;
  candidate:Candidate;
  isExistingContact:boolean;
  isBlocked:boolean;
}) {
  if (input.isBlocked) return false;
  if (input.viewer.verifiedOnly && !input.candidate.verified) return false;
  if (!outboundAllows(input.viewer,input.candidate.verifiedCountryCode)) return false;
  if (!inboundAllows(input.candidate,input.viewer.viewerCountryCode,input.isExistingContact)) return false;
  return true;
}

import { db } from "../../db/pool.js";
import { CountryLinqaPolicyResolver } from "./policyResolver.js";
import { canExposePerson } from "./eligibility.js";

export class PeopleDiscoveryService {
  private resolver=new CountryLinqaPolicyResolver();

  async list(viewerUserId:string,limit=30,cursor?:string) {
    const policy=await this.resolver.resolve(viewerUserId);

    const rows=await db.query(`
      SELECT
        u.id AS user_id,
        u.display_name,
        u.verified_country_code,
        COALESCE(u.identity_verified,FALSE) AS verified,
        COALESCE(p.inbound_mode,'MY_COUNTRY') AS inbound_mode,
        COALESCE(p.inbound_country_codes,'{}') AS inbound_country_codes,
        p.inbound_region_code,
        COALESCE(p.discovery_paused,FALSE) AS discovery_paused,
        EXISTS(
          SELECT 1 FROM contacts c
          WHERE c.owner_user_id=$1 AND c.contact_user_id=u.id
        ) AS existing_contact,
        EXISTS(
          SELECT 1 FROM user_blocks b
          WHERE (b.blocker_user_id=$1 AND b.blocked_user_id=u.id)
             OR (b.blocker_user_id=u.id AND b.blocked_user_id=$1)
        ) AS blocked
      FROM users u
      LEFT JOIN country_discovery_policy_v2 p ON p.user_id=u.id
      WHERE u.id<>$1
      ORDER BY u.id
      LIMIT $2
    `,[viewerUserId,Math.min(limit*4,120)]);

    const visible=rows.rows.filter((r:any)=>canExposePerson({
      viewer:policy,
      candidate:{
        userId:r.user_id,
        verifiedCountryCode:r.verified_country_code,
        verified:r.verified,
        inboundMode:r.inbound_mode,
        inboundCountryCodes:r.inbound_country_codes,
        inboundRegionCode:r.inbound_region_code,
        discoveryPaused:r.discovery_paused
      },
      isExistingContact:r.existing_contact,
      isBlocked:r.blocked
    })).slice(0,limit);

    return {
      items:visible,
      nextCursor:visible.length===limit ? visible.at(-1)?.user_id : null
    };
  }
}

import { db } from "../../db/pool.js";
import { countriesForRegion } from "./regionCatalog.js";
import type { EffectiveDiscoveryPolicy } from "./types.js";

export class CountryLinqaPolicyResolver {
  async resolve(userId:string):Promise<EffectiveDiscoveryPolicy> {
    const r = await db.query(`
      SELECT
        u.id,
        u.verified_country_code,
        p.outbound_mode,
        p.inbound_mode,
        p.outbound_country_codes,
        p.inbound_country_codes,
        p.outbound_region_code,
        p.inbound_region_code,
        p.verified_only,
        p.discovery_paused,
        p.allow_businesses,
        p.allow_communities,
        p.allow_events
      FROM users u
      LEFT JOIN country_discovery_policy_v2 p ON p.user_id=u.id
      WHERE u.id=$1
    `,[userId]);

    if (!r.rowCount) throw Object.assign(new Error("USER_NOT_FOUND"),{status:404});

    const x=r.rows[0];
    if (!x.verified_country_code)
      throw Object.assign(new Error("COUNTRY_PROFILE_REQUIRED"),{status:409});

    return {
      viewerUserId:userId,
      viewerCountryCode:x.verified_country_code,
      outboundMode:x.outbound_mode ?? "MY_COUNTRY",
      outboundCountryCodes:x.outbound_country_codes ?? [],
      outboundRegionCode:x.outbound_region_code ?? null,
      inboundMode:x.inbound_mode ?? "MY_COUNTRY",
      inboundCountryCodes:x.inbound_country_codes ?? [],
      inboundRegionCode:x.inbound_region_code ?? null,
      verifiedOnly:x.verified_only ?? true,
      discoveryPaused:x.discovery_paused ?? false,
      allowBusinesses:x.allow_businesses ?? true,
      allowCommunities:x.allow_communities ?? true,
      allowEvents:x.allow_events ?? true
    };
  }

  outboundCountries(p:EffectiveDiscoveryPolicy):string[]|null {
    if (p.outboundMode==="GLOBAL") return null;
    if (p.outboundMode==="MY_COUNTRY") return [p.viewerCountryCode];
    if (p.outboundMode==="SELECTED_COUNTRIES") return p.outboundCountryCodes;
    return countriesForRegion(p.outboundRegionCode);
  }
}

import type { EffectiveDiscoveryPolicy } from "./types.js";
import { countriesForRegion } from "./regionCatalog.js";

export function outboundCountryScope(policy:EffectiveDiscoveryPolicy) {
  if (policy.outboundMode==="GLOBAL") return null;
  if (policy.outboundMode==="MY_COUNTRY") return [policy.viewerCountryCode];
  if (policy.outboundMode==="SELECTED_COUNTRIES") return policy.outboundCountryCodes;
  return countriesForRegion(policy.outboundRegionCode);
}

export function appendCountryPredicate(
  params:any[],
  countryColumn:string,
  countries:string[]|null
) {
  if (countries===null) return {sql:"",params};
  params.push(countries);
  return {
    sql:` AND ${countryColumn} = ANY($${params.length}::text[]) `,
    params
  };
}

import { CountryLinqaPolicyResolver } from "./policyResolver.js";

export class CountryLinqaUnifiedSearch {
  private resolver=new CountryLinqaPolicyResolver();

  async search(input:{
    userId:string;
    query:string;
    types:Array<"PEOPLE"|"POSTS"|"EVENTS"|"BUSINESSES"|"COMMUNITIES">;
    limit:number;
  }) {
    const policy=await this.resolver.resolve(input.userId);

    // Production integration:
    // 1. Normalize query.
    // 2. Resolve policy country scope.
    // 3. Query each requested vertical.
    // 4. Apply block/safety/inbound eligibility before exposing PEOPLE.
    // 5. Apply recommendation toggles to events/businesses/communities.
    // 6. Rank only eligible candidates.
    // 7. Return opaque cursor, never raw relevance internals.
    return {
      query:input.query,
      scope:{
        mode:policy.outboundMode,
        countries:this.resolver.outboundCountries(policy)
      },
      results:[],
      nextCursor:null
    };
  }
}

    inboundCountryCodes:[],
