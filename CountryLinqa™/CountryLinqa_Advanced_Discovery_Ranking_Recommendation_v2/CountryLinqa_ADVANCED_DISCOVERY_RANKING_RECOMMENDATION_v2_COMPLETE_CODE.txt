# CountryLinqa™ Two-Country Physical Certification

This is an execution template. Do not mark PASS until tests are performed.

## Required evidence
- Release Git SHA
- Server artifact digest
- Mobile build number / binary digest
- Database migration level
- Device A model / OS
- Device B model / OS
- Test account IDs (internal evidence only)
- Verified country for A
- Verified country for B
- UTC timestamps
- Screen recordings/screenshots
- Relevant server logs / request IDs

## Scenario A — My Country
Device A: verified country A
Device B: verified country B
1. Set A outbound = MY_COUNTRY.
2. Confirm B is not exposed to A if B is in another country.
3. Confirm eligible same-country control account appears.
4. Confirm no client-side override can expose B.

## Scenario B — Selected Countries
1. Add country B to A's selected countries.
2. Confirm B appears only if B's inbound policy permits country A.
3. Remove country B.
4. Confirm B disappears after cache invalidation.

## Scenario C — Candidate inbound restriction
1. B inbound = MY_COUNTRY.
2. A remains outside B's country.
3. Even if A outbound includes B, B must not appear.
4. Change B inbound to include A's country.
5. Confirm eligibility becomes true.

## Scenario D — Region
1. A selects a supported region containing B.
2. Confirm B appears if mutually eligible.
3. Confirm country outside selected region does not appear.

## Scenario E — Worldwide
1. Use an Elite entitlement.
2. Set A outbound = GLOBAL.
3. Confirm global scope does not bypass B's inbound restrictions.

## Scenario F — Safety overrides
- Block B from A and verify immediate disappearance.
- Pause B discovery and verify B disappears for strangers.
- Set B to CONTACTS_ONLY and verify only existing contacts retain eligibility.

## Scenario G — Filters
- Language
- Interest
- Profession
- Verified only
- Online now
Verify each filter narrows an already eligible set and never broadens geographic eligibility.

## Scenario H — Saved search
1. Save filter set.
2. Restart application.
3. Re-run saved search.
4. Confirm server re-evaluates current policy rather than replaying stale results.

## Scenario I — Pagination/cache
1. Load page 1 and page 2.
2. Ensure no duplicates.
3. Change policy.
4. Confirm old cursor/cache cannot leak out-of-scope candidates.

## PASS requirements
Every scenario must pass on both physical devices and evidence must be bound to the exact deployed Git SHA.

export interface AbuseStore {
  increment(key:string,windowSeconds:number):Promise<number>;
}

export class DiscoveryAbuseGuard {
  constructor(private store:AbuseStore){}

  async assertAllowed(input:{
    userId:string;
    installationId:string;
    action:"SEARCH"|"PAGE"|"PROFILE_OPEN"|"SAVE_SEARCH";
  }) {
    const limits={
      SEARCH:{window:60,max:30},
      PAGE:{window:60,max:60},
      PROFILE_OPEN:{window:60,max:90},
      SAVE_SEARCH:{window:3600,max:20}
    } as const;

    const rule=limits[input.action];
    const key=`clq:abuse:${input.action}:${input.userId}:${input.installationId}`;
    const count=await this.store.increment(key,rule.window);

    if (count>rule.max)
      throw Object.assign(new Error("DISCOVERY_RATE_LIMITED"),{
        status:429,
        retryAfter:rule.window
      });
  }
}

import { db } from "../../../db/pool.js";
import { CountryLinqaPolicyResolver } from "../policyResolver.js";
import { canExposePerson } from "../eligibility.js";
import { validateFilters } from "./filterValidator.js";
import { rankAndDiversify } from "./ranker.js";
import { encodeCursor } from "./cursor.js";

export class AdvancedDiscoveryService {
  private policyResolver=new CountryLinqaPolicyResolver();

  async people(input:{
    userId:string;
    filters:any;
    limit:number;
  }) {
    const started=Date.now();
    const filters=validateFilters(input.filters);
    const policy=await this.policyResolver.resolve(input.userId);

    const viewerFeatureResult=await db.query(`
      SELECT
        u.id,
        u.verified_country_code,
        COALESCE(f.languages,'{}') languages,
        COALESCE(f.interests,'{}') interests,
        f.profession_code
      FROM users u
      LEFT JOIN countrylinqa_user_discovery_features f ON f.user_id=u.id
      WHERE u.id=$1
    `,[input.userId]);

    const viewerRow=viewerFeatureResult.rows[0];
    const viewer={
      userId:input.userId,
      countryCode:viewerRow.verified_country_code,
      languages:viewerRow.languages,
      interests:viewerRow.interests,
      professionCode:viewerRow.profession_code
    };

    // Fetch an oversampled pool. For very large installations, push more
    // eligibility predicates into SQL and/or a search index before ranking.
    const pool=await db.query(`
      SELECT
        u.id user_id,
        u.verified_country_code country_code,
        COALESCE(u.identity_verified,FALSE) verified,
        COALESCE(f.languages,'{}') languages,
        COALESCE(f.interests,'{}') interests,
        f.profession_code,
        f.last_active_at,
        COALESCE(p.inbound_mode,'MY_COUNTRY') inbound_mode,
        COALESCE(p.inbound_country_codes,'{}') inbound_country_codes,
        p.inbound_region_code,
        COALESCE(p.discovery_paused,FALSE) discovery_paused,
        COALESCE(pc.profile_quality,0.5) profile_quality,
        COALESCE(pc.safety_quality,0.5) safety_quality,
        CASE WHEN f.last_active_at > now()-interval '24 hours' THEN 1.0 ELSE 0.3 END freshness,
        (
          SELECT COUNT(*)::int FROM contacts a
          JOIN contacts b ON b.contact_user_id=a.contact_user_id
          WHERE a.owner_user_id=$1 AND b.owner_user_id=u.id
        ) mutual_connections,
        EXISTS(
          SELECT 1 FROM contacts c
          WHERE c.owner_user_id=$1 AND c.contact_user_id=u.id
        ) existing_contact,
        EXISTS(
          SELECT 1 FROM user_blocks b
          WHERE (b.blocker_user_id=$1 AND b.blocked_user_id=u.id)
             OR (b.blocker_user_id=u.id AND b.blocked_user_id=$1)
        ) blocked
      FROM users u
      LEFT JOIN countrylinqa_user_discovery_features f ON f.user_id=u.id
      LEFT JOIN country_discovery_policy_v2 p ON p.user_id=u.id
      LEFT JOIN user_profile_quality pc ON pc.user_id=u.id
      WHERE u.id<>$1
      LIMIT 500
    `,[input.userId]);

    const eligible=pool.rows.filter((r:any)=>{
      if (filters.verifiedOnly && !r.verified) return false;
      if (filters.onlineOnly && (!r.last_active_at ||
          Date.now()-new Date(r.last_active_at).getTime()>15*60*1000)) return false;
      if (filters.languages?.length &&
          !filters.languages.some((x:string)=>r.languages.includes(x))) return false;
      if (filters.interests?.length &&
          !filters.interests.some((x:string)=>r.interests.includes(x))) return false;
      if (filters.profession && r.profession_code!==filters.profession) return false;

      return canExposePerson({
        viewer:policy,
        candidate:{
          userId:r.user_id,
          verifiedCountryCode:r.country_code,
          verified:r.verified,
          inboundMode:r.inbound_mode,
          inboundCountryCodes:r.inbound_country_codes,
          inboundRegionCode:r.inbound_region_code,
          discoveryPaused:r.discovery_paused
        },
        isExistingContact:r.existing_contact,
        isBlocked:r.blocked
      });
    });

    const ranked=rankAndDiversify(viewer,eligible.map((r:any)=>({
      userId:r.user_id,
      countryCode:r.country_code,
      verified:r.verified,
      languages:r.languages,
      interests:r.interests,
      professionCode:r.profession_code,
      lastActiveAt:r.last_active_at,
      mutualConnections:r.mutual_connections,
      profileQuality:Number(r.profile_quality),
      safetyQuality:Number(r.safety_quality),
      freshness:Number(r.freshness)
    })),Math.min(input.limit,50));

    const last=ranked.at(-1);
    return {
      items:ranked.map(({score,...x})=>({...x,rankingScore:score})),
      nextCursor:last ? encodeCursor({
        lastScore:last.score,
        lastUserId:last.userId,
        policyVersion:(policy as any).version ?? 1,
        issuedAt:Date.now()
      }) : null,
      meta:{latencyMs:Date.now()-started}
    };
  }
}

import crypto from "node:crypto";

export interface RedisLike {
  get(k:string):Promise<string|null>;
  set(k:string,v:string,mode:string,ttl:number):Promise<unknown>;
  del(...keys:string[]):Promise<unknown>;
  scan(cursor:string,...args:any[]):Promise<[string,string[]]>;
}

export function discoveryCacheKey(input:{
  userId:string;
  policyVersion:number;
  surface:string;
  filters:unknown;
  cursor?:string|null;
}) {
  const digest=crypto.createHash("sha256")
    .update(JSON.stringify({filters:input.filters,cursor:input.cursor ?? null}))
    .digest("hex")
    .slice(0,24);
  return `clq:disc:v2:${input.userId}:${input.policyVersion}:${input.surface}:${digest}`;
}

export async function cached<T>(
  redis:RedisLike,
  key:string,
  ttlSeconds:number,
  loader:()=>Promise<T>
):Promise<T> {
  const hit=await redis.get(key);
  if (hit) return JSON.parse(hit);
  const value=await loader();
  await redis.set(key,JSON.stringify(value),"EX",ttlSeconds);
  return value;
}

// Prefer versioned keys. Policy updates increment policy version,
// naturally making old cache entries unreachable. Expiration cleans them up.

import crypto from "node:crypto";

const secret=process.env.COUNTRYLINQA_CURSOR_SECRET;
if (!secret) throw new Error("COUNTRYLINQA_CURSOR_SECRET_REQUIRED");

export type CursorPayload={
  lastScore:number;
  lastUserId:string;
  policyVersion:number;
  issuedAt:number;
};

export function encodeCursor(p:CursorPayload) {
  const body=Buffer.from(JSON.stringify(p)).toString("base64url");
  const sig=crypto.createHmac("sha256",secret).update(body).digest("base64url");
  return `${body}.${sig}`;
}

export function decodeCursor(token:string):CursorPayload {
  const [body,sig]=token.split(".");
  if (!body || !sig) throw Object.assign(new Error("INVALID_CURSOR"),{status:400});
  const expected=crypto.createHmac("sha256",secret).update(body).digest("base64url");
  if (!crypto.timingSafeEqual(Buffer.from(sig),Buffer.from(expected)))
    throw Object.assign(new Error("INVALID_CURSOR"),{status:400});
  const p=JSON.parse(Buffer.from(body,"base64url").toString("utf8"));
  if (Date.now()-p.issuedAt>15*60*1000)
    throw Object.assign(new Error("CURSOR_EXPIRED"),{status:400});
  return p;
}

import type { DiscoveryFilters } from "./types.js";

const COUNTRY=/^[A-Z]{2}$/;
const SAFE_TOKEN=/^[\p{L}\p{N} ._+#&/-]{1,60}$/u;

export function validateFilters(raw:DiscoveryFilters):DiscoveryFilters {
  const out:DiscoveryFilters={};

  if (raw.query) out.query=raw.query.trim().slice(0,80);

  if (raw.countries) {
    out.countries=[...new Set(raw.countries.map(x=>x.toUpperCase()))].slice(0,50);
    if (out.countries.some(x=>!COUNTRY.test(x)))
      throw Object.assign(new Error("INVALID_COUNTRY_FILTER"),{status:422});
  }

  if (raw.languages)
    out.languages=[...new Set(raw.languages.filter(x=>SAFE_TOKEN.test(x)))].slice(0,10);

  if (raw.interests)
    out.interests=[...new Set(raw.interests.filter(x=>SAFE_TOKEN.test(x)))].slice(0,20);

  if (raw.profession) {
    if (!SAFE_TOKEN.test(raw.profession))
      throw Object.assign(new Error("INVALID_PROFESSION_FILTER"),{status:422});
    out.profession=raw.profession;
  }

  if (raw.region) out.region=raw.region.slice(0,20);
  out.verifiedOnly=raw.verifiedOnly ?? false;
  out.onlineOnly=raw.onlineOnly ?? false;
  return out;
}

import type { RankCandidate, RankedCandidate, ViewerFeatures } from "./types.js";

const clamp=(n:number,min=0,max=1)=>Math.max(min,Math.min(max,n));
const overlap=(a:string[],b:string[])=>{
  if (!a.length || !b.length) return 0;
  const B=new Set(b.map(x=>x.toLowerCase()));
  return a.filter(x=>B.has(x.toLowerCase())).length / Math.max(a.length,b.length);
};

export function rankCandidate(
  viewer:ViewerFeatures,
  c:RankCandidate
):RankedCandidate {
  const interest=overlap(viewer.interests,c.interests);
  const language=overlap(viewer.languages,c.languages);
  const profession=
    viewer.professionCode && c.professionCode===viewer.professionCode ? 1 : 0;
  const mutual=clamp(c.mutualConnections/10);
  const quality=clamp(c.profileQuality);
  const safety=clamp(c.safetyQuality);
  const freshness=clamp(c.freshness);

  // Geographic eligibility is intentionally NOT a ranking weight.
  // It must already have been enforced as a hard policy gate.
  const score =
    interest * 0.25 +
    language * 0.15 +
    profession * 0.10 +
    mutual * 0.18 +
    quality * 0.10 +
    safety * 0.17 +
    freshness * 0.05;

  const reasons:string[]=[];
  if (interest>=0.2) reasons.push("SHARED_INTERESTS");
  if (language>=0.2) reasons.push("SHARED_LANGUAGE");
  if (profession) reasons.push("SIMILAR_PROFESSION");
  if (mutual>0) reasons.push("MUTUAL_CONNECTIONS");
  if (c.verified) reasons.push("VERIFIED_PROFILE");

  return {...c,score,reasons};
}

export function rankAndDiversify(
  viewer:ViewerFeatures,
  candidates:RankCandidate[],
  limit:number
) {
  const ranked=candidates.map(c=>rankCandidate(viewer,c))
    .sort((a,b)=>b.score-a.score);

  // Prevent a single profession from dominating a page.
  const professionCounts=new Map<string,number>();
  const result:RankedCandidate[]=[];

  for (const c of ranked) {
    const key=c.professionCode ?? "NONE";
    const count=professionCounts.get(key) ?? 0;
    if (count>=Math.max(3,Math.ceil(limit*0.35))) continue;
    result.push(c);
    professionCounts.set(key,count+1);
    if (result.length>=limit) break;
  }

  return result;
}

import { db } from "../../../db/pool.js";
import { validateFilters } from "./filterValidator.js";

export class SavedSearchRepository {
  async list(userId:string) {
    return (await db.query(`
      SELECT id,name,query_text,filters,alerts_enabled,created_at,updated_at
      FROM countrylinqa_saved_searches
      WHERE user_id=$1
      ORDER BY updated_at DESC
      LIMIT 100
    `,[userId])).rows;
  }

  async create(userId:string,input:any) {
    const filters=validateFilters(input.filters ?? {});
    return (await db.query(`
      INSERT INTO countrylinqa_saved_searches(
        user_id,name,query_text,filters,alerts_enabled
      ) VALUES($1,$2,$3,$4::jsonb,$5)
      RETURNING *
    `,[
      userId,
      String(input.name ?? "").trim().slice(0,80),
      input.query ? String(input.query).trim().slice(0,80) : null,
      JSON.stringify(filters),
      Boolean(input.alertsEnabled)
    ])).rows[0];
  }

  async remove(userId:string,id:string) {
    await db.query(
      `DELETE FROM countrylinqa_saved_searches WHERE id=$1 AND user_id=$2`,
      [id,userId]
    );
  }
}
