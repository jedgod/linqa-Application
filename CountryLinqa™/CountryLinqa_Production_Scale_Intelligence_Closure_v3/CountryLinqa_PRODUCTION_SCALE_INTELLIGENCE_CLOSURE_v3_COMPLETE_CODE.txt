# CountryLinqa Production Rollout

## Phase 0 — disabled
Deploy schema/code with discovery kill switch available. No new traffic.

## Phase 1 — internal
Enable staff/internal accounts only. Validate dashboards, traces, Redis and workers.

## Phase 2 — 1%
Enable deterministic 1% cohort. Hold at least one observation window.
Rollback if error/latency/privacy gates breach.

## Phase 3 — 5%
Enable 5%. Validate cache hit rate, moderation, recommendation quality,
saved-search worker success and abuse telemetry.

## Phase 4 — 25%
Increase only after SLOs remain green and no privacy/security regressions exist.

## Phase 5 — 50%
Run peak load and worker backlog checks.

## Phase 6 — 100%
Only after exact-SHA physical two-country certification is PASS.

## Emergency rollback
1. globalDiscoveryOff=true
2. recommendationWorkersOff=true
3. alertsOff=true
4. worldwideOff=true if issue is global-scope specific
5. Preserve audit/telemetry.
6. Do not delete evidence.
7. Diagnose before re-enabling.

Never use a mobile client flag as the sole kill switch. Server enforcement is authoritative.

const server=process.env.SERVER_ARTIFACT;

import crypto from "node:crypto";

export function cacheKey(x:{
  userId:string; policyVersion:number; featureVersion:number;
  surface:string; filters:unknown; cursor?:string|null;
}) {
  const digest=crypto.createHash("sha256")
    .update(JSON.stringify({f:x.filters,c:x.cursor ?? null}))
    .digest("hex").slice(0,24);
  return `clq:v3:${x.userId}:p${x.policyVersion}:f${x.featureVersion}:${x.surface}:${digest}`;
}

export const TTL={
  PEOPLE:45,
  FEED:60,
  EVENTS:120,
  BUSINESSES:180,
  COMMUNITIES:180,
  SAVED_SEARCH:120
} as const;

// Versioned keys eliminate synchronous wildcard invalidation from the hot path.
// Policy/profile feature changes increment versions. Old entries expire naturally.

export function userBucket(userId:string) {
  let h=2166136261;
  for (const c of userId) h=Math.imul(h^c.charCodeAt(0),16777619);
  return (h>>>0)%100;
}

export function enabledForUser(flag:boolean,rolloutPercent:number,userId:string) {
  return flag && userBucket(userId)<Math.max(0,Math.min(100,rolloutPercent));
}

import { db } from "../../../db/pool.js";

export type CandidateQuery={
  viewerUserId:string;
  countryScope:string[]|null;
  languages:string[];
  interests:string[];
  profession?:string;
  verifiedOnly:boolean;
  onlineOnly:boolean;
  afterUserId?:string;
  limit:number;
};

export class IndexedCandidateRepository {
  async fetch(q:CandidateQuery) {
    const p:any[]=[q.viewerUserId];
    const where:string[]=["u.id <> $1","COALESCE(f.discoverable,TRUE)=TRUE"];

    if (q.countryScope) {
      p.push(q.countryScope);
      where.push(`u.verified_country_code = ANY($${p.length}::text[])`);
    }
    if (q.profession) {
      p.push(q.profession);
      where.push(`f.profession_code=$${p.length}`);
    }
    if (q.verifiedOnly) where.push("COALESCE(u.identity_verified,FALSE)=TRUE");
    if (q.onlineOnly) where.push("f.last_active_at > now()-interval '15 minutes'");
    if (q.languages.length) {
      p.push(q.languages);
      where.push(`f.languages && $${p.length}::text[]`);
    }
    if (q.interests.length) {
      p.push(q.interests);
      where.push(`f.interests && $${p.length}::text[]`);
    }
    if (q.afterUserId) {
      p.push(q.afterUserId);
      where.push(`u.id > $${p.length}::uuid`);
    }

    p.push(Math.min(Math.max(q.limit,1),250));

    return (await db.query(`
      SELECT
        u.id user_id,
        u.verified_country_code country_code,
        COALESCE(u.identity_verified,FALSE) verified,
        COALESCE(f.languages,'{}') languages,
        COALESCE(f.interests,'{}') interests,
        f.profession_code,
        f.last_active_at,
        COALESCE(dp.inbound_mode,'MY_COUNTRY') inbound_mode,
        COALESCE(dp.inbound_country_codes,'{}') inbound_country_codes,
        dp.inbound_region_code,
        COALESCE(dp.discovery_paused,FALSE) discovery_paused
      FROM users u
      LEFT JOIN countrylinqa_user_discovery_features f ON f.user_id=u.id
      LEFT JOIN country_discovery_policy_v2 dp ON dp.user_id=u.id
      WHERE ${where.join(" AND ")}
      ORDER BY u.id
      LIMIT $${p.length}
    `,p)).rows;
  }
}

}

export type ModerationDecision={
  allowed:boolean;
  reason?:string;
};

export interface ModerationProvider {
  canAppearInDiscovery(userId:string):Promise<ModerationDecision>;
  canViewDiscovery(userId:string):Promise<ModerationDecision>;
}

export class CountryLinqaModerationGate {
  constructor(private provider:ModerationProvider){}

  async assertViewer(userId:string) {
    const d=await this.provider.canViewDiscovery(userId);
    if (!d.allowed) throw Object.assign(new Error("DISCOVERY_VIEW_RESTRICTED"),{status:403});
  }

  async candidateAllowed(userId:string) {
    const d=await this.provider.canAppearInDiscovery(userId);
    return d.allowed;
  }
}
