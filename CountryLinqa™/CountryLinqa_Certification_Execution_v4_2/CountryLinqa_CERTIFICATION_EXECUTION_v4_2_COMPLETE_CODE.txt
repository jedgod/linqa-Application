# What to do when the gate is red

Do not create a new architecture pack.

- Migration failure -> correct only the failing migration/repo schema assumption.
- Typecheck failure -> correct only the reported contract/import.
- DB test failure -> reproduce against dedicated test PostgreSQL and patch the failing behavior.
- Redis test failure -> preserve cache-miss fallback and privacy recheck.
- Skip detected -> remove accidental hard-coded skip/only; do not force unrelated environment-dependent tests to run without prerequisites.
- Physical failure -> preserve evidence, patch observed behavior, rebuild, generate new artifact hash, rerun failed scenario and regression matrix.
- SHA mismatch -> never edit evidence to match; rebuild/retest the exact release.

# CountryLinqa™ Certification Execution v4.2

This is NOT another architecture pack. It is the execution layer after v4.1.

It adds:
- preflight certification runner
- dedicated PostgreSQL safety guard
- CI result/evidence collector
- release readiness report
- Android physical-certification evidence template
- deterministic release gate
- rollback checklist

It intentionally does NOT add:
- Online Now/presence discovery
- another moderation service
- another CountryLinqa certification authority
- speculative iOS code

Use the repository's existing `scripts/countrylinqa-certification.mjs` as the
authoritative exact-SHA/physical evidence verifier.

# Release execution order

1. Review current `git status`.
2. Stage CountryLinqa-only changes; use `git add -p` on shared files.
3. Commit CountryLinqa separately.
4. Push the branch.
5. Run `countrylinqa-release-gate.yml`.
6. Require PostgreSQL tests to execute with zero skips.
7. Fix only failures actually produced by CI.
8. When green, create the Android release artifact.
9. Hash server and Android artifacts.
10. Run two-device/two-country Android certification.
11. Feed evidence to the existing `countrylinqa-certification.mjs`.
12. Keep iOS explicitly blocked until a real iOS project exists.
13. Approve rollout only when the repository's authoritative release gate is green.

const raw=process.env.DATABASE_URL;
if(!raw){
  console.error("DATABASE_URL missing.");
  process.exit(2);
}

const u=new URL(raw);
const db=u.pathname.replace(/^\\//,"").toLowerCase();
const host=u.hostname.toLowerCase();

const allowedByName=/(^|[_-])(test|ci)([_-]|$)/.test(db);
const explicit=process.env.LINQA_ALLOW_CERT_DB==="1";

if(!allowedByName && !explicit){
  console.error(
    `Refusing certification DB: "${db}". `+
    `Use a dedicated database whose name contains test/ci, `+
    `or explicitly set LINQA_ALLOW_CERT_DB=1 after verifying safety.`
  );
  process.exit(3);
}

if(["prod","production","linqa"].includes(db) && !explicit){
  console.error("Refusing production-like database.");
  process.exit(4);
}

console.log(`Dedicated certification database accepted: ${host}/${db}`);

import {spawnSync} from "node:child_process";
import fs from "node:fs";

function run(label,cmd,args,cwd="."){
  console.log(`\n=== ${label} ===`);
  const r=spawnSync(cmd,args,{
    cwd,stdio:"inherit",shell:process.platform==="win32",
    env:process.env
  });
  if((r.status ?? 1)!==0) throw new Error(`${label}_FAILED`);
}

const dirty=spawnSync("git",["status","--porcelain"],{
  encoding:"utf8",shell:process.platform==="win32"
}).stdout?.trim();

if(dirty){
  console.error("Working tree is not clean.");
  console.error("Commit/stash unrelated edits before release certification.");
  process.exit(2);
}

run("Skip audit","node",["scripts/audit-vitest-skips.mjs"]);
run("DB test gate","node",["scripts/countrylinqa-db-test-gate.mjs"]);

if(fs.existsSync("server/package.json")){
  run("Server build","npm",["run","build"],"server");
  run("Server tests","npm",["test"],"server");
}
if(fs.existsSync("clients/mobile/package.json")){
  run("Mobile typecheck","npm",["run","typecheck"],"clients/mobile");
}

console.log("\nCOUNTRYLINQA_CERT_PREFLIGHT=PASS");

import {spawnSync} from "node:child_process";

function run(cmd,args,cwd="."){
  const r=spawnSync(cmd,args,{
    cwd,stdio:"inherit",shell:process.platform==="win32",
    env:{...process.env,LINQA_REQUIRE_DB_TESTS:"1"}
  });
  if((r.status ?? 1)!==0) process.exit(r.status ?? 1);
}

run("node",["scripts/assert-dedicated-test-db.mjs"]);
run("node",["scripts/audit-vitest-skips.mjs"]);
run("node",["scripts/countrylinqa-db-test-gate.mjs"]);

// Keep existing repo certification script authoritative.
run("node",["scripts/countrylinqa-certification.mjs","--software-only"]);

console.log("COUNTRYLINQA_SOFTWARE_CERTIFICATION=PASS");

import fs from "node:fs";
import {execFileSync} from "node:child_process";

const sha=execFileSync("git",["rev-parse","HEAD"],{encoding:"utf8"}).trim();
const status=execFileSync("git",["status","--porcelain"],{encoding:"utf8"}).trim();

const report={
  generatedAt:new Date().toISOString(),
  gitSha:sha,
  cleanWorktree:status.length===0,
  software:{
    serverTests:process.env.CLQ_SERVER_TESTS ?? "UNKNOWN",
    dbIntegration:process.env.CLQ_DB_TESTS ?? "UNKNOWN",
    redisTests:process.env.CLQ_REDIS_TESTS ?? "UNKNOWN",
    mobileTypecheck:process.env.CLQ_MOBILE_TYPECHECK ?? "UNKNOWN",
    workflow:process.env.CLQ_WORKFLOW ?? "NOT_RUN"
  },
  physical:{
    twoCountryAndroid:process.env.CLQ_ANDROID_PHYSICAL ?? "PENDING",
    ios:process.env.CLQ_IOS_PHYSICAL ?? "BLOCKED_NO_IOS_PROJECT"
  }
};

report.releaseReady =
  report.cleanWorktree &&
  Object.values(report.software).every(x=>x==="PASS") &&
  report.physical.twoCountryAndroid==="PASS";

fs.mkdirSync("certification/countrylinqa",{recursive:true});
fs.writeFileSync(
  "certification/countrylinqa/release-readiness.json",
  JSON.stringify(report,null,2)
);
console.log(JSON.stringify(report,null,2));
if(!report.releaseReady) process.exitCode=5;
