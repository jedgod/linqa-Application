Take a completed G0-G5 implementation and prevent GA until the exact release candidate
has passed dependency security, Expo 57 migration verification, physical-device tests,

1. No environment flag can turn a release gate green.
2. Evidence must bind to exact Git SHA, artifact SHA-256, release version, environment, and producer.
3. Required physical-device tests may not be satisfied by simulator/emulator evidence.
4. Skipped, todo, mock-only, or fake-provider results cannot satisfy release-blocking checks.
5. Evidence expires according to policy.
6. A code change after security review creates a new RC and invalidates affected evidence.
7. GA promotion is automatic only after all exact-revision checks pass and human approvals exist.

export type AdvisorySeverity = "low" | "moderate" | "high" | "critical";

export interface Advisory {
  id: string;
  packageName: string;
  severity: AdvisorySeverity;
  direct: boolean;
  productionReachable: boolean;
  patchedVersion?: string;
  findingUrl?: string;
  acceptedRiskId?: string;
}

export interface AcceptedRisk {
  id: string;
  advisoryIds: string[];
  approvedBySecurity: boolean;
  expiresAt: string;
  compensatingControls: string[];
  rationale: string;
}

export function evaluateDependencySecurity(
  advisories: Advisory[],
  acceptedRisks: AcceptedRisk[],
  now = new Date(),
) {
  const validRisk = new Map<string, AcceptedRisk>();
  for (const risk of acceptedRisks) {
    if (
      risk.approvedBySecurity &&
      Date.parse(risk.expiresAt) > now.getTime() &&
      risk.compensatingControls.length > 0
    ) {
      validRisk.set(risk.id, risk);
    }
  }

  const blockers = advisories.filter(a => {
    if (!a.productionReachable) return false;
    if (!["critical", "high"].includes(a.severity)) return false;
    if (!a.acceptedRiskId) return true;
    const risk = validRisk.get(a.acceptedRiskId);
    return !risk || !risk.advisoryIds.includes(a.id);
  });

  return {
    green: blockers.length === 0,
    blockers,
    totals: {
      all: advisories.length,
      critical: advisories.filter(a => a.severity === "critical").length,
      high: advisories.filter(a => a.severity === "high").length,
      productionReachable: advisories.filter(a => a.productionReachable).length,
    },
  };
}

import { EvidenceRecord, GateEvaluation } from "./types";

export interface EvidenceRequirement {
  controlId: string;
  maxAgeHours: number;
  exactRevision: boolean;
  exactArtifact: boolean;
  physicalDeviceRequired?: boolean;
  productionEquivalentRequired?: boolean;
  allowedKinds: EvidenceRecord["executionKind"][];
}

export function evaluateEvidence(
  requirement: EvidenceRequirement,
  evidence: EvidenceRecord[],
  release: { gitSha: string; artifactSha256: string; now?: Date },
): GateEvaluation {
  const now = release.now ?? new Date();
  const reasons: string[] = [];
  const matching = evidence.filter(e => e.controlId === requirement.controlId && e.result === "PASS");

  const valid = matching.filter(e => {
    if (requirement.exactRevision && e.gitSha !== release.gitSha) {
      reasons.push(`${e.id}:REVISION_MISMATCH`);
      return false;
    }

    if (requirement.exactArtifact && e.artifactSha256 !== release.artifactSha256) {
      reasons.push(`${e.id}:ARTIFACT_MISMATCH`);
      return false;
    }

    if (!requirement.allowedKinds.includes(e.executionKind)) {
      reasons.push(`${e.id}:EXECUTION_KIND_NOT_ALLOWED`);
      return false;
    }

    const produced = Date.parse(e.producedAt);
    if (!Number.isFinite(produced)) {
      reasons.push(`${e.id}:INVALID_TIMESTAMP`);
      return false;
    }

    const ageHours = (now.getTime() - produced) / 3_600_000;
    if (ageHours > requirement.maxAgeHours) {
      reasons.push(`${e.id}:EVIDENCE_EXPIRED`);
      return false;
    }

    if (e.expiresAt && Date.parse(e.expiresAt) <= now.getTime()) {
      reasons.push(`${e.id}:EVIDENCE_EXPLICITLY_EXPIRED`);
      return false;
    }

    if (requirement.physicalDeviceRequired && !e.device?.physical) {
      reasons.push(`${e.id}:PHYSICAL_DEVICE_REQUIRED`);
      return false;
    }

    if (e.test) {
      if (e.test.skipped) {
        reasons.push(`${e.id}:SKIPPED_TEST`);
        return false;
      }
      if (e.test.todo) {
        reasons.push(`${e.id}:TODO_TEST`);
        return false;
      }
      if (e.test.mockProviderUsed) {
        reasons.push(`${e.id}:MOCK_PROVIDER_USED`);
        return false;
      }
      if (requirement.productionEquivalentRequired && !e.test.productionEquivalentDependencies) {
        reasons.push(`${e.id}:NOT_PRODUCTION_EQUIVALENT`);
        return false;
      }
    }

    return true;
  });

  return {
    controlId: requirement.controlId,
    passed: valid.length > 0,
    evidenceIds: valid.map(x => x.id),
    reasons: valid.length ? [] : reasons.length ? reasons : ["NO_VALID_EVIDENCE"],
  };
}

export interface ExternalReviewEvidence {
  id: string;
  reviewType: "CRYPTO_REVIEW" | "PEN_TEST" | "PRIVACY_LEGAL";
  gitSha: string;
  artifactSha256?: string;
  completedAt: string;
  expiresAt?: string;
  approved: boolean;
  criticalOpen: number;
  highOpen: number;
  reportSha256: string;
}

export function assertExternalEvidenceBound(
  e: ExternalReviewEvidence,
  release: { gitSha: string; artifactSha256: string },
  now = new Date(),
) {
  if (e.gitSha !== release.gitSha) throw new Error("EXTERNAL_REVIEW_SHA_MISMATCH");
  if (e.artifactSha256 && e.artifactSha256 !== release.artifactSha256) {
    throw new Error("EXTERNAL_REVIEW_ARTIFACT_MISMATCH");
  }
  if (!e.approved) throw new Error("EXTERNAL_REVIEW_NOT_APPROVED");
  if (e.criticalOpen > 0) throw new Error("EXTERNAL_REVIEW_CRITICAL_OPEN");
  if (e.reviewType === "CRYPTO_REVIEW" && e.highOpen > 0) {
    throw new Error("CRYPTO_REVIEW_HIGH_OPEN");
  }
  if (e.expiresAt && Date.parse(e.expiresAt) <= now.getTime()) {
    throw new Error("EXTERNAL_REVIEW_EXPIRED");
  }
}

import { ReleaseStage } from "./types";

export interface PromotionInput {
  currentStage: ReleaseStage;
  requestedStage: ReleaseStage;
  g0g5Green: boolean;
  dependencySecurityGreen: boolean;
  physicalIosGreen: boolean;
  physicalAndroidGreen: boolean;
  exactRevisionExternalEvidenceGreen: boolean;
  canaryGreen: boolean;
  formalApprovalsGreen: boolean;
}

const ORDER: ReleaseStage[] = [
  "DEVELOPMENT",
  "RC_CANDIDATE",
  "RC_FROZEN",
  "CANARY",
  "LIMITED_AVAILABILITY",
  "GA",
];

export function assertPromotionAllowed(i: PromotionInput) {
  const current = ORDER.indexOf(i.currentStage);
  const requested = ORDER.indexOf(i.requestedStage);

  if (current < 0 || requested < 0 || requested !== current + 1) {
    throw new Error("INVALID_RELEASE_STAGE_TRANSITION");
  }

  if (requested >= ORDER.indexOf("RC_FROZEN") && !i.dependencySecurityGreen) {
    throw new Error("PROMOTION_BLOCKED_DEPENDENCY_SECURITY");
  }

  if (requested >= ORDER.indexOf("CANARY")) {
    if (!i.g0g5Green) throw new Error("PROMOTION_BLOCKED_G0_G5");
    if (!i.physicalIosGreen || !i.physicalAndroidGreen) {
      throw new Error("PROMOTION_BLOCKED_DEVICE_CERTIFICATION");
    }
    if (!i.exactRevisionExternalEvidenceGreen) {
      throw new Error("PROMOTION_BLOCKED_EXTERNAL_EVIDENCE");
    }
  }

  if (i.requestedStage === "GA") {
    if (!i.canaryGreen) throw new Error("PROMOTION_BLOCKED_CANARY");
    if (!i.formalApprovalsGreen) throw new Error("PROMOTION_BLOCKED_APPROVALS");
  }
}

export interface ReleaseEvidenceManifest {
  releaseVersion: string;
  gitSha: string;
  artifactSha256: string;
  evidence: Array<{ id: string; sha256: string }>;
  generatedAt: string;
  manifestSha256?: string;
}

export function canonicalManifestPayload(m: ReleaseEvidenceManifest) {
  return JSON.stringify({
    releaseVersion: m.releaseVersion,
    gitSha: m.gitSha,
    artifactSha256: m.artifactSha256,
    evidence: [...m.evidence].sort((a,b) => a.id.localeCompare(b.id)),
    generatedAt: m.generatedAt,
  });
}

export function finalizeManifest(m: ReleaseEvidenceManifest) {
  return {
    ...m,
    manifestSha256: crypto
      .createHash("sha256")
      .update(canonicalManifestPayload(m))
      .digest("hex"),
  };
}

import fs from "node:fs/promises";

const RELEASE_BLOCKING_PATTERNS = [
  /\b(it|test|describe)\.skip\b/g,
  /\b(it|test)\.todo\b/g,
  /\bskipIf\s*\(/g,
  /\bLINQA_.*_REAL\s*===?\s*["']1["']/g,
  /\bmockProviderUsed\s*:\s*true\b/g,
  /\bFake[A-Z][A-Za-z0-9_]*(Provider|Client|Repository|Service)\b/g,
];

export interface Detection {
  file: string;
  line: number;
  pattern: string;
  text: string;
}

export async function detectForbiddenReleasePatterns(files: string[]): Promise<Detection[]> {
  const hits: Detection[] = [];

  for (const file of files) {
    const content = await fs.readFile(file, "utf8");
    const lines = content.split(/\r?\n/);

    lines.forEach((text, index) => {
      for (const pattern of RELEASE_BLOCKING_PATTERNS) {
        pattern.lastIndex = 0;
        if (pattern.test(text)) {
          hits.push({
            file,
            line: index + 1,
            pattern: pattern.toString(),
            text: text.trim(),
          });
        }
      }
    });
  }

  return hits;
}

export interface RawTestCase {
  name: string;
  status: "passed" | "failed" | "skipped" | "todo";
  durationMs?: number;
  tags?: string[];
}

export interface NormalizedTestSummary {
  total: number;
  passed: number;
  failed: number;
  skipped: number;
  todo: number;
  releaseBlockingSkipped: string[];
}

export function normalizeTests(cases: RawTestCase[]): NormalizedTestSummary {
  const releaseBlockingSkipped = cases
    .filter(t =>
      (t.status === "skipped" || t.status === "todo") &&
      t.tags?.includes("release-blocking")
    )
    .map(t => t.name);

  return {
    total: cases.length,
    passed: cases.filter(x => x.status === "passed").length,
    failed: cases.filter(x => x.status === "failed").length,
    skipped: cases.filter(x => x.status === "skipped").length,
    todo: cases.filter(x => x.status === "todo").length,
    releaseBlockingSkipped,
  };
}

export function assertReleaseTestsComplete(summary: NormalizedTestSummary) {
  if (summary.failed > 0) throw new Error("RELEASE_TEST_FAILURES");
  if (summary.releaseBlockingSkipped.length > 0) {
    throw new Error(
      `RELEASE_BLOCKING_TESTS_SKIPPED:${summary.releaseBlockingSkipped.join(",")}`
    );
  }
}

import fs from "node:fs/promises";
import path from "node:path";

const roots = process.argv.slice(2);
if (!roots.length) roots.push("tests");

const patterns = [
  ["SKIP", /\b(it|test|describe)\.skip\b/],
  ["TODO", /\b(it|test)\.todo\b/],
  ["SKIP_IF", /\bskipIf\s*\(/],
  ["FAKE_PROVIDER", /\bFake[A-Z][A-Za-z0-9_]*(Provider|Client|Repository|Service)\b/],
  ["REAL_FLAG", /\bLINQA_.*_REAL\b/]
];

const hits = [];

async function walk(p) {
  const stat = await fs.stat(p);
  if (stat.isDirectory()) {
    for (const entry of await fs.readdir(p)) {
      await walk(path.join(p, entry));
    }
    return;
  }

  if (!/\.(ts|tsx|js|mjs|kt|swift)$/.test(p)) return;
  const lines = (await fs.readFile(p, "utf8")).split(/\r?\n/);
  lines.forEach((line, i) => {
    patterns.forEach(([name, rx]) => {
      if (rx.test(line)) hits.push({ file:p, line:i+1, type:name, text:line.trim() });
    });
  });
}

for (const r of roots) {
  try { await walk(r); } catch {}
}

if (hits.length) {
  console.error(JSON.stringify(hits, null, 2));
  process.exit(2);
}

console.log("No forbidden release-test patterns detected");

import fs from "node:fs/promises";
import path from "node:path";

const root = process.env.LINQA_EVIDENCE_ROOT ?? "release-evidence/current";
const gitSha = process.env.GIT_SHA;
const artifactSha = process.env.ARTIFACT_SHA256;
const releaseVersion = process.env.RELEASE_VERSION;

if (!gitSha || !artifactSha || !releaseVersion) {
  throw new Error("GIT_SHA, ARTIFACT_SHA256 and RELEASE_VERSION are required");
}

const failures = [];

async function walk(p) {
  for (const entry of await fs.readdir(p, { withFileTypes:true })) {
    const full = path.join(p, entry.name);
    if (entry.isDirectory()) await walk(full);
    else if (entry.name.endsWith(".json")) {
      try {
        const x = JSON.parse(await fs.readFile(full, "utf8"));
        if (x.gitSha && x.gitSha !== gitSha) failures.push(`${full}:gitSha`);
        if (x.releaseVersion && x.releaseVersion !== releaseVersion) failures.push(`${full}:releaseVersion`);
        if (x.artifactSha256 && x.artifactSha256 !== artifactSha) failures.push(`${full}:artifactSha256`);
      } catch {}
    }
  }
}

await walk(root);

if (failures.length) {
  console.error("Evidence binding failures:", failures);
  process.exit(2);
}

console.log("All discovered evidence is bound to the current release identity");
