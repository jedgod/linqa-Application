# Linqa Production Gap Closure Pack v1.0

This pack implements the highest-priority gaps identified in the Linqa audit:

- Truthful runtime capability readiness with NOT_CONFIGURED / DEGRADED / READY.
- Production SMS + email delivery provider composite with fail-closed production startup.
- Canonical signed device key-package wire format and server-side verification.
- Short-lived, action/user/device-bound, single-use signed step-up tokens.
- Production-only secure-core bridge selection with no demo fallback in production.
- Durable encrypted outbox and relay/lease/ack message path.
- Versioned WASM and UniFFI binding surfaces for Rust secure-core.
- Provider registry consolidation.
- Transactional migration ledger with checksums and advisory locking.
- Executable two-client E2E/security acceptance harness.
- Mobile/web capability gating and release feature gating.

This is an integration pack designed to merge into the existing Linqa monorepo. Adapt import aliases and ORM/DB wrappers to the repository's existing conventions.

# Linqa Release Gates

## G0 Truthfulness
No required production capability may report READY from database health alone. Production must not start with development providers.

## G1 Identity
Phone verification, device enrollment, recovery-email verification, signed key package validation, transparency append/proof, and cryptographic single-use step-up must pass.

## G2 1:1 Secure Messaging
Two fresh clients must authenticate, enroll, publish/verify keys, encrypt locally, persist durable outbox, relay ciphertext only, wake/lease, decrypt locally and ACK.

Must survive: sender restart, receiver restart, server restart, offline recipient, websocket loss, duplicate submission.

Must reject: replay, ciphertext tamper, sender/device substitution, expired/revoked package, unsupported version/suite, plaintext message API.

## G3 Native
G2 must pass Web<->Web plus Web<->one native platform using UniFFI secure-core and OS/hardware-protected device keys.

## G4 Provider/Migration
Provider registry consolidation and migration ledger clean-install/upgrade tests pass.

## G5 GA External Assurance
Independent crypto review, penetration test, provider/KMS/TLS deployment review, backup restore, multi-replica/multi-region testing, SLO paging, privacy/legal approval, staffed incident response, signed native release candidates, app-store validation, canary rollback and formal go/no-go.

export type OutboxState = "CREATED" | "ENCRYPTED" | "QUEUED" | "SENDING" | "ACCEPTED_BY_RELAY" | "DELIVERED" | "FAILED_RETRYABLE" | "FAILED_PERMANENT";

export interface MobileOutboxRow {
  messageId: string;
  accountId: string;
  conversationId: string;
  recipientDeviceId: string;
  encryptedEnvelopeB64: string;
  state: OutboxState;
  attempts: number;
  nextAttemptAt: number;
}

export interface MobileOutboxDb {
  transaction<T>(fn: () => Promise<T>): Promise<T>;
  insert(row: MobileOutboxRow): Promise<void>;
  due(limit: number): Promise<MobileOutboxRow[]>;
  update(messageId: string, patch: Partial<MobileOutboxRow>): Promise<void>;
}

export class OutboxWorker {
  constructor(private db: MobileOutboxDb, private send: (envelopeB64: string) => Promise<void>) {}
  async flush(limit = 50): Promise<void> {
    for (const row of await this.db.due(limit)) {
      try {
        await this.db.update(row.messageId, { state: "SENDING" });
        await this.send(row.encryptedEnvelopeB64);
        await this.db.update(row.messageId, { state: "ACCEPTED_BY_RELAY" });
      } catch {
        const attempts = row.attempts + 1;
        const backoff = Math.min(60_000, 1000 * 2 ** Math.min(attempts, 6));
        await this.db.update(row.messageId, { state: "FAILED_RETRYABLE", attempts, nextAttemptAt: Date.now() + backoff });
      }
    }
  }
}

import type { SecureCoreBridge } from "./SecureCoreBridge";
import { FailClosedCoreBridge } from "./SecureCoreBridge";

export async function createProductionCoreBridge(env: { mode: string; secureCoreRequired: boolean }): Promise<SecureCoreBridge> {
  if (env.mode === "production" && !env.secureCoreRequired) {
    throw new Error("Production misconfiguration: secureCoreRequired must be true");
  }
  try {
    const wasm = await import("linqa-secure-core-wasm");
    const core = new wasm.WasmSecureCore();
    return {
      version: core.version(), protection: "BROWSER_PROTECTED",
      generateKeyPackage: (x) => core.generate_key_package(x),
      verifyKeyPackage: (x) => core.verify_key_package(x),
      encryptMessage: (x) => core.encrypt_message(x),
      decryptMessage: (x) => core.decrypt_message(x),
      verifyTransparencyProof: (x) => core.verify_transparency_proof(x),
    } as SecureCoreBridge;
  } catch (e) {
    if (env.mode === "production") throw new Error(`Secure core failed to load: ${String(e)}`);
    return new FailClosedCoreBridge();
  }
}

import type { SecureCoreBridge } from "../core/SecureCoreBridge";

export interface OutboxRecord {
  messageId: string;
  conversationId: string;
  recipientUserId: string;
  recipientDeviceId: string;
  encryptedEnvelope: Uint8Array;
  state: "ENCRYPTED" | "QUEUED" | "SENDING" | "ACCEPTED_BY_RELAY" | "FAILED_RETRYABLE" | "FAILED_PERMANENT";
  attempts: number;
  nextAttemptAt: number;
}

export interface EncryptedOutboxRepository {
  nextSequence(conversationId: string, recipientDeviceId: string): Promise<number>;
  insert(record: OutboxRecord): Promise<void>;
  mark(messageId: string, state: OutboxRecord["state"], patch?: Partial<OutboxRecord>): Promise<void>;
}

export interface RelayApi {
  send(envelope: Uint8Array): Promise<{ accepted: boolean; duplicate: boolean }>;
}

export class ProductionMessagingCoordinator {
  constructor(private core: SecureCoreBridge, private outbox: EncryptedOutboxRepository, private relay: RelayApi) {}

  async send(input: { messageId: string; conversationId: string; recipientUserId: string; recipientDeviceId: string; text: string }): Promise<void> {
    const sequence = await this.outbox.nextSequence(input.conversationId, input.recipientDeviceId);
    const encrypted = await this.core.encryptMessage({
      conversationId: input.conversationId,
      recipientUserId: input.recipientUserId,
      recipientDeviceId: input.recipientDeviceId,
      plaintext: new TextEncoder().encode(input.text),
      sequence,
    });
    await this.outbox.insert({ ...input, encryptedEnvelope: encrypted, state: "ENCRYPTED", attempts: 0, nextAttemptAt: Date.now() });
    await this.flushOne(input.messageId, encrypted);
  }

  private async flushOne(messageId: string, encrypted: Uint8Array) {
    try {
      await this.outbox.mark(messageId, "SENDING");
      await this.relay.send(encrypted);
      await this.outbox.mark(messageId, "ACCEPTED_BY_RELAY");
    } catch {
      await this.outbox.mark(messageId, "FAILED_RETRYABLE", { attempts: 1, nextAttemptAt: Date.now() + 2_000 });
    }
  }
}

export interface RuntimeSnapshot { capabilities: Record<string, { state: CapabilityState; checkedAt: string; reason?: string }> }

export function requireCapability(snapshot: RuntimeSnapshot, key: string): void {
  const cap = snapshot.capabilities[key];
  if (!cap || cap.state !== "READY") throw new Error(`CAPABILITY_UNAVAILABLE:${key}:${cap?.state ?? "MISSING"}`);
}

export function isCapabilityReady(snapshot: RuntimeSnapshot, key: string): boolean {
  return snapshot.capabilities[key]?.state === "READY";
}

//! Linqa secure-core binding surface.
//! Keep cryptographic protocol details inside this crate; clients call high-level operations only.

use serde::{Deserialize, Serialize};
use thiserror::Error;

#[derive(Debug, Error)]
pub enum CoreError {
    #[error("invalid input")]
    InvalidInput,
    #[error("authentication failed")]
    AuthenticationFailed,
    #[error("replay detected")]
    ReplayDetected,
    #[error("unsupported wire version")]
    UnsupportedWireVersion,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct DeviceIdentity {
    pub device_id: String,
    pub identity_public_key_b64: String,
    pub signing_public_key_b64: String,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct EncryptRequest {
    pub conversation_id: String,
    pub recipient_user_id: String,
    pub recipient_device_id: String,
    pub plaintext: Vec<u8>,
    pub sequence: u64,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct EncryptedEnvelope {
    pub wire_version: u8,
    pub message_id: String,
    pub ciphertext_b64: String,
    pub nonce_b64: String,
    pub aad_b64: String,
    pub sender_signature_b64: String,
}

pub trait SecureCore: Send + Sync {
    fn create_device_identity(&self, device_id: &str) -> Result<DeviceIdentity, CoreError>;
    fn generate_key_package(&self, user_id: &str, device_id: &str, expires_at_ms: u64) -> Result<Vec<u8>, CoreError>;
    fn verify_key_package(&self, bytes: &[u8]) -> Result<(), CoreError>;
    fn encrypt_message(&self, req: EncryptRequest) -> Result<EncryptedEnvelope, CoreError>;
    fn decrypt_message(&self, envelope: &[u8]) -> Result<Vec<u8>, CoreError>;
    fn verify_transparency_proof(&self, proof: &[u8]) -> Result<(), CoreError>;
}

import crypto from "node:crypto";
import type { DeliveryProvider } from "../providers/delivery";

export type ChallengePurpose = "REGISTER_PHONE" | "LOGIN_PHONE" | "VERIFY_RECOVERY_EMAIL" | "RECOVER_ACCOUNT";

export interface RateLimiter {
  consume(key: string, limit: number, windowSeconds: number): Promise<boolean>;
}

export interface ChallengeStore {
  put(input: { id: string; destinationHash: string; codeHash: string; purpose: ChallengePurpose; expiresAt: Date; attempts: number }): Promise<void>;
  get(id: string): Promise<{ id: string; destinationHash: string; codeHash: string; purpose: ChallengePurpose; expiresAt: Date; attempts: number } | null>;
  incrementAttempts(id: string): Promise<void>;
  consume(id: string): Promise<void>;
}

const sha256 = (value: string) => crypto.createHash("sha256").update(value).digest("hex");

export class AuthChallengeService {
  constructor(private delivery: DeliveryProvider, private store: ChallengeStore, private limiter: RateLimiter) {}

  async issueSms(params: { e164: string; purpose: ChallengePurpose; ip: string; deviceId: string }): Promise<{ challengeId: string }> {
    await this.enforceLimits(params.e164, params.ip, params.deviceId, params.purpose);
    const challengeId = crypto.randomUUID();
    const code = crypto.randomInt(100000, 1000000).toString();
    await this.store.put({
      id: challengeId,
      destinationHash: sha256(params.e164),
      codeHash: sha256(`${challengeId}:${code}`),
      purpose: params.purpose,
      expiresAt: new Date(Date.now() + 10 * 60_000),
      attempts: 0,
    });
    await this.delivery.sendSms({
      toE164: params.e164,
      body: `Your Linqa verification code is ${code}. It expires in 10 minutes.`,
      idempotencyKey: challengeId,
    });
    return { challengeId };
  }

  async issueEmail(params: { email: string; purpose: ChallengePurpose; ip: string; deviceId: string }): Promise<{ challengeId: string }> {
    await this.enforceLimits(params.email.toLowerCase(), params.ip, params.deviceId, params.purpose);
    const challengeId = crypto.randomUUID();
    const code = crypto.randomInt(100000, 1000000).toString();
    await this.store.put({
      id: challengeId,
      destinationHash: sha256(params.email.toLowerCase()),
      codeHash: sha256(`${challengeId}:${code}`),
      purpose: params.purpose,
      expiresAt: new Date(Date.now() + 10 * 60_000),
      attempts: 0,
    });
    await this.delivery.sendEmail({
      to: params.email,
      subject: "Verify your Linqa recovery email",
      text: `Your Linqa verification code is ${code}. It expires in 10 minutes.`,
      idempotencyKey: challengeId,
    });
    return { challengeId };
  }

  async verify(params: { challengeId: string; code: string }): Promise<boolean> {
    const row = await this.store.get(params.challengeId);
    if (!row || row.expiresAt.getTime() < Date.now() || row.attempts >= 5) return false;
    const expected = Buffer.from(row.codeHash, "hex");
    const actual = Buffer.from(sha256(`${params.challengeId}:${params.code}`), "hex");
    const valid = expected.length === actual.length && crypto.timingSafeEqual(expected, actual);
    if (!valid) { await this.store.incrementAttempts(params.challengeId); return false; }
    await this.store.consume(params.challengeId);
    return true;
  }

  private async enforceLimits(identifier: string, ip: string, deviceId: string, purpose: ChallengePurpose) {
    const keys: Array<[string, number, number]> = [
      [`auth:id:${sha256(identifier)}:${purpose}`, 5, 900],
      [`auth:ip:${ip}:${purpose}`, 20, 900],
      [`auth:device:${deviceId}:${purpose}`, 10, 900],
    ];
    for (const [key, limit, window] of keys) {
      if (!(await this.limiter.consume(key, limit, window))) throw new Error("RATE_LIMITED");
    }
  }
}

import crypto from "node:crypto";
import fs from "node:fs/promises";
import path from "node:path";

export interface SqlClient {
  query(sql: string, params?: unknown[]): Promise<{ rows: any[] }>;
}

const checksum = (s: string) => crypto.createHash("sha256").update(s).digest("hex");

export async function migrate(db: SqlClient, directory: string, appVersion: string): Promise<void> {
  await db.query(`CREATE TABLE IF NOT EXISTS schema_migrations (
    version VARCHAR(200) PRIMARY KEY,
    checksum VARCHAR(64) NOT NULL,
    applied_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    execution_ms INTEGER NOT NULL,
    application_version VARCHAR(100),
    dirty BOOLEAN NOT NULL DEFAULT FALSE
  )`);
  await db.query("SELECT pg_advisory_lock(hashtext('linqa-schema-migrations'))");
  try {
    const files = (await fs.readdir(directory)).filter(f => /^\d+.*\.sql$/.test(f)).sort();
    for (const file of files) {
      const sql = await fs.readFile(path.join(directory, file), "utf8");
      const sum = checksum(sql);
      const existing = await db.query("SELECT checksum, dirty FROM schema_migrations WHERE version=$1", [file]);
      if (existing.rows[0]) {
        if (existing.rows[0].dirty) throw new Error(`Dirty migration: ${file}`);
        if (existing.rows[0].checksum !== sum) throw new Error(`Migration checksum mismatch: ${file}`);
        continue;
      }
      const start = Date.now();
      await db.query("BEGIN");
      try {
        await db.query("INSERT INTO schema_migrations(version, checksum, execution_ms, application_version, dirty) VALUES($1,$2,0,$3,true)", [file, sum, appVersion]);
        await db.query(sql);
        await db.query("UPDATE schema_migrations SET dirty=false, execution_ms=$2 WHERE version=$1", [file, Date.now() - start]);
        await db.query("COMMIT");
      } catch (e) {
        await db.query("ROLLBACK");
        throw e;
      }
    }
  } finally {
    await db.query("SELECT pg_advisory_unlock(hashtext('linqa-schema-migrations'))");
  }
}

  signatureB64: string;
}

export function canonicalKeyPackagePayload(p: Omit<LinqaKeyPackageV1, "signatureB64">): Buffer {
  return Buffer.from(JSON.stringify({
    wireVersion: p.wireVersion,
    userId: p.userId,
    deviceId: p.deviceId,
    identityKeyB64: p.identityKeyB64,
    signingKeyB64: p.signingKeyB64,
    preKeyB64: p.preKeyB64,
    pqPreKeyB64: p.pqPreKeyB64,
    algorithmSuite: p.algorithmSuite,
    createdAt: p.createdAt,
    expiresAt: p.expiresAt,
    nonceB64: p.nonceB64,
  }));
}

export interface EnrolledDeviceIdentity {
  userId: string;
  deviceId: string;
  signingPublicKeyPem: string;
  revokedAt?: Date | null;
}

export function verifyKeyPackage(pkg: LinqaKeyPackageV1, enrolled: EnrolledDeviceIdentity, now = Date.now()): void {
  if (pkg.wireVersion !== KEY_PACKAGE_WIRE_VERSION) throw new Error("UNSUPPORTED_KEY_PACKAGE_VERSION");
  if (pkg.algorithmSuite !== SUPPORTED_KEY_PACKAGE_SUITE) throw new Error("UNSUPPORTED_KEY_PACKAGE_SUITE");
  if (pkg.userId !== enrolled.userId || pkg.deviceId !== enrolled.deviceId) throw new Error("KEY_PACKAGE_DEVICE_BINDING_FAILED");
  if (enrolled.revokedAt) throw new Error("DEVICE_REVOKED");
  if (pkg.createdAt > now + 5 * 60_000) throw new Error("KEY_PACKAGE_CREATED_IN_FUTURE");
  if (pkg.expiresAt <= now) throw new Error("KEY_PACKAGE_EXPIRED");
  if (pkg.expiresAt - pkg.createdAt > 30 * 24 * 60 * 60_000) throw new Error("KEY_PACKAGE_LIFETIME_TOO_LONG");

  const { signatureB64, ...unsigned } = pkg;
  const ok = crypto.verify(
    null,
    canonicalKeyPackagePayload(unsigned),
    enrolled.signingPublicKeyPem,
    Buffer.from(signatureB64, "base64"),
  );
  if (!ok) throw new Error("KEY_PACKAGE_SIGNATURE_INVALID");
}

export interface PgLike { query(sql: string, params?: unknown[]): Promise<{ rows: any[]; rowCount?: number }> }

export class PostgresRelayRepository implements RelayRepository {
  constructor(private readonly db: PgLike) {}

  async enqueue(env: EncryptedEnvelopeV1): Promise<"INSERTED" | "DUPLICATE"> {
    const result = await this.db.query(
      `INSERT INTO message_relay(
        message_id, conversation_id, sender_user_id, sender_device_id,
        recipient_user_id, recipient_device_id, sequence, envelope
      ) VALUES($1,$2,$3,$4,$5,$6,$7,$8::jsonb)
      ON CONFLICT DO NOTHING RETURNING message_id`,
      [env.messageId, env.conversationId, env.senderUserId, env.senderDeviceId,
       env.recipientUserId, env.recipientDeviceId, env.sequence, JSON.stringify(env)],
    );
    return result.rows.length ? "INSERTED" : "DUPLICATE";
  }

  async lease(recipientUserId: string, recipientDeviceId: string, limit: number, leaseSeconds: number) {
    const leaseId = crypto.randomUUID();
    const result = await this.db.query(
      `WITH candidates AS (
         SELECT message_id FROM message_relay
          WHERE recipient_user_id=$1 AND recipient_device_id=$2
            AND (state='QUEUED' OR (state='LEASED' AND lease_expires_at < NOW()))
          ORDER BY created_at ASC
          FOR UPDATE SKIP LOCKED
          LIMIT $3
       )
       UPDATE message_relay m
          SET state='LEASED', lease_id=$4::uuid,
              lease_expires_at=NOW()+($5::text || ' seconds')::interval
         FROM candidates c
        WHERE m.message_id=c.message_id
       RETURNING m.envelope, m.lease_id`,
      [recipientUserId, recipientDeviceId, limit, leaseId, leaseSeconds],
    );
    return result.rows.map(r => ({ ...r.envelope, leaseId: r.lease_id }));
  }

  async acknowledge(messageId: string, leaseId: string, recipientDeviceId: string): Promise<boolean> {
    const result = await this.db.query(
      `UPDATE message_relay SET state='ACKNOWLEDGED', acknowledged_at=NOW(), lease_id=NULL, lease_expires_at=NULL
        WHERE message_id=$1::uuid AND lease_id=$2::uuid AND recipient_device_id=$3 AND state='LEASED'
        RETURNING message_id`,
      [messageId, leaseId, recipientDeviceId],
    );
    return result.rows.length === 1;
  }
}

export type CapabilityState = "NOT_CONFIGURED" | "DEGRADED" | "READY";

export interface DependencyHealth {
  name: string;
  state: CapabilityState;
  identity?: string;
  reason?: string;
}

export interface CapabilityHealth {
  key: string;
  state: CapabilityState;
  provider?: string;
  version?: string;
  dependencies: DependencyHealth[];
  checkedAt: string;
  lastReadyAt?: string;
  reason?: string;
}

export interface CapabilityProbe {
  readonly key: string;
  probe(): Promise<CapabilityHealth>;
}

export class CapabilityRegistry {
  private readonly probes = new Map<string, CapabilityProbe>();
  private readonly lastReady = new Map<string, string>();

  register(probe: CapabilityProbe): this {
    if (this.probes.has(probe.key)) throw new Error(`Duplicate capability probe: ${probe.key}`);
    this.probes.set(probe.key, probe);
    return this;
  }

  async snapshot(): Promise<Record<string, CapabilityHealth>> {
    const entries = await Promise.all([...this.probes.values()].map(async (probe) => {
      const result = await probe.probe();
      if (result.state === "READY") {
        const now = new Date().toISOString();
        this.lastReady.set(probe.key, now);
        result.lastReadyAt = now;
      } else {
        result.lastReadyAt = this.lastReady.get(probe.key);
      }
      return [probe.key, result] as const;
    }));
    return Object.fromEntries(entries);
  }

  async requireReady(keys: string[]): Promise<void> {
    const snapshot = await this.snapshot();
    const failures = keys.filter((key) => snapshot[key]?.state !== "READY");
    if (failures.length) {
      throw new Error(`Required capabilities not READY: ${failures.join(", ")}`);
    }
  }
}

export const capability = (
  key: string,
  state: CapabilityState,
  dependencies: DependencyHealth[] = [],
  extra: Partial<CapabilityHealth> = {},
): CapabilityHealth => ({
  key,
  state,
  dependencies,
  checkedAt: new Date().toISOString(),
  ...extra,
});

import crypto from "node:crypto";

export type SensitiveAction =
  | "PHONE_CHANGE" | "RECOVERY_EMAIL_CHANGE" | "ACCOUNT_RECOVERY"
  | "DEVICE_REVOKE" | "DATA_EXPORT" | "ACCOUNT_DELETE" | "PASSKEY_REMOVE"
  | "BACKUP_KEY_ROTATE" | "COUNTRY_OVERRIDE" | "LINQASEND_FUNDING_SOURCE_CHANGE";

export interface StepUpClaims {
  iss: "linqa";
  aud: "linqa-sensitive-actions";
  sub: string;
  deviceId: string;
  action: SensitiveAction;
  authn: "PASSKEY" | "DEVICE_PROOF";
  jti: string;
  iat: number;
  exp: number;
}

export interface StepUpConsumptionStore {
  consumeOnce(jti: string, expiresAtSeconds: number): Promise<boolean>;
}

const b64u = (b: Buffer) => b.toString("base64url");

export class StepUpService {
  constructor(private readonly privateKeyPem: string, private readonly publicKeyPem: string, private readonly consumed: StepUpConsumptionStore) {}

  issue(input: { userId: string; deviceId: string; action: SensitiveAction; authn: StepUpClaims["authn"]; ttlSeconds?: number }): string {
    const now = Math.floor(Date.now() / 1000);
    const claims: StepUpClaims = {
      iss: "linqa", aud: "linqa-sensitive-actions", sub: input.userId, deviceId: input.deviceId,
      action: input.action, authn: input.authn, jti: crypto.randomUUID(), iat: now, exp: now + Math.min(input.ttlSeconds ?? 300, 300),
    };
    const header = { alg: "EdDSA", typ: "JWT", kid: "stepup-v1" };
    const signingInput = `${b64u(Buffer.from(JSON.stringify(header)))}.${b64u(Buffer.from(JSON.stringify(claims)))}`;
    const sig = crypto.sign(null, Buffer.from(signingInput), this.privateKeyPem);
    return `${signingInput}.${b64u(sig)}`;
  }

  async verifyAndConsume(token: string, expected: { userId: string; deviceId: string; action: SensitiveAction }): Promise<StepUpClaims> {
    const parts = token.split(".");
    if (parts.length !== 3) throw new Error("STEP_UP_MALFORMED");
    const [h, p, s] = parts;
    if (!crypto.verify(null, Buffer.from(`${h}.${p}`), this.publicKeyPem, Buffer.from(s, "base64url"))) throw new Error("STEP_UP_SIGNATURE_INVALID");
    const claims = JSON.parse(Buffer.from(p, "base64url").toString("utf8")) as StepUpClaims;
    const now = Math.floor(Date.now() / 1000);
    if (claims.iss !== "linqa" || claims.aud !== "linqa-sensitive-actions") throw new Error("STEP_UP_AUDIENCE_INVALID");
    if (claims.exp <= now || claims.iat > now + 30) throw new Error("STEP_UP_EXPIRED");
    if (claims.sub !== expected.userId || claims.deviceId !== expected.deviceId || claims.action !== expected.action) throw new Error("STEP_UP_BINDING_FAILED");
    if (!(await this.consumed.consumeOnce(claims.jti, claims.exp))) throw new Error("STEP_UP_REPLAYED");
    return claims;
  }
}
