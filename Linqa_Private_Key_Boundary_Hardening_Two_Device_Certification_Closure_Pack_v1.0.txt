package com.linqa.security

import java.io.Closeable
import java.util.concurrent.atomic.AtomicBoolean

/**
 * Narrow, zeroizable secret container.
 * Never convert to String, log, serialize, or return from UI/repository APIs.
 */
class SecretBytes private constructor(private var value: ByteArray?) : Closeable {
    private val closed = AtomicBoolean(false)

    companion object {
        fun takeOwnership(bytes: ByteArray): SecretBytes = SecretBytes(bytes)
    }

    fun <T> useBytes(block: (ByteArray) -> T): T {
        check(!closed.get()) { "secret_closed" }
        return block(value ?: error("secret_missing"))
    }

    override fun close() {
        if (closed.compareAndSet(false, true)) {
            value?.fill(0)
            value = null
        }
    }
}

package com.linqa.security

import android.security.keystore.KeyGenParameterSpec
import android.security.keystore.KeyInfo
import android.security.keystore.KeyProperties
import java.security.KeyStore
import javax.crypto.KeyGenerator
import javax.crypto.SecretKey
import javax.crypto.SecretKeyFactory

class AndroidKeystoreWrapKey(
    private val alias: String = "linqa.identity.wrap.v1"
) {
    private val ks = KeyStore.getInstance("AndroidKeyStore").apply { load(null) }

    fun getOrCreate(): SecretKey {
        (ks.getKey(alias, null) as? SecretKey)?.let { return it }

        val generator = KeyGenerator.getInstance(
            KeyProperties.KEY_ALGORITHM_AES,
            "AndroidKeyStore"
        )
        generator.init(
            KeyGenParameterSpec.Builder(
                alias,
                KeyProperties.PURPOSE_ENCRYPT or KeyProperties.PURPOSE_DECRYPT
            )
                .setBlockModes(KeyProperties.BLOCK_MODE_GCM)
                .setEncryptionPaddings(KeyProperties.ENCRYPTION_PADDING_NONE)
                .setKeySize(256)
                .setRandomizedEncryptionRequired(true)
                .build()
        )
        return generator.generateKey()
    }

    fun isHardwareBacked(): Boolean = runCatching {
        val key = getOrCreate()
        val factory = SecretKeyFactory.getInstance(key.algorithm, "AndroidKeyStore")
        val info = factory.getKeySpec(key, KeyInfo::class.java) as KeyInfo
        info.isInsideSecureHardware
    }.getOrDefault(false)
}

package com.linqa.security

import android.content.Context
import android.util.Base64
import androidx.core.content.edit
import java.time.Instant
import javax.crypto.Cipher
import javax.crypto.spec.GCMParameterSpec

/**
 * Seals the enrollment identity payload with an AES-256 Android Keystore wrapping key.
 * No general-purpose unseal API exists.
 */
class DeviceIdentityStore(
    context: Context,
    private val wrapKey: AndroidKeystoreWrapKey = AndroidKeystoreWrapKey()
) {
    private val prefs = context.getSharedPreferences(
        "linqa_identity_secure_v1",
        Context.MODE_PRIVATE
    )

    fun sealEnrollmentPayload(payload: ByteArray) {
        val cipher = Cipher.getInstance("AES/GCM/NoPadding")
        cipher.init(Cipher.ENCRYPT_MODE, wrapKey.getOrCreate())

        val ciphertext = try {
            cipher.doFinal(payload)
        } finally {
            payload.fill(0)
        }

        try {
            prefs.edit(commit = true) {
                putInt("version", 1)
                putString("nonce", Base64.encodeToString(cipher.iv, Base64.NO_WRAP))
                putString("ciphertext", Base64.encodeToString(ciphertext, Base64.NO_WRAP))
                putString("createdAt", Instant.now().toString())
            }
        } finally {
            ciphertext.fill(0)
        }
    }

    /**
     * Security/core-module-only bootstrap method.
     * Do not make public.
     */
    internal fun unsealForNativeImport(): SecretBytes {
        val nonceString = requireNotNull(prefs.getString("nonce", null)) {
            "identity_nonce_missing"
        }
        val ciphertextString = requireNotNull(prefs.getString("ciphertext", null)) {
            "identity_ciphertext_missing"
        }

        val nonce = Base64.decode(nonceString, Base64.NO_WRAP)
        val ciphertext = Base64.decode(ciphertextString, Base64.NO_WRAP)

        val cipher = Cipher.getInstance("AES/GCM/NoPadding")
        cipher.init(
            Cipher.DECRYPT_MODE,
            wrapKey.getOrCreate(),
            GCMParameterSpec(128, nonce)
        )

        return try {
            SecretBytes.takeOwnership(cipher.doFinal(ciphertext))
        } finally {
            nonce.fill(0)
            ciphertext.fill(0)
        }
    }

    fun hardwareBacked(): Boolean = wrapKey.isHardwareBacked()

    fun clearSealedIdentity() {
        prefs.edit(commit = true) { clear() }
    }
}

pub fn import_enrollment_payload(
    payload_bytes: &mut [u8]
) -> Result<IdentityImportResult, String> {
    let parsed: EnrollmentPayload = serde_json::from_slice(payload_bytes)
        .map_err(|_| "invalid_identity_payload".to_string())?;

    let mut ed = B64.decode(parsed.ed25519_private_key_b64.as_bytes())
        .map_err(|_| "invalid_ed25519_private_key".to_string())?;
    let mut x = B64.decode(parsed.x25519_private_key_b64.as_bytes())
        .map_err(|_| "invalid_x25519_private_key".to_string())?;

    let result = (|| {
        let (ed_pub, x_pub) = derive_public_identity_using_existing_linqa_types(&ed, &x)?;
        let (ml_private, ml_public) = generate_or_restore_existing_ml_kem_768()?;

        let fp = identity_fingerprint(&ed_pub, &x_pub, &ml_public);

        let identity = CoreDeviceIdentity {
            version: parsed.version,
            user_id: parsed.user_id,
            device_id: parsed.device_id,
            ed25519_private: Zeroizing::new(ed.clone()),
            x25519_private: Zeroizing::new(x.clone()),
            ml_kem_768_private: Zeroizing::new(ml_private),
            ed25519_public_b64: B64.encode(&ed_pub),
            x25519_public_b64: B64.encode(&x_pub),
            ml_kem_768_public_b64: B64.encode(&ml_public),
        };

        let output = IdentityImportResult {
            identity_version: identity.version,
            ed25519_public_key_b64: identity.ed25519_public_b64.clone(),
            x25519_public_key_b64: identity.x25519_public_b64.clone(),
            ml_kem_768_public_key_b64: identity.ml_kem_768_public_b64.clone(),
            core_identity_fingerprint: fp,
        };

        *identity_slot().write() = Some(identity);
        Ok(output)
    })();

    ed.zeroize();
    x.zeroize();
    payload_bytes.zeroize();
    result
}

#[no_mangle]
pub extern "system"
fn Java_com_linqa_core_NativeCore_importDeviceIdentitySealedPayload(
    mut env: JNIEnv,
    _class: JClass,
    payload: JByteArray,
) -> jobject {

            .map_err(|_| "jni_identity_copy_failed".to_string())?;

        let imported = import_enrollment_payload(&mut bytes);

            .map_err(|_| "jni_array_length_failed".to_string())?;

            .map_err(|_| "jni_java_array_wipe_failed".to_string())?;

        let imported = imported?;

        let cls = env.find_class("com/linqa/core/NativeIdentityImportResult")
            .map_err(|_| "jni_result_class_missing".to_string())?;

        let ed = env.new_string(imported.ed25519_public_key_b64)
            .map_err(|_| "jni_ed_public_failed".to_string())?;
        let x = env.new_string(imported.x25519_public_key_b64)
            .map_err(|_| "jni_x_public_failed".to_string())?;
        let ml = env.new_string(imported.ml_kem_768_public_key_b64)
            .map_err(|_| "jni_ml_public_failed".to_string())?;
        let fp = env.new_string(imported.core_identity_fingerprint)
            .map_err(|_| "jni_fp_failed".to_string())?;

        let obj = env.new_object(
            cls,
            "(ILjava/lang/String;Ljava/lang/String;Ljava/lang/String;Ljava/lang/String;)V",

                JValue::Object(&JObject::from(ed)),
                JValue::Object(&JObject::from(x)),
                JValue::Object(&JObject::from(ml)),
                JValue::Object(&JObject::from(fp)),
            ],

        Ok(obj.into_raw())

        Ok(value) => value,
        Err(code) => {

        postRevokeEnvelopeCount: Int
    ) = record(
        "REVOKED_DEVICE_BLOCKED",
        mapOf(
            "revokedDeviceId" to revokedDeviceId,
            "postRevokeEnvelopeCount" to postRevokeEnvelopeCount

        if (postRevokeEnvelopeCount == 0) "PASS" else "FAIL"
    )

    private fun record(
        event: String,
        evidence: Map<String, Any?> = emptyMap(),
        status: String = "PASS"
    ) {
        recorder.record(
            CertificationEvent(
                runId = runId,
                gitSha = gitSha,
                deviceRole = role,
                deviceId = deviceId,
                userId = userId,
                event = event,
                status = status,
                evidence = evidence
            )
        )
    }

    with open(path,encoding="utf-8") as f:
        for line in f:
            if line.strip():
                events.append(json.loads(line))

if not events:
    raise SystemExit("FAIL: no real device evidence")

for e in events:
    if e["gitSha"] != sha:
        raise SystemExit("FAIL: device evidence SHA mismatch")
    if e["status"] != "PASS":
        raise SystemExit(f"FAIL: failed device event {e}")

roles={e["deviceRole"] for e in events}
if not {"alice","bob"} <= roles:
    raise SystemExit("FAIL: Alice and Bob evidence both required")

sends={}
receives={}
for e in events:
    if e["event"]=="SEND_ACCEPTED":
        sends[e["evidence"]["clientMessageId"]]=e
    if e["event"]=="RECEIVE_DECRYPTED":
        receives[e["evidence"]["clientMessageId"]]=e

pairs=set()
for mid,send in sends.items():
    receive=receives.get(mid)
    if not receive:
        continue
    if send["deviceRole"] == receive["deviceRole"]:
        continue
    if send["evidence"]["plaintextSha256"] != receive["evidence"]["plaintextSha256"]:
        raise SystemExit(f"FAIL: digest mismatch for {mid}")
    pairs.add((send["deviceRole"],receive["deviceRole"]))

if ("alice","bob") not in pairs:
    raise SystemExit("FAIL: Alice -> Bob not proven")
if ("bob","alice") not in pairs:
    raise SystemExit("FAIL: Bob -> Alice not proven")

for required in [
    "DUPLICATE_RETRY",
    "TAMPER_REJECTED",
    "KEY_ROTATED",
    "REVOKED_DEVICE_BLOCKED"
]:
    if not any(e["event"]==required for e in events):
        raise SystemExit(f"FAIL: {required} missing")

rotation=[
    e for e in events
    if e["event"]=="KEY_ROTATED"
    and e["evidence"].get("oldPackageRejected") is True
]
if not rotation:
    raise SystemExit("FAIL: stale key package rejection not proven")

revoked=[
    e for e in events
    if e["event"]=="REVOKED_DEVICE_BLOCKED"
    and e["evidence"].get("postRevokeEnvelopeCount")==0
]
if not revoked:
    raise SystemExit("FAIL: revoked-device isolation not proven")

offline=False
by_role=defaultdict(list)
for e in events:
    by_role[e["deviceRole"]].append(e)

for role,seq in by_role.items():
    names=[e["event"] for e in seq]
    if "OFFLINE_ENTER" not in names or "OFFLINE_EXIT" not in names:
        continue
    i=names.index("OFFLINE_ENTER")
    j=names.index("OFFLINE_EXIT",i+1)
    if any(x=="RECEIVE_DECRYPTED" for x in names[j+1:]):
        offline=True

    raise SystemExit("FAIL: offline -> reconnect -> decrypt sequence not proven")

    "status":"GREEN",
    "requirements":{
        "aliceToBob":True,
        "bobToAlice":True,
        "offlineDelivery":True,
        "duplicateRetry":True,
        "tamperRejected":True,
        "keyRotation":True,
        "revokedDeviceIsolation":True
    }
}

    "w",
    encoding="utf-8"

    json.dump(summary,f,indent=2,sort_keys=True)

    })

}

mp=os.path.join(evdir,"manifest.json")
with open(mp,"w",encoding="utf-8") as f:
