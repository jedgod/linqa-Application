import { twMerge } from "tailwind-merge";

export function cn(...inputs: ClassValue[]) {

import React from "react";
import {
  Home, FolderClosed, Plus, Share2, MoreHorizontal, ArrowLeft, Search,
  Menu, Info, Bell, FileText, FileSpreadsheet, Folder, LockKeyhole
} from "lucide-react";
import { cn } from "@/lib/cn";

export function PhoneFrame({children}:{children:React.ReactNode}) {
  return (
    <div className="relative mx-auto flex h-[844px] w-full max-w-[390px] flex-col
                    overflow-hidden rounded-[44px] border border-white/10
                    bg-vault-bg shadow-phone">
      <div className="flex h-11 items-center justify-between px-6 text-[12px] font-semibold">
        <span>9:41</span><span>▮▮▮  ⌁  ▰</span>
      </div>
      <div className="min-h-0 flex-1">{children}</div>
      <div className="pointer-events-none absolute bottom-2 left-1/2 h-1 w-32
                      -translate-x-1/2 rounded-full bg-white/80"/>
    </div>
  );
}

export function Card({children,className}:{children:React.ReactNode,className?:string}) {
  return (
    <div className={cn("rounded-2xl border border-white/[0.06] bg-white/[0.045]",className)}>
      {children}
    </div>
  );
}

export function PageHeader({
  title,subtitle,back,search,menu,info,bell
}:{
  title:string;subtitle?:string;back?:boolean;search?:boolean;menu?:boolean;
  info?:boolean;bell?:boolean
}) {
  return (
    <div className="px-5 pt-3">
      <div className="flex items-center gap-3">
        {back && <button className="grid h-10 w-10 place-items-center rounded-full">
          <ArrowLeft className="h-5 w-5"/>
        </button>}
        <div className="min-w-0 flex-1">
          <h1 className="truncate text-[22px] font-semibold">{title}</h1>
          {subtitle && <p className="mt-1 text-xs text-slate-400">{subtitle}</p>}
        </div>
        {search && <Search className="h-5 w-5"/>}
        {menu && <Menu className="h-5 w-5"/>}
        {info && <Info className="h-5 w-5"/>}
        {bell && <Bell className="h-5 w-5"/>}
      </div>
    </div>
  );
}

export function ProgressBar({value}:{value:number}) {
  return (
    <div className="h-1.5 overflow-hidden rounded-full bg-white/10">
      <div className="h-full rounded-full bg-blue-500"
           style={{width:`${Math.max(0,Math.min(value,100))}%`}}/>
    </div>
  );
}

export function FileIcon({type}:{type:"pdf"|"doc"|"sheet"|"folder"}) {
  if(type==="folder") {
    return <Folder className="h-8 w-8 fill-amber-400 text-amber-400"/>;
  }
  if(type==="sheet") {
    return <span className="grid h-10 w-9 place-items-center rounded-md bg-green-500">
      <FileSpreadsheet className="h-5 w-5"/>
    </span>;
  }
  return <span className={cn("grid h-10 w-9 place-items-center rounded-md",
    type==="pdf"?"bg-red-500":"bg-blue-500")}>
    <FileText className="h-5 w-5"/>
  </span>;
}

export function BottomNav({active}:{active:"home"|"files"|"share"|"more"}) {
  const items = [
    ["home","Home",Home],["files","Files",FolderClosed],["plus","",Plus],
    ["share","Share",Share2],["more","More",MoreHorizontal]
  ] as const;
  return (
    <div className="absolute inset-x-0 bottom-0 border-t border-white/10
                    bg-[#081523]/95 px-3 pb-5 pt-2 backdrop-blur">
      <div className="grid grid-cols-5 items-end">
        {items.map(([id,label,Icon])=>{
          if(id==="plus") return <button key={id}
            className="mx-auto grid h-14 w-14 -translate-y-4 place-items-center
                       rounded-full bg-blue-600 ring-4 ring-[#081523] shadow-glow">
            <Icon className="h-7 w-7"/>
          </button>;
          const on = id===active;
          return <button key={id}
            className={cn("flex flex-col items-center gap-1 text-[10px]",
              on?"text-blue-400":"text-slate-400")}>
            <Icon className="h-[18px] w-[18px]"/><span>{label}</span>
          </button>
        })}
      </div>
    </div>
  );
}

export function Screen({
  children,active,bottomNav=true
}:{
  children:React.ReactNode;active:"home"|"files"|"share"|"more";bottomNav?:boolean
}) {
  return (
    <div className="relative flex h-full flex-col">
      <main className="flex-1 overflow-y-auto pb-24">{children}</main>
      {bottomNav && <BottomNav active={active}/>}
    </div>
  );
}

export function VaultLogo() {
  return <div className="grid h-9 w-9 place-items-center rounded-xl bg-blue-600/20
                         ring-1 ring-blue-500/30">
    <LockKeyhole className="h-5 w-5 text-blue-400"/>
  </div>
}

SECURITY UI RULES
================================================
SOCIAL -> VAULT
- Save to Vault is user initiated.
- File passes Intake Gateway.
- File is quarantined.
- File is scanned.
- User confirms save.
- Independent Vault object is created.

VAULT -> SOCIAL
- No direct social browsing of Vault.
- UI says "Export to Social (Copy)".
- Explicit confirmation.
- Step-up auth when required.
- Backend policy can deny business export.
- New social object is created.
- Original Vault object remains private.

PRIVATE VAULT
- biometric / passkey / PIN
- short auto-lock
- no background previews while locked
- notification redaction

// features/vault/api/client.ts
export async function vaultFetch<T>(path:string, init:RequestInit={}):Promise<T>{
  const response=await fetch(path,{
    ...init,
    credentials:"include",
    headers:{
      "Content-Type":"application/json",
      ...(init.headers??{})
    }
  });

  if(!response.ok){
    const body=await response.json().catch(()=>({}));
    throw new Error(body.error ?? `VAULT_HTTP_${response.status}`);
  }

  return response.json();
}

// features/vault/hooks/useScanStatus.ts
"use client";

import { useEffect,useState } from "react";

export function useScanStatus(uploadId:string){
  const [state,setState]=useState("SCANNING");
  const [progress,setProgress]=useState(0);

  useEffect(()=>{
    let stopped=false;

    const run=async()=>{
      while(!stopped){
        const r=await fetch(`/v1/vault/uploads/${uploadId}/status`,{
          credentials:"include"
        });
        if(r.ok){
          const d=await r.json();
          setState(d.state);
          setProgress(d.progress??0);
          if(["AWAITING_CONFIRMATION","FAILED","COMPLETED"].includes(d.state)) break;
        }
        await new Promise(x=>setTimeout(x,1500));
      }
    };

    run();
    return()=>{stopped=true};
  },[uploadId]);

  return {state,progress};
}

FINAL IMPLEMENTATION PRINCIPLE
================================================
The mobile UI can initiate operations, but it cannot enforce the Vault trust
boundary by itself. Backend IAM, authorization, KMS policy, object-storage
policy, tenant policy, audit, and the Export Gateway remain authoritative.

SOCIAL CAN SEND INFORMATION INTO THE VAULT WHEN THE USER REQUESTS IT.
VAULT CANNOT EXPOSE INFORMATION OUT TO SOCIAL WITHOUT EXPLICIT AUTHORIZATION.
