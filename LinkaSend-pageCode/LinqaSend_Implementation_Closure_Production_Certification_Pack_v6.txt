// This is an adapter pattern, not a hard-coded dependency.
// Provider SDK/API versions must be pinned and verified at integration time.

export class StripeFundingAdapter
  implements FundingProviderAdapter
{
  readonly providerId =
    "stripe";

  constructor(
    private readonly client:
      StripeCompatibleClient,
  ) {}

  async authorize(
    input:
      FundingAuthorizationRequest,
  ): Promise<FundingAuthorizationResult> {
    const result =
      await this.client.createPaymentIntent(
        {
          amount:
            safeBigIntToProviderInteger(
              input.amountMinor,
            ),

          currency:
            input.currency.toLowerCase(),

          payment_method:
            input.fundingSourceToken,

          confirmation_method:
            "automatic",

          confirm:
            true,

          metadata: {
            linqa_transfer_id:
              input.transferId,
          },

          return_url:
            input.returnUrl,
        },
        {
          idempotencyKey:
            input.idempotencyKey,
        },
      );

    if (
      result.status ===
      "succeeded"
    ) {
      return {
        status:
          "AUTHORIZED",
        reference:
          result.id,
      };
    }

    if (
      result.status ===
        "requires_action" ||
      result.status ===
        "requires_source_action"
    ) {
      return {
        status:
          "REQUIRES_ACTION",
        reference:
          result.id,
        clientAction:
          result.next_action,
      };
    }

    if (
      result.status ===
      "requires_payment_method"
    ) {
      return {
        status:
          "DECLINED",
        reference:
          result.id,
        failureCode:
          "PAYMENT_METHOD_DECLINED",
      };
    }

    return {
      status: "FAILED",
      reference:
        result.id,
      failureCode:
        `UNHANDLED_PROVIDER_STATUS:${result.status}`,
    };
  }

  async refund(input: {
    reference: string;
    amountMinor: bigint;
    currency: string;
    idempotencyKey: string;
  }) {
    const refund =
      await this.client.createRefund(
        {
          payment_intent:
            input.reference,

          amount:
            safeBigIntToProviderInteger(
              input.amountMinor,
            ),
        },
        {
          idempotencyKey:
            input.idempotencyKey,
        },
      );

    return {
      status:
        refund.status ===
        "succeeded"
          ? "REFUNDED"
          : "FAILED",

      reference:
        refund.id,
    } as const;
  }
}

function safeBigIntToProviderInteger(
  value: bigint,
) {
  if (
    value >
    BigInt(
      Number.MAX_SAFE_INTEGER,
    )
  ) {
    throw new Error(
      "PROVIDER_AMOUNT_EXCEEDS_SAFE_INTEGER",
    );
  }

  return Number(value);
}

export class PgBaseRepository {
  constructor(
    protected readonly db:
      PgPool,
  ) {}

  async tx<T>(
    fn: (
      client: PgClient,
    ) => Promise<T>,
  ) {
    const client =
      await this.db.connect();

    try {
      await client.query(
        "BEGIN",
      );

      const result =
        await fn(client);

      await client.query(
        "COMMIT",
      );

      return result;
    } catch (error) {
      await client.query(
        "ROLLBACK",
      );

      throw error;
    } finally {
      client.release();
    }
  }
}

export class PgLedgerRepository
  extends PgBaseRepository
{
  async postBalancedJournal(args: {
    referenceType: string;
    referenceId: string;
    description: string;

    postings: Array<{
      accountId: string;
      direction:
        | "DEBIT"
        | "CREDIT";
      amountMinor: bigint;
      currency: string;
    }>;
  }) {
    assertBalancedByCurrency(
      args.postings,
    );

    return this.tx(
      async (client) => {
        const journal =
          await client.query(
            `
            INSERT INTO linqasend_ledger_journals_v3 (
              reference_type,
              reference_id,
              description
            )
            VALUES ($1,$2,$3)
            RETURNING *
            `,
            [
              args.referenceType,
              args.referenceId,
              args.description,
            ],
          );

        const journalId =
          journal.rows[0].id;

        for (
          const posting
          of args.postings
        ) {
          await client.query(
            `
            INSERT INTO linqasend_ledger_postings_v3 (
              journal_id,
              account_id,
              direction,
              amount_minor,
              currency
            )
            VALUES ($1,$2,$3,$4,$5)
            `,
            [
              journalId,
              posting.accountId,
              posting.direction,
              posting.amountMinor.toString(),
              posting.currency,
            ],
          );
        }

        return (
          journal.rows[0]
        );
      },
    );
  }
}

function assertBalancedByCurrency(
  postings: Array<{
    direction:
      | "DEBIT"
      | "CREDIT";
    amountMinor: bigint;
    currency: string;
  }>,
) {
  const currencies =
    new Set(
      postings.map(
        (x) =>
          x.currency,
      ),
    );

  for (
    const currency
    of currencies
  ) {
    let debits = 0n;
    let credits = 0n;

    for (
      const posting
      of postings
    ) {
      if (
        posting.currency !==
        currency
      ) {
        continue;
      }

      if (
        posting.direction ===
        "DEBIT"
      ) {
        debits +=
          posting.amountMinor;
      } else {
        credits +=
          posting.amountMinor;
      }
    }

    if (
      debits !==
      credits
    ) {
      throw new Error(
        `UNBALANCED_LEDGER:${currency}`,
      );
    }
  }
}

import {
  createHash,
} from "crypto";

export class CsvBankStatementParser
  implements
    BankStatementParser
{
  async parse(
    input:
      Buffer | string,
  ) {
    const text =
      Buffer.isBuffer(
        input,
      )
        ? input.toString(
            "utf8",
          )
        : input;

    const lines =
      text
        .split(/\r?\n/)
        .filter(Boolean);

    const [
      headerLine,
      ...rows
    ] = lines;

    const headers =
      headerLine.split(",");

    return rows.map(
      (line) => {
        const cells =
          splitCsvLine(
            line,
          );

        const row =
          Object.fromEntries(
            headers.map(
              (h, i) => [
                h.trim(),
                cells[i]?.trim(),
              ],
            ),
          );

        const rawRecordHash =
          createHash("sha256")
            .update(line)
            .digest("hex");

        return {
          sourceId:
            row.source_id,

          externalRecordId:
            row.transaction_id,

          accountReference:
            row.account_ref,

          bookingDate:
            row.booking_date,

          valueDate:
            row.value_date ||
            undefined,

          amountMinor:
            parseDecimalToMinor(
              row.amount,
              row.currency,
            ),

          currency:
            row.currency.toUpperCase(),

          direction:
            row.direction.toUpperCase() as
              | "DEBIT"
              | "CREDIT",

          bankReference:
            row.bank_reference ||
            undefined,

          counterpartyReference:
            row.counterparty_reference ||
            undefined,

          rawRecordHash,
        };
      },
    );
  }
}

function splitCsvLine(
  line: string,
) {
  // Replace with production CSV parser library for quoted/escaped CSV.
  return line.split(",");
}

export class BankStatementIngestor {
  constructor(
    private readonly records:
      ExternalSettlementRecordRepository,

    private readonly dedupe:
      IngestionDedupeService,
  ) {}

  async ingest(args: {
    sourceType:
      "BANK_STATEMENT";
    sourceId: string;

    records:
      NormalizedBankStatementRecord[];
  }) {
    let inserted = 0;
    let duplicates = 0;

    for (
      const record
      of args.records
    ) {
      const duplicate =
        await this.dedupe.exists({
          sourceType:
            args.sourceType,
          sourceId:
            args.sourceId,
          externalRecordId:
            record.externalRecordId,
        });

      if (duplicate) {
        duplicates++;
        continue;
      }

      await this.records.insert({
        sourceType:
          args.sourceType,

        sourceId:
          args.sourceId,

        externalRecordId:
          record.externalRecordId,

        accountReference:
          record.accountReference,

        amountMinor:
          record.amountMinor,

        currency:
          record.currency,

        direction:
          record.direction,

        valueDate:
          record.valueDate,

        externalTimestamp:
          new Date(
            record.bookingDate,
          ),

        rawPayloadHash:
          record.rawRecordHash,
      });

      inserted++;
    }

    return {
      inserted,
      duplicates,
    };
  }
}

export class PhysicalTransactionCertifier {
  constructor(
    private readonly runner:
      LiveTransferRunner,

    private readonly evidence:
      EvidenceBinderService,

    private readonly repo:
      CertificationEvidenceRepository,
  ) {}

  async run(
    test:
      PhysicalTransactionCase,
    context: {
      gitSha: string;
      buildId: string;
    },
  ): Promise<
    PhysicalTransactionEvidence
  > {
    const startedAt =
      new Date();

    const base = {
      caseId:
        test.caseId,

      gitSha:
        context.gitSha,

      buildId:
        context.buildId,

      sourceCountry:
        test.sourceCountry,

      destinationCountry:
        test.destinationCountry,

      startedAt:
        startedAt.toISOString(),

      result:
        "NOT_RUN" as
        CertificationResult,

      checks: {} as
        Record<
          string,
          boolean
        >,

      evidenceFiles:
        [] as string[],
    };

    try {
      if (
        test.expectedCapability ===
        "BLOCKED"
      ) {
        const result =
          await this.runner.tryBlockedTransfer(
            test,
          );

        const passed =
          result.blocked ===
          true;

        const evidence = {
          ...base,

          completedAt:
            new Date()
              .toISOString(),

          result:
            passed
              ? "PASS"
              : "FAIL",

          checks: {
            capabilityBlocked:
              passed,
            noFundingOccurred:
              result.noFundingOccurred,
            noLedgerPosting:
              result.noLedgerPosting,
            noPayout:
              result.noPayout,
          },

          failureCode:
            passed
              ? undefined
              : "EXPECTED_BLOCK_NOT_ENFORCED",
        };

        await this.repo.save(
          evidence,
        );

        return evidence;
      }

      const run =
        await this.runner.executeLiveTransfer(
          test,
        );

      const checks = {
        transferCreated:
          !!run.transferId,

        securityPassed:
          run.securityPassed,

        fundingAuthorized:
          test.realFundingRequired
            ? run.fundingAuthorized
            : true,

        ledgerBalanced:
          run.ledgerBalanced,

        recipientCredited:
          run.recipientCredited,

        payoutConfirmed:
          test.realPayoutRequired
            ? run.payoutConfirmed
            : true,

        reconciliationMatched:
          run.reconciliationMatched,

        exactSha:
          run.serverGitSha ===
          context.gitSha,

        sameBuild:
          run.serverBuildId ===
          context.buildId,
      };

      const allPass =
        Object.values(
          checks,
        ).every(Boolean);

      const evidence = {
        ...base,

        transferId:
          run.transferId,

        providerReference:
          run.providerReference,

        completedAt:
          new Date()
            .toISOString(),

        result:
          allPass
            ? "PASS"
            : "FAIL",

        checks,

        evidenceFiles:
          run.evidenceFiles,
      };

      await this.repo.save(
        evidence,
      );

      return evidence;
    } catch (error) {
      const evidence = {
        ...base,

        completedAt:
          new Date()
            .toISOString(),

        result:
          "FAIL" as const,

        failureCode:
          error instanceof Error
            ? error.message
            : "UNKNOWN_FAILURE",
      };

      await this.repo.save(
        evidence,
      );

      return evidence;
    }
  }
}

export type ReconciliationFailureMode =
  | "MISSING_EXTERNAL"
  | "MISSING_INTERNAL"
  | "AMOUNT_MISMATCH"
  | "DUPLICATE_EXTERNAL"
  | "LATE_EXTERNAL"
  | "WRONG_CURRENCY"
  | "WRONG_PROVIDER_REFERENCE";

export class ReconciliationFailureInjector {
  constructor(
    private readonly sandbox:
      ReconciliationSandboxRepository,
  ) {}

  async inject(args: {
    transferId: string;
    mode:
      ReconciliationFailureMode;
  }) {
    switch (args.mode) {
      case "MISSING_EXTERNAL":
        return this.sandbox.removeExternalRecord(
          args.transferId,
        );

      case "MISSING_INTERNAL":
        return this.sandbox.removeInternalProjection(
          args.transferId,
        );

      case "AMOUNT_MISMATCH":
        return this.sandbox.offsetExternalAmount(
          args.transferId,
          100n,
        );

      case "DUPLICATE_EXTERNAL":
        return this.sandbox.duplicateExternalRecord(
          args.transferId,
        );

      case "LATE_EXTERNAL":
        return this.sandbox.delayExternalAvailability(
          args.transferId,
        );

      case "WRONG_CURRENCY":
        return this.sandbox.changeExternalCurrency(
          args.transferId,
          "ZZZ",
        );

      case "WRONG_PROVIDER_REFERENCE":
        return this.sandbox.changeProviderReference(
          args.transferId,
          `BROKEN-${args.transferId}`,
        );
    }
  }
}

import {
  createHash,
} from "crypto";

export interface EvidenceManifest {
  gitSha: string;
  buildId: string;

  createdAt: string;

  environment:
    | "staging"
    | "preprod"
    | "production";

  corridor:
    string;

  files: Array<{
    path: string;
    sha256: string;
  }>;

  testResults: Array<{
    suite: string;
    result:
      | "PASS"
      | "FAIL";
  }>;

  manifestHash: string;
}

export class EvidenceBinderService {
  async create(args: {
    gitSha: string;
    buildId: string;
    environment:
      "staging"
      | "preprod"
      | "production";
    corridor: string;
    files: Array<{
      path: string;
      content: Buffer;
    }>;
    testResults: Array<{
      suite: string;
      result:
        | "PASS"
        | "FAIL";
    }>;
  }): Promise<EvidenceManifest> {
    const files =
      args.files.map(
        (file) => ({
          path:
            file.path,

          sha256:
            createHash(
              "sha256",
            )
              .update(
                file.content,
              )
              .digest(
                "hex",
              ),
        }),
      );

    const base = {
      gitSha:
        args.gitSha,

      buildId:
        args.buildId,

      createdAt:
        new Date()
          .toISOString(),

      environment:
        args.environment,

      corridor:
        args.corridor,

      files,

      testResults:
        args.testResults,
    };

    const manifestHash =
      createHash("sha256")
        .update(
          JSON.stringify(
            base,
          ),
        )
        .digest("hex");

    return {
      ...base,
      manifestHash,
    };
  }
}

export class EvidenceFreshnessService {
  constructor(
    private readonly repo:
      CertificationEvidenceRepository,
  ) {}

  async assertFresh(args: {
    corridor: string;
    gitSha: string;
    maxAgeHours: number;
  }) {
    const evidence =
      await this.repo.getLatestPassing(
        args.corridor,
        args.gitSha,
      );

    if (!evidence) {
      throw new Error(
        "PASSING_EVIDENCE_NOT_FOUND",
      );
    }

    const ageMs =
      Date.now() -
      evidence.completedAt.getTime();

    const maxAgeMs =
      args.maxAgeHours *
      3_600_000;

    if (
      ageMs >
      maxAgeMs
    ) {
      throw new Error(
        "CERTIFICATION_EVIDENCE_STALE",
      );
    }

    return evidence;
  }
}

  destinationCurrency: string;

  payoutProviderId?: string;
  fxProviderId?: string;

  legalApprovalReference?: string;
  partnerApprovalReference?: string;

  testCaseIds: string[];

  evidenceManifestHash?: string;

  certifiedGitSha?: string;

  certifiedAt?: Date;

  status:
    | "DRAFT"
    | "TESTING"
    | "BLOCKED"
    | "CERTIFIED"
    | "SUSPENDED";
}

export class CorridorCertificationService {
  constructor(
    private readonly repo:
      CorridorCertificationRepository,

    private readonly evidence:
      EvidenceFreshnessService,

    private readonly providers:
      ProviderCertificationService,
  ) {}

  async certify(args: {
    corridorId: string;
    gitSha: string;
    maxEvidenceAgeHours: number;
  }) {
    const corridor =
      await this.repo.findById(
        args.corridorId,
      );

    if (!corridor) {
      throw new Error(
        "CORRIDOR_CERTIFICATION_NOT_FOUND",
      );
    }

    if (
      !corridor.legalApprovalReference
    ) {
      throw new Error(
        "LEGAL_APPROVAL_MISSING",
      );
    }

    if (
      !corridor.partnerApprovalReference
    ) {
      throw new Error(
        "PARTNER_APPROVAL_MISSING",
      );
    }

    for (
      const providerId
      of [
        corridor.fundingProviderId,
        corridor.payoutProviderId,
        corridor.fxProviderId,
      ].filter(Boolean)
    ) {
      await this.providers.assertCertified({
        providerId:
          providerId!,
        gitSha:
          args.gitSha,
      });
    }

    const fresh =
      await this.evidence.assertFresh({
        corridor:
          corridor.corridorId,
        gitSha:
          args.gitSha,
        maxAgeHours:
          args.maxEvidenceAgeHours,
      });

    return this.repo.markCertified({
      corridorId:
        corridor.corridorId,

      certifiedGitSha:
        args.gitSha,

      evidenceManifestHash:
        fresh.manifestHash,

      certifiedAt:
        new Date(),
    });
  }
}

export class LinqaSendGaGateService {
  constructor(
    private readonly corridors:
      CorridorCertificationRepository,

    private readonly evidence:
      EvidenceFreshnessService,

    private readonly eod:
      EodControlRepository,

    private readonly breaks:
      ReconciliationBreakRepository,

    private readonly safeguarding:
      SafeguardingAccountService,

    private readonly truth:
      TestTruthfulnessGate,

    private readonly release:
      ReleaseManifestRepository,
  ) {}

  async evaluate(args: {
    gitSha: string;
    buildId: string;

    enabledCorridorIds:
      string[];

    evidenceMaxAgeHours:
      number;

    testInventory:
      TestInventoryResult;
  }) {
    const failures: string[] = [];

    try {
      this.truth.assertClean(
        args.testInventory,
      );
    } catch (e) {
      failures.push(
        e instanceof Error
          ? e.message
          : "TEST_TRUTHFULNESS_FAILED",
      );
    }

    for (
      const corridorId
      of args.enabledCorridorIds
    ) {
      const cert =
        await this.corridors.findById(
          corridorId,
        );

      if (
        !cert ||
        cert.status !==
          "CERTIFIED"
      ) {
        failures.push(
          `CORRIDOR_NOT_CERTIFIED:${corridorId}`,
        );

        continue;
      }

      if (
        cert.certifiedGitSha !==
        args.gitSha
      ) {
        failures.push(
          `CORRIDOR_SHA_MISMATCH:${corridorId}`,
        );
      }

      try {
        await this.evidence.assertFresh({
          corridor:
            corridorId,
          gitSha:
            args.gitSha,
          maxAgeHours:
            args.evidenceMaxAgeHours,
        });
      } catch (error) {
        failures.push(
          error instanceof Error
            ? `${corridorId}:${error.message}`
            : `${corridorId}:EVIDENCE_FAILURE`,
        );
      }
    }

    const openCritical =
      await this.breaks.countAllCriticalOpen();

    if (
      openCritical > 0
    ) {
      failures.push(
        `CRITICAL_RECONCILIATION_BREAKS:${openCritical}`,
      );
    }

    const latestEod =
      await this.eod.latestAll();

    for (
      const item
      of latestEod
    ) {
      if (
        item.status ===
        "RED"
      ) {
        failures.push(
          `EOD_RED:${item.currency}`,
        );
      }
    }

    const release =
      await this.release.findByGitSha(
        args.gitSha,
      );

    if (
      !release
    ) {
      failures.push(
        "RELEASE_MANIFEST_MISSING",
      );
    } else if (
      release.buildId !==
      args.buildId
    ) {
      failures.push(
        "RELEASE_BUILD_ID_MISMATCH",
      );
    }

    return {
      gate:
        failures.length ===
        0
          ? "GREEN"
          : "RED",

      gitSha:
        args.gitSha,

      buildId:
        args.buildId,

      failures,
    };
  }
}

export class ReleaseManifestService {
  async build(args: {
    gitSha: string;
    buildId: string;

    migrationHead: string;

    enabledCorridors:
      string[];

    certifiedProviders:
      string[];

    containerImageDigests:
      string[];

    mobileBuilds:
      LinqaSendReleaseManifest["mobileBuilds"];

    serverArtifact:
      Buffer;

    configuration:
      unknown;
  }) {
    const serverArtifactHash =
      sha256(
        args.serverArtifact,
      );

    const configurationHash =
      sha256(
        Buffer.from(
          stableJson(
            args.configuration,
          ),
        ),
      );

    const base = {
      gitSha:
        args.gitSha,

      buildId:
        args.buildId,

      createdAt:
        new Date()
          .toISOString(),

      databaseMigrationHead:
        args.migrationHead,

      enabledCorridors:
        [...args.enabledCorridors]
          .sort(),

      certifiedProviders:
        [...args.certifiedProviders]
          .sort(),

      containerImageDigests:
        [...args.containerImageDigests]
          .sort(),

      mobileBuilds:
        args.mobileBuilds,

      serverArtifactHash,

      configurationHash,
    };

    return {
      ...base,

      manifestHash:
        sha256(
          Buffer.from(
            stableJson(
              base,
            ),
          ),
        ),
    };
  }
}

function sha256(
  input: Buffer,
) {
  return createHash(
    "sha256",
  )
    .update(
      input,
    )
    .digest(
      "hex",
    );
}

function stableJson(
  value: unknown,
) {
  return JSON.stringify(
    value,
    Object.keys(
      value as object,
    ).sort(),
  );
}

31. BANK INGESTION SECURITY REQUIREMENTS
================================================================================

1. Source account identity must be configured server-side.

2. Ingestion file must be hashed before parsing.

3. Duplicate files should be rejected/deduped using file hash.

4. Duplicate bank records dedupe using:
   source_type + source_id + external_record_id.

5. Do not infer settlement success from filename.

6. Reject malformed currency codes.

7. Reject invalid dates.

8. Parse amounts using decimal-to-minor exact arithmetic.

9. Store raw row hash.

10. Store import batch ID.

11. Preserve original record reference.

12. Import is append-only.

13. Corrections use a later correcting statement/report.

14. Never delete prior external evidence merely because bank sends replacement.

15. Reconciliation determines economic effect.


================================================================================
32. RECONCILIATION FAILURE CERTIFICATION
================================================================================

Before GA, prove:

- Missing provider payout creates break.
- Missing bank settlement creates break.
- Duplicate provider payout does not duplicate credit.
- Wrong amount creates exact delta.
- Wrong currency is never silently converted.
- Delayed statement can resolve on later run.
- Provider replay does not duplicate settlement.
- Ledger mismatch produces alert.
- Safeguarding shortfall produces RED.
- Critical reconciliation break blocks GA gate.
- Suspense cannot be cleared without reason/evidence.
- Reversal resolves economic position using correcting entries.
