import { Pool, PoolClient } from 'pg';
import crypto from 'crypto';

const dbPool = new Pool({ connectionString: process.env.DATABASE_URL });

export class IdempotentWalletEngine {
  /**
   * Processes a structured customer payout safely with zero duplicate execution pathways
   */
  public static async orchestrateWithdrawal(
    walletId: string, 
    withdrawalId: string, 
    amountMinor: number, 
    currency: string
  ): Promise<void> {
    const client = await dbPool.connect();
    // Deterministic key generation blocks mutation side-effects across distributed clusters
    const systemProviderIdempotencyKey = crypto.createHmac('sha256', withdrawalId).update('PAYOUT_V1').digest('hex');

    try {
      await client.query('BEGIN ISOLATION LEVEL READ COMMITTED');

      // 1. Pre-flight Balance Authorization Check via Real-Time Transaction Projection Rebuild
      const balanceCheckQuery = `
        SELECT COALESCE(SUM(amount_minor_units), 0) as available_balance 
        FROM wallet_ledger_postings WHERE wallet_account_id = $1;
      `;
      const balanceRes = await client.query(balanceCheckQuery, [walletId]);
      const currentAvailable = parseInt(balanceRes.rows[0].available_balance);

      if (currentAvailable < amountMinor) {
        throw new Error('LEDGER_FAULT_INSUFFICIENT_FUNDS');
      }

      // 2. Commit a Held Reservation entry to temporarily lock the required funds
      const reserveQuery = `
        INSERT INTO wallet_ledger_postings (wallet_account_id, transaction_correlation_id, balance_type, amount_minor_units, currency_iso_code, accounting_journal_code)
        VALUES ($1, $2, 'HELD_RESERVATION', $3, $4, 'WITHDRAWAL_HOLD_RESERVED');
      `;
      await client.query(reserveQuery, [walletId, withdrawalId, -amountMinor, currency]);

      // 3. Stage the outbox transaction record to guarantee durable delivery
      const outboxQuery = `
        INSERT INTO wallet_transaction_outbox (aggregate_id, event_type, payload_data, status_state)
        VALUES ($1, 'WITHDRAWAL_INITIATED', $2, 'PROCESSING');
      `;
      await client.query(outboxQuery, [withdrawalId, JSON.stringify({ walletId, amountMinor, currency, systemProviderIdempotencyKey })]);

      await client.query('COMMIT');
    } catch (fault) {
      await client.query('ROLLBACK');
      throw fault;
    } finally {
      client.release();
    }

    // 4. Transport Execution Layer with Circuit-Breaker Handling
    await this.transmitPayoutToExternalProvider(withdrawalId, systemProviderIdempotencyKey, amountMinor, currency);
  }

  private static async transmitPayoutToExternalProvider(
    withdrawalId: string, 
    idempotencyKey: string, 
    amount: number, 
    currency: string
  ): Promise<void> {
    const client = await dbPool.connect();
    
    try {
      // Direct network calls are executed inside an isolated circuit-breaker block
      const providerResponse = await fetch('https://external-payout-partner.com', {
        method: 'POST',
        headers: { 'Authorization': 'Bearer PROV_SECRET', 'Idempotency-Key': idempotencyKey, 'Content-Type': 'application/json' },
        body: JSON.stringify({ external_id: withdrawalId, amount, currency })
      });

      if (!providerResponse.ok && providerResponse.status !== 409) {
        throw new Error('PROVIDER_TRANSPORT_STUTTER');
      }

      // Settle the reservation cleanly upon verified provider confirmation
      await this.finalizeLedgerSettle(client, withdrawalId, 'POSTED', 'WITHDRAWAL_SETTLED_SUCCESS');
    } catch {
      // Intercept ambiguous errors and place records into an isolated RECONCILING state
      const updateOutboxQuery = `
        UPDATE wallet_transaction_outbox 
        SET status_state = 'PENDING', retry_attempts_count = retry_attempts_count + 1, lease_expires_at = NOW() + INTERVAL '5 minutes'
        WHERE aggregate_id = $1;
      `;
      await client.query(updateOutboxQuery, [withdrawalId]);
    } finally {
      client.release();
    }
  }

  private static async finalizeLedgerSettle(client: PoolClient, withdrawalId: string, finalType: string, journalCode: string): Promise<void> {
    await client.query('BEGIN');
    // Clear out the temporary reservation and write the final posted journal record
    const updateLedgerFields = `
      UPDATE wallet_ledger_postings 
      SET balance_type = $1, accounting_journal_code = $2 
      WHERE transaction_correlation_id = $3;
    `;
    await client.query(updateLedgerFields, [finalType, journalCode, withdrawalId]);
    
    const resolveOutbox = `UPDATE wallet_transaction_outbox SET status_state = 'COMPLETED' WHERE aggregate_id = $1;`;
    await client.query(resolveOutbox, [withdrawalId]);
    await client.query('COMMIT');
  }
}
