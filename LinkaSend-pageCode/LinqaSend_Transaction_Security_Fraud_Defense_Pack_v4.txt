import { ForbiddenException, Injectable } from "@nestjs/common";

@Injectable()
export class DeviceBindingService {
  constructor(private readonly devices: TrustedPaymentDeviceRepository) {}

  async assertUsable(args: {
    userId: string;
    installationId: string;
    deviceId: string;
  }) {
    const device = await this.devices.find(args);

    if (!device) {
      return {
        trustState: "UNKNOWN" as const,
        requiresStepUp: true,
      };
    }

    if (device.trustState === "REVOKED") {
      throw new ForbiddenException("PAYMENT_DEVICE_REVOKED");
    }

    return {
      trustState: device.trustState,
      requiresStepUp: device.trustState !== "TRUSTED",
      deviceBindingKeyId: device.deviceBindingKeyId,
      firstSeenAt: device.firstSeenAt,
    };
  }
}

import { Injectable } from "@nestjs/common";

@Injectable()
export class SecurityChangeCooldownService {
  constructor(private readonly events: SecurityChangeEventRepository) {}

  async evaluate(userId: string) {
    const recent = await this.events.listRecent(
      userId,
      new Date(Date.now() - 72 * 60 * 60 * 1000),
    );

    let until: Date | undefined;
    const reasons: string[] = [];

    for (const event of recent) {
      let hours = 0;

      switch (event.eventType) {
        case "ACCOUNT_RECOVERY_COMPLETED":
        case "PRIMARY_PHONE_CHANGED":
        case "RECOVERY_PHONE_CHANGED":
          hours = 24;
          break;

        case "PASSWORD_RESET":
        case "RECOVERY_EMAIL_CHANGED":
        case "DEVICE_ENROLLED":
          hours = 6;
          break;
      }

      if (hours > 0) {
        reasons.push(event.eventType);
        const candidate = new Date(
          event.eventAt.getTime() + hours * 60 * 60 * 1000,
        );
        if (!until || candidate > until) until = candidate;
      }
    }

    return {
      active: !!until && until > new Date(),
      cooldownUntil: until,
      reasons,
    };
  }
}

import { createHash } from "crypto";
import { Injectable } from "@nestjs/common";

@Injectable()
export class TransactionSignatureService {
  buildTransactionHash(args: {
    transferId: string;
    senderUserId: string;
    recipientId: string;
    quoteId: string;
    sourceAmountMinor: bigint;
    sourceCurrency: string;
    destinationAmountMinor: bigint;
    destinationCurrency: string;
    fundingSourceId: string;
    transferClass: string;
  }) {
    const canonical = [
      "LINQASEND_TX_V4",
      args.transferId,
      args.senderUserId,
      args.recipientId,
      args.quoteId,
      args.sourceAmountMinor.toString(),
      args.sourceCurrency.toUpperCase(),
      args.destinationAmountMinor.toString(),
      args.destinationCurrency.toUpperCase(),
      args.fundingSourceId,
      args.transferClass,
    ].join("|");

    return createHash("sha256").update(canonical).digest("hex");
  }
}

import { createHash, randomBytes } from "crypto";
import { Injectable } from "@nestjs/common";

@Injectable()
export class TransactionChallengeService {
  constructor(
    private readonly repo: TransactionChallengeRepository,
    private readonly signatures: TransactionSignatureService,
  ) {}

  async create(args: {
    transfer: SecureTransferView;
    userId: string;
    installationId: string;
    deviceId: string;
  }) {
    const transactionHash = this.signatures.buildTransactionHash({
      transferId: args.transfer.id,
      senderUserId: args.transfer.senderUserId,
      recipientId: args.transfer.recipientId,
      quoteId: args.transfer.quoteId,
      sourceAmountMinor: args.transfer.sourceAmountMinor,
      sourceCurrency: args.transfer.sourceCurrency,
      destinationAmountMinor: args.transfer.destinationAmountMinor,
      destinationCurrency: args.transfer.destinationCurrency,
      fundingSourceId: args.transfer.fundingSourceId,
      transferClass: args.transfer.transferClass,
    });

    const nonce = randomBytes(32).toString("base64url");
    const challengeHash = createHash("sha256")
      .update(`${nonce}|${transactionHash}`)
      .digest("hex");

    return this.repo.create({
      transferId: args.transfer.id,
      userId: args.userId,
      challengeNonce: nonce,
      challengeHash,
      expectedTransactionHash: transactionHash,
      installationId: args.installationId,
      deviceId: args.deviceId,
      expiresAt: new Date(Date.now() + 120_000),
    });
  }
}

import { ForbiddenException, Injectable } from "@nestjs/common";
import { randomUUID } from "crypto";

@Injectable()
export class TransactionAuthorizationService {
  constructor(
    private readonly challenges: TransactionChallengeRepository,
    private readonly authorizations: TransactionAuthorizationRepository,
    private readonly assertions: DeviceAssertionVerifier,
    private readonly signer: TransactionTokenSigner,
    private readonly redis: RedisService,
  ) {}

  async authorize(args: {
    challengeId: string;
    userId: string;
    installationId: string;
    deviceId: string;
    assertion: DevicePaymentAssertion;
    authMethod:
      | "PASSKEY"
      | "BIOMETRIC_DEVICE_ASSERTION"
      | "DEVICE_CREDENTIAL";
  }) {
    const challenge = await this.challenges.findActive(args.challengeId);

    if (!challenge) {
      throw new ForbiddenException("PAYMENT_CHALLENGE_EXPIRED");
    }

    if (
      challenge.userId !== args.userId ||
      challenge.installationId !== args.installationId ||
      challenge.deviceId !== args.deviceId
    ) {
      throw new ForbiddenException("PAYMENT_CHALLENGE_BINDING_MISMATCH");
    }

    const verified = await this.assertions.verify({
      challenge: challenge.challengeNonce,
      expectedTransactionHash: challenge.expectedTransactionHash,
      assertion: args.assertion,
      userId: args.userId,
      deviceId: args.deviceId,
    });

    if (!verified.ok) {
      throw new ForbiddenException("PAYMENT_ASSERTION_INVALID");
    }

    await this.challenges.consume(challenge.id);

    const jti = randomUUID();
    const expiresAt = new Date(Date.now() + 120_000);

    await this.authorizations.create({
      transferId: challenge.transferId,
      userId: args.userId,
      jti,
      transactionHash: challenge.expectedTransactionHash,
      deviceBindingKeyId: verified.deviceBindingKeyId,
      authMethod: args.authMethod,
      expiresAt,
    });

    await this.redis.set(`linqasend:v4:auth:${jti}`, "ACTIVE", 120);

    return this.signer.sign({
      sub: args.userId,
      transferId: challenge.transferId,
      transactionHash: challenge.expectedTransactionHash,
      purpose: "LINQASEND_TRANSFER_CONFIRM",
      aud: "linqasend-money-movement",
      jti,
      exp: Math.floor(expiresAt.getTime() / 1000),
    });
  }

  async consume(
    token: string,
    expected: {
      userId: string;
      transferId: string;
      transactionHash: string;
    },
  ) {
    const claims = await this.signer.verify(token);

    if (
      claims.sub !== expected.userId ||
      claims.transferId !== expected.transferId ||
      claims.transactionHash !== expected.transactionHash ||
      claims.purpose !== "LINQASEND_TRANSFER_CONFIRM" ||
      claims.aud !== "linqasend-money-movement"
    ) {
      throw new ForbiddenException("PAYMENT_AUTH_TOKEN_MISMATCH");
    }

    const state = await this.redis.getDel(`linqasend:v4:auth:${claims.jti}`);

    if (state !== "ACTIVE") {
      throw new ForbiddenException("PAYMENT_AUTH_TOKEN_USED_OR_EXPIRED");
    }

    await this.authorizations.markConsumed(claims.jti);
    return claims;
  }
}

import { Injectable, TooManyRequestsException } from "@nestjs/common";

@Injectable()
export class VelocityLimitService {
  constructor(private readonly redis: RedisService) {}

  async enforce(args: {
    userId: string;
    amountMinor: bigint;
    currency: string;
    isNewRecipient: boolean;
    policy: {
      maxTransfers1h: number;
      maxTransfers24h: number;
      newRecipientMaxTransfers24h?: number;
      maxAmount24hMinor?: bigint;
    };
  }) {
    const oneHour = await this.redis.incrWithTtl(
      `linqasend:v4:velocity:${args.userId}:1h`,
      3600,
    );

    if (oneHour > args.policy.maxTransfers1h) {
      throw new TooManyRequestsException("TRANSFER_VELOCITY_1H_EXCEEDED");
    }

    const oneDay = await this.redis.incrWithTtl(
      `linqasend:v4:velocity:${args.userId}:24h`,
      86400,
    );

    if (oneDay > args.policy.maxTransfers24h) {
      throw new TooManyRequestsException("TRANSFER_VELOCITY_24H_EXCEEDED");
    }

    if (
      args.isNewRecipient &&
      args.policy.newRecipientMaxTransfers24h !== undefined
    ) {
      const count = await this.redis.incrWithTtl(
        `linqasend:v4:velocity:${args.userId}:new-recipient:24h`,
        86400,
      );

      if (count > args.policy.newRecipientMaxTransfers24h) {
        throw new TooManyRequestsException(
          "NEW_RECIPIENT_VELOCITY_EXCEEDED",
        );
      }
    }

    if (args.policy.maxAmount24hMinor !== undefined) {
      const total = await this.redis.addBigIntWithTtl(
        `linqasend:v4:velocity:${args.userId}:amount:${args.currency}:24h`,
        args.amountMinor,
        86400,
      );

      if (total > args.policy.maxAmount24hMinor) {
        throw new TooManyRequestsException("TRANSFER_AMOUNT_24H_EXCEEDED");
      }
    }
  }
}

import { Injectable } from "@nestjs/common";

@Injectable()
export class RecipientTrustService {
  constructor(private readonly repo: RecipientTrustRepository) {}

  async evaluate(senderUserId: string, recipientId: string) {
    const row = await this.repo.find(senderUserId, recipientId);

    if (!row) {
      return {
        isNew: true,
        trustState: "NEW" as const,
        successfulTransfers: 0,
        relationshipAgeDays: 0,
      };
    }

    return {
      isNew: row.successfulTransferCount === 0,
      trustState: row.trustState,
      successfulTransfers: row.successfulTransferCount,
      relationshipAgeDays: row.firstTransferAt
        ? Math.max(
            0,
            Math.floor(
              (Date.now() - row.firstTransferAt.getTime()) / 86_400_000,
            ),
          )
        : 0,
    };
  }
}

import { Injectable } from "@nestjs/common";

@Injectable()
export class FundingFraudService {
  constructor(
    private readonly fingerprints: FundingFingerprintRepository,
    private readonly failures: FundingFailureRepository,
  ) {}

  async collect(userId: string, fundingSourceId: string) {
    const source = await this.fingerprints.findByFundingSource(
      fundingSourceId,
    );

    const failedAttempts = await this.failures.countRecent(
      userId,
      24 * 60 * 60,
    );

    if (!source) {
      return {
        fundingSourceAgeDays: 0,
        fundingSourceReuseAcrossAccounts: 0,
        priorFailedFundingAttempts: failedAttempts,
      };
    }

    const reuse = await this.fingerprints.countOwnersForFingerprint(
      source.provider,
      source.providerFingerprint,
    );

    return {
      fundingSourceAgeDays: Math.max(
        0,
        Math.floor(
          (Date.now() - source.firstSeenAt.getTime()) / 86_400_000,
        ),
      ),
      fundingSourceReuseAcrossAccounts: reuse,
      priorFailedFundingAttempts: failedAttempts,
    };
  }
}

import { Injectable } from "@nestjs/common";

@Injectable()
export class PaymentRiskEngineService {
  readonly modelVersion = "linqasend-risk-v4.1";

  assess(s: PaymentRiskSignals): PaymentRiskResult {
    let score = 0;
    const reasons: string[] = [];

    const add = (condition: boolean, points: number, reason: string) => {
      if (condition) {
        score += points;
        reasons.push(reason);
      }
    };

    add(s.blockedRelationship, 100, "BLOCKED_RELATIONSHIP");
    add(s.deviceTrust === "REVOKED", 100, "REVOKED_DEVICE");
    add(s.deviceTrust === "RISKY", 35, "RISKY_DEVICE");
    add(s.deviceTrust === "UNKNOWN", 15, "UNKNOWN_DEVICE");

    add(s.accountAgeDays < 7, 12, "YOUNG_ACCOUNT");
    add(s.deviceAgeDays < 1, 15, "NEW_DEVICE");

    add(s.recentPasswordReset, 20, "RECENT_PASSWORD_RESET");
    add(s.recentRecoveryChange, 25, "RECENT_RECOVERY_CHANGE");
    add(s.recentPhoneChange, 30, "RECENT_PHONE_CHANGE");
    add(s.recentDeviceEnrollment, 15, "RECENT_DEVICE_ENROLLMENT");

    add(s.recipientIsNew, 12, "NEW_RECIPIENT");
    add(
      s.priorSuccessfulTransfers === 0 && s.relationshipAgeDays < 1,
      12,
      "NO_ESTABLISHED_RELATIONSHIP",
    );

    add(s.transferCount1h >= 4, 20, "HIGH_VELOCITY_1H");
    add(s.transferCount24h >= 10, 15, "HIGH_VELOCITY_24H");
    add(s.amountDeviationRatio >= 3, 20, "UNUSUAL_AMOUNT");

    add(s.fundingSourceAgeDays < 1, 18, "NEW_FUNDING_SOURCE");
    add(
      s.fundingSourceReuseAcrossAccounts >= 3,
      25,
      "FUNDING_SOURCE_MULTI_ACCOUNT_REUSE",
    );

    add(s.countryMismatch, 15, "COUNTRY_MISMATCH");
    add(s.impossibleTravelSignal, 35, "IMPOSSIBLE_TRAVEL");

    add(s.priorChargebacks >= 1, 20, "PRIOR_CHARGEBACK");
    add(
      s.priorFailedFundingAttempts >= 3,
      20,
      "REPEATED_FUNDING_FAILURES",
    );

    add(s.quoteAgeSeconds > 60, 20, "STALE_QUOTE");

    score = Math.min(100, score);

    let decision: PaymentRiskDecision = "ALLOW";

    if (score >= 90) decision = "FREEZE";
    else if (score >= 80) decision = "DECLINE";
    else if (score >= 65) decision = "HOLD";
    else if (score >= 50) decision = "REVIEW";
    else if (score >= 30) decision = "WARN";
    else if (score >= 15) decision = "STEP_UP";

    return {
      score,
      decision,
      reasons,
      modelVersion: this.modelVersion,
    };
  }
}

    const has = (r: string) => risk.reasons.includes(r);

      warnings.push({

        severity: "WARN",
        messageKey: "linqasend.scam.new_recipient",
      });
    }

    if (
      has("RECENT_PASSWORD_RESET") ||
      has("RECENT_RECOVERY_CHANGE") ||
      has("RECENT_PHONE_CHANGE")
    ) {
      warnings.push({
        code: "RECENT_ACCOUNT_SECURITY_CHANGE",
        severity: "HIGH",
        messageKey: "linqasend.scam.security_change",
      });
    }

    if (has("FUNDING_SOURCE_MULTI_ACCOUNT_REUSE")) {
      warnings.push({
        code: "FUNDING_SOURCE_REUSE",
        severity: "HIGH",
        messageKey: "linqasend.scam.funding_source_reuse",
      });
    }

    if (has("IMPOSSIBLE_TRAVEL")) {
      warnings.push({
        code: "LOCATION_ANOMALY",
        severity: "CRITICAL",
        messageKey: "linqasend.scam.location_anomaly",
      });
    }

    return warnings;
  }

import { ForbiddenException, Injectable } from "@nestjs/common";

@Injectable()
export class TransferSecurityOrchestratorService {
  constructor(
    private readonly devices: DeviceBindingService,
    private readonly cooldowns: SecurityChangeCooldownService,
    private readonly recipients: RecipientTrustService,
    private readonly velocity: VelocityLimitService,
    private readonly signalCollector: PaymentRiskSignalCollector,
    private readonly risk: PaymentRiskEngineService,
    private readonly scam: ScamDefenseService,
    private readonly riskRepo: PaymentRiskAssessmentRepository,
    private readonly warningsRepo: ScamWarningRepository,
    private readonly audit: SecurityAuditEvidenceService,
  ) {}

  async evaluate(args: {
    transfer: SecureTransferView;
    senderUserId: string;
    installationId: string;
    deviceId: string;
    velocityPolicy: {
      maxTransfers1h: number;
      maxTransfers24h: number;
      newRecipientMaxTransfers24h?: number;
      maxAmount24hMinor?: bigint;
    };
  }): Promise<SecurityDecision> {
    const device = await this.devices.assertUsable({
      userId: args.senderUserId,
      installationId: args.installationId,
      deviceId: args.deviceId,
    });

    const cooldown = await this.cooldowns.evaluate(args.senderUserId);

    const recipientTrust = await this.recipients.evaluate(
      args.senderUserId,
      args.transfer.recipientId,
    );

    if (recipientTrust.trustState === "BLOCKED") {
      throw new ForbiddenException("RECIPIENT_BLOCKED");
    }

    await this.velocity.enforce({
      userId: args.senderUserId,
      amountMinor: args.transfer.sourceAmountMinor,
      currency: args.transfer.sourceCurrency,
      isNewRecipient: recipientTrust.isNew,
      policy: args.velocityPolicy,
    });

    const signals = await this.signalCollector.collect({
      transfer: args.transfer,
      senderUserId: args.senderUserId,
      installationId: args.installationId,
      deviceId: args.deviceId,
      deviceTrust: device.trustState,
      recipientTrust,
    });

    const result = this.risk.assess(signals);

    await this.riskRepo.insert({
      transferId: args.transfer.id,
      userId: args.senderUserId,
      score: result.score,
      decision: result.decision,
      modelVersion: result.modelVersion,
      reasons: result.reasons,
      signals: {
        ...signals,
        amountSent24hMinor: signals.amountSent24hMinor.toString(),
      },
    });

    if (result.decision === "FREEZE") {
      await this.audit.append({
        actorType: "SYSTEM",
        actorId: "linqasend-risk-v4",
        transferId: args.transfer.id,
        eventType: "TRANSFER_SECURITY_FREEZE",
        evidence: {
          score: result.score,
          reasons: result.reasons,
        },
      });

      return {
        risk: result,
        requireStepUp: true,
        requireScamWarning: false,
        warnings: [],
        blocked: true,
        blockReason: "SECURITY_FREEZE",
      };
    }

    if (["DECLINE", "HOLD", "REVIEW"].includes(result.decision)) {
      return {
        risk: result,
        requireStepUp: true,
        requireScamWarning: false,
        warnings: [],
        blocked: true,
        blockReason:
          result.decision === "DECLINE"
            ? "SECURITY_DECLINE"
            : result.decision === "HOLD"
              ? "SECURITY_HOLD"
              : "SECURITY_REVIEW",
      };
    }

    const warnings = this.scam.buildWarnings(result);

    for (const warning of warnings) {
      await this.warningsRepo.insert({
        transferId: args.transfer.id,
        userId: args.senderUserId,
        warningCode: warning.code,
        severity: warning.severity,
        metadata: {},
      });
    }

    if (cooldown.active) {
      return {
        risk: result,
        requireStepUp: true,
        requireScamWarning: warnings.length > 0,
        warnings,
        securityCooldownUntil: cooldown.cooldownUntil?.toISOString(),
        blocked: true,
        blockReason: "SECURITY_CHANGE_COOLDOWN",
      };
    }

    return {
      risk: result,
      requireStepUp: true,
      requireScamWarning: warnings.length > 0,
      warnings,
      blocked: false,
    };
  }
}

import { createHash } from "crypto";
import { ForbiddenException, Injectable } from "@nestjs/common";

@Injectable()
export class WebhookSecurityService {
  constructor(
    private readonly providers: WebhookProviderRegistry,
    private readonly events: ProviderWebhookEventRepository,
  ) {}

  async verifyAndRegister(args: {
    provider: string;
    rawBody: Buffer;
    headers: Record<string, string | string[] | undefined>;
  }) {
    const adapter = this.providers.get(args.provider);

    const parsed = await adapter.parseAndVerify({
      rawBody: args.rawBody,
      headers: args.headers,
    });

    if (!parsed.signatureValid) {
      throw new ForbiddenException("WEBHOOK_SIGNATURE_INVALID");
    }

    if (
      Math.abs(Date.now() - parsed.signedTimestamp.getTime()) >
      adapter.timestampToleranceMs
    ) {
      throw new ForbiddenException("WEBHOOK_TIMESTAMP_OUT_OF_RANGE");
    }

    const payloadHash = createHash("sha256")
      .update(args.rawBody)
      .digest("hex");

    const existing = await this.events.findByProviderEventId(
      args.provider,
      parsed.eventId,
    );

    if (existing) {
      if (existing.payloadHash !== payloadHash) {
        throw new ForbiddenException("WEBHOOK_EVENT_ID_PAYLOAD_MISMATCH");
      }

      return {
        duplicate: true,
        event: existing,
      };
    }

    const event = await this.events.insert({
      provider: args.provider,
      providerEventId: parsed.eventId,
      eventType: parsed.eventType,
      signatureVerified: true,
      timestampVerified: true,
      payloadHash,
      replayDetected: false,
    });

    return {
      duplicate: false,
      event,
      parsed,
    };
  }
}

import { createHash } from "crypto";
import { Injectable } from "@nestjs/common";

@Injectable()
export class LedgerIntegrityService {
  constructor(
    private readonly ledger: LedgerReadRepository,
    private readonly integrity: LedgerIntegrityRepository,
  ) {}

  async sealJournal(journalId: string) {
    const journal = await this.ledger.getJournalWithPostings(journalId);

    if (!journal) {
      throw new Error("LEDGER_JOURNAL_NOT_FOUND");
    }

    const previous = await this.integrity.getLatestBefore(
      journal.createdAt,
    );

    const postingText = [...journal.postings]
      .sort((a, b) => a.id.localeCompare(b.id))
      .map((p) =>
        [
          p.id,
          p.accountId,
          p.direction,
          p.amountMinor.toString(),
          p.currency,
        ].join("|"),
      )
      .join("\n");

    const canonical = [
      "LINQASEND_LEDGER_V4",
      journal.id,
      journal.referenceType,
      journal.referenceId,
      journal.description,
      previous?.journalHash ?? "",
      postingText,
    ].join("\n");

    const journalHash = createHash("sha256")
      .update(canonical)
      .digest("hex");

    await this.integrity.insert({
      journalId: journal.id,
      previousHash: previous?.journalHash,
      journalHash,
      postingCount: journal.postings.length,
    });

    return journalHash;
  }
}

      transferId: args.transferId,
      eventType: "PRIVILEGED_MONEY_ACTION_REQUESTED",
      evidence: {
        actionId: action.id,
        actionType: args.actionType,
        reasonCode: args.reasonCode,
      },
    });

    return action;
  }

  async approveAction(approverUserId: string, actionId: string) {
    await this.scopes.require(
      approverUserId,
      "ops:money:privileged:approve",
    );

    const action = await this.actions.findById(actionId);

    if (!action) throw new Error("PRIVILEGED_ACTION_NOT_FOUND");

    if (action.requestedBy === approverUserId) {
      throw new ForbiddenException(
        "MAKER_CHECKER_SELF_APPROVAL_DENIED",
      );
    }

    return this.actions.approve({
      actionId,
      approvedBy: approverUserId,
    });
  }
}

async function request<T>(
  path: string,
  token: string,
  init: RequestInit = {},
): Promise<T> {
  const response = await fetch(
    `${API}/v1/linqasend/security${path}`,
    {
      ...init,
      headers: {
        "Content-Type": "application/json",
        Authorization: `Bearer ${token}`,
        ...(init.headers || {}),
      },
    },
  );

  if (!response.ok) {
    const body = await response.json().catch(() => ({}));
    throw new Error(
      body.message || body.error || `HTTP_${response.status}`,
    );
  }

  return response.json();
}
