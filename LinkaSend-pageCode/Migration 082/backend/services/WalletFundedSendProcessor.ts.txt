//backend/services/WalletFundedSendProcessor.ts

import { Pool } from 'pg';
import crypto from 'crypto';

const dbPool = new Pool({ connectionString: process.env.DATABASE_URL });

export class WalletFundedSendProcessor {
  /**
   * Executes a wallet-funded international money transfer featuring multi-currency FX leg splitting
   */
  public static async executeWalletFundedSend(params: {
    walletId: string;
    transferCorrelationId: string;
    sourceAmountMinor: number;
    sourceCurrency: string;
    destinationCurrency: string;
    exchangeRate: number;
    providerCode: string;
  }): Promise<void> {
    const client = await dbPool.connect();
    const providerOperationKey = crypto.createHmac('sha256', params.transferCorrelationId).update('SEND_FX_V1').digest('hex');

    try {
      await client.query('BEGIN ISOLATION LEVEL READ COMMITTED');

      // 1. Authoritative Pre-Flight Check: Project current available balance from the subledger
      const checkBalanceQuery = `
        SELECT COALESCE(SUM(amount_minor_units), 0) as available_balance 
        FROM wallet_ledger_postings WHERE wallet_account_id = $1 AND balance_type = 'POSTED';
      `;
      const balanceRes = await client.query(checkBalanceQuery, [params.walletId]);
      const currentAvailable = parseInt(balanceRes.rows[0].available_balance);

      if (currentAvailable < params.sourceAmountMinor) {
        throw new Error('LEDGER_FAULT_INSUFFICIENT_FUNDS');
      }

      // 2. Commit Source Currency Debit Posting entry to lock funds
      const debitQuery = `
        INSERT INTO wallet_ledger_postings (wallet_account_id, transaction_correlation_id, provider_operation_key, balance_type, amount_minor_units, currency_iso_code, accounting_journal_code)
        VALUES ($1, $2, $3, 'POSTED', $4, $5, 'WALLET_DEBIT_INTERNATIONAL_SEND');
      `;
      await client.query(debitQuery, [params.walletId, params.transferCorrelationId, providerOperationKey, -params.sourceAmountMinor, params.sourceCurrency]);

      // 3. Calculate conversion parameters and execute the FX leg entry assignment
      const destinationAmountMinor = Math.floor(params.sourceAmountMinor * params.exchangeRate);
      const fxLegQuery = `
        INSERT INTO wallet_fx_ledger_legs (transaction_correlation_id, source_currency, source_amount_minor, destination_currency, destination_amount_minor, applied_exchange_rate, spread_margin_minor_units)
        VALUES ($1, $2, $3, $4, $5, $6, $7);
      `;
      const dummySpreadMargin = 100; // Internal profit optimization target (1.00 base unit baseline)
      await client.query(fxLegQuery, [params.transferCorrelationId, params.sourceCurrency, params.sourceAmountMinor, params.destinationCurrency, destinationAmountMinor, params.exchangeRate, dummySpreadMargin]);

      // 4. Update Provider Custody holding tracking records to match expected settlement outbox loads
      const custodyUpdateQuery = `
        UPDATE linqa_provider_custody_accounts 
        SET custody_balance_minor_units = custody_balance_minor_units + $1
        WHERE provider_code_identifier = $2 AND currency_iso_code = $3;
      `;
      await client.query(custodyUpdateQuery, [destinationAmountMinor, params.providerCode, params.destinationCurrency]);

      await client.query('COMMIT');
    } catch (fault) {
      await client.query('ROLLBACK');
      throw fault;
    } finally {
      client.release();
    }
  }
}
