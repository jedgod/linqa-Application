//backend/metrics/WalletMetricsExporter.ts

import client from 'prom-client';
import express from 'express';

// Initialize a clean metrics registry sandbox
const walletRegistry = new client.Registry();

client.collectDefaultMetrics({ register: walletRegistry });

// Counter metric measuring aggregate cross-currency exchange profit spreads
export const fxSpreadMarginCounter = new client.Counter({
  name: 'linqasend_wallet_fx_spread_margin_total',
  help: 'Total aggregated volume of captured revenue margins generated by cross-currency exchange spreads.',
  labelNames: ['source_currency', 'destination_currency'],
});

walletRegistry.registerMetric(fxSpreadMarginCounter);

const metricsServer = express();

/**
 * Public metrics scraping route mapped into Prometheus monitoring networks
 */
metricsServer.get('/metrics', async (req, res) => {
  try {
    res.set('Content-Type', walletRegistry.contentType);
    res.end(await walletRegistry.metrics());
  } catch (err) {
    res.status(500).end(err);
  }
});

export const startMetricsServer = (port: number) => {
  metricsServer.listen(port, () => {
    console.log(`📊 [Telemetry] Prometheus metrics server active at port ${port}/metrics`);
  });
};
