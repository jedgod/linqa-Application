LINQASEND™ — REGULATED ENTITY RESOLVER + REAL-MONEY READINESS PACK v7
Missing Real-User / Real-Money Control Layer
Format: Notepad-friendly TXT

PURPOSE
--------------------------------------------------------------------------------
This module sits between LinqaSend Capability Engine v3 and Security/Fraud v4.

It answers one critical question before any real user money can move:

"Which regulated / approved entity is responsible for each financial leg
of this transfer?"

Linqa remains the technology, user-experience, orchestration, messaging,
risk-display, status and reconciliation platform.

Real-money movement is allowed only when an approved financial-partner
responsibility map exists for the exact:
- source country
- destination country
- transfer class
- source currency
- destination currency
- funding method
- payout method
- customer/account model
- provider arrangement
- effective policy/version

POSITION
--------------------------------------------------------------------------------

Authenticated User
        |
        v
Recipient Resolver
        |
        v
Transfer Classifier
        |
        v
Country / Corridor Capability Engine v3
        |
        v
[ REGULATED ENTITY RESOLVER v7 ]
        |
        +-- Who accepts customer funds?
        +-- Who legally holds / safeguards customer funds?
        +-- Who provides customer account / stored value?
        +-- Who transmits value?
        +-- Who performs FX?
        +-- Who performs domestic payout?
        +-- Who performs cross-border payout?
        +-- Who performs KYC/CIP?
        +-- Who performs AML transaction monitoring?
        +-- Who performs sanctions screening?
        +-- Who manages disputes / chargebacks?
        +-- Who settles and reconciles?
        |
        v
Security / Fraud Defense v4
        |
        v
Money Movement
        |
        v
Settlement / Treasury / Safeguarding v5
        |
        v
Implementation Closure / Certification v6

FAIL-CLOSED RULE
--------------------------------------------------------------------------------
If the exact transfer does not resolve to a complete, active, approved,
effective responsibility map:

DO NOT QUOTE AS EXECUTABLE.
DO NOT AUTHORIZE FUNDING.
DO NOT POST CUSTOMER MONEY MOVEMENT.
DO NOT PAY OUT.

export class ResponsibilityRequirementService {
  derive(args: {
    transferClass: string;
    fundingMode: string;
    receiveMode: string;

    sourceCurrency: string;
    destinationCurrency: string;

    custodyModel:
      CustodyModel;
  }): RequiredResponsibilities {
    const crossBorder =
      args.transferClass ===
        "INTERNAL_CROSS_BORDER" ||
      args.transferClass ===
        "EXTERNAL_CROSS_BORDER";

    const requiresFx =
      args.sourceCurrency !==
      args.destinationCurrency;

    const result:
      RequiredResponsibilities = {
        CUSTOMER_ONBOARDING:
          "REQUIRED",

        KYC_CIP:
          "REQUIRED",

        AML_MONITORING:
          "REQUIRED",

        SANCTIONS_SCREENING:
          "REQUIRED",

        ACCEPT_CUSTOMER_FUNDS:
          args.fundingMode ===
            "LINQA_BALANCE"
            ? "OPTIONAL"
            : "REQUIRED",

        CUSTOMER_FUNDS_CUSTODY:
          args.custodyModel ===
            "NO_CUSTOMER_FUNDS"
            ? "NOT_APPLICABLE"
            : "REQUIRED",

        SAFEGUARDING:
          [
            "PARTNER_CUSTODY",
            "FBO_OR_EQUIVALENT_PARTNER_ACCOUNT",
            "PARTNER_ISSUED_STORED_VALUE",
          ].includes(
            args.custodyModel,
          )
            ? "REQUIRED"
            : "NOT_APPLICABLE",

        CUSTOMER_ACCOUNT_ISSUER:
          args.receiveMode ===
            "LINQA_BALANCE"
            ? "REQUIRED"
            : "OPTIONAL",

        STORED_VALUE_ISSUER:
          args.receiveMode ===
            "LINQA_BALANCE"
            ? "REQUIRED"
            : "NOT_APPLICABLE",

        DOMESTIC_TRANSMISSION:
          crossBorder
            ? "NOT_APPLICABLE"
            : "REQUIRED",

        CROSS_BORDER_TRANSMISSION:
          crossBorder
            ? "REQUIRED"
            : "NOT_APPLICABLE",

        FX_EXECUTION:
          requiresFx
            ? "REQUIRED"
            : "NOT_APPLICABLE",

        CARD_ACQUIRING:
          [
            "DEBIT_CARD",
            "CREDIT_CARD",
          ].includes(
            args.fundingMode,
          )
            ? "REQUIRED"
            : "NOT_APPLICABLE",

        BANK_DEBIT:
          args.fundingMode ===
            "BANK"
            ? "REQUIRED"
            : "NOT_APPLICABLE",

        MOBILE_MONEY_PAYOUT:
          args.receiveMode ===
            "MOBILE_MONEY"
            ? "REQUIRED"
            : "NOT_APPLICABLE",

        CARD_PUSH_PAYOUT:
          args.receiveMode ===
            "CARD"
            ? "REQUIRED"
            : "NOT_APPLICABLE",

        LOCAL_BANK_PAYOUT:
          args.receiveMode ===
            "BANK"
            ? "REQUIRED"
            : "NOT_APPLICABLE",

        DISPUTES:
          "REQUIRED",

        CHARGEBACKS:
          [
            "DEBIT_CARD",
            "CREDIT_CARD",
          ].includes(
            args.fundingMode,
          )
            ? "REQUIRED"
            : "OPTIONAL",

        REFUNDS:
          "REQUIRED",

        SETTLEMENT:
          "REQUIRED",

        RECONCILIATION:
          "REQUIRED",
      };

    return result;
  }
}

import {
  Injectable,
} from "@nestjs/common";

@Injectable()
export class RegulatedEntityResolverService {
  constructor(
    private readonly responsibilityRepo:
      FinancialResponsibilityRepository,

    private readonly entityRepo:
      EntityRegistryRepository,

    private readonly contracts:
      PartnerContractRepository,

    private readonly requirements:
      ResponsibilityRequirementService,
  ) {}

  async resolve(args: {
    sourceCountry: string;
    destinationCountry: string;

    transferClass: string;

    sourceCurrency: string;
    destinationCurrency: string;

    fundingMode: string;
    receiveMode: string;

    at?: Date;
  }): Promise<
    RegulatedEntityResolution
  > {
    const at =
      args.at ??
      new Date();

    const map =
      await this.responsibilityRepo.findActive({
        sourceCountry:
          args.sourceCountry,

        destinationCountry:
          args.destinationCountry,

        transferClass:
          args.transferClass,

        sourceCurrency:
          args.sourceCurrency,

        destinationCurrency:
          args.destinationCurrency,

        fundingMode:
          args.fundingMode,

        receiveMode:
          args.receiveMode,

        at,
      });

    if (!map) {
      return {
        responsibilityMapId:
          "NONE",

        policyVersion:
          0,

        sourceCountry:
          args.sourceCountry,

        destinationCountry:
          args.destinationCountry,

        transferClass:
          args.transferClass,

        custodyModel:
          "NO_CUSTOMER_FUNDS",

        authorities: [],

        legalApprovalReference:
          "",

        partnerApprovalReference:
          "",

        effectiveAt:
          at.toISOString(),

        complete:
          false,

        missingResponsibilities: [],

        blockedReasons: [
          "FINANCIAL_RESPONSIBILITY_MAP_NOT_CONFIGURED",
        ],
      };
    }

    const required =
      this.requirements.derive({
        transferClass:
          args.transferClass,

        fundingMode:
          args.fundingMode,

        receiveMode:
          args.receiveMode,

        sourceCurrency:
          args.sourceCurrency,

        destinationCurrency:
          args.destinationCurrency,

        custodyModel:
          map.custodyModel,
      });

    const authorities:
      FinancialExecutionAuthority[] =
        [];

    const missing:
      FinancialResponsibility[] =
        [];

    const blockedReasons:
      string[] =
        [];

    for (
      const [
        responsibility,
        requirement,
      ]
      of Object.entries(
        required,
      ) as Array<
        [
          FinancialResponsibility,
          ResponsibilityRequirement,
        ]
      >
    ) {
      if (
        requirement ===
        "NOT_APPLICABLE"
      ) {
        continue;
      }

      const entityId =
        getEntityForResponsibility(
          map,
          responsibility,
        );

      if (!entityId) {
        if (
          requirement ===
          "REQUIRED"
        ) {
          missing.push(
            responsibility,
          );
        }

        continue;
      }

      const entity =
        await this.entityRepo.findById(
          entityId,
        );

      if (!entity) {
        missing.push(
          responsibility,
        );

        blockedReasons.push(
          `ENTITY_NOT_FOUND:${responsibility}:${entityId}`,
        );

        continue;
      }

      if (
        entity.status !==
        "ACTIVE"
      ) {
        blockedReasons.push(
          `ENTITY_NOT_ACTIVE:${responsibility}:${entity.id}`,
        );

        continue;
      }

      if (
        entity.dueDiligenceStatus !==
        "APPROVED"
      ) {
        blockedReasons.push(
          `ENTITY_DUE_DILIGENCE_NOT_APPROVED:${entity.id}`,
        );

        continue;
      }

      const contract =
        await this.contracts.findEffectiveContract({
          entityId:
            entity.id,

          responsibility,

          sourceCountry:
            args.sourceCountry,

          destinationCountry:
            args.destinationCountry,

          sourceCurrency:
            args.sourceCurrency,

          destinationCurrency:
            args.destinationCurrency,

          at,
        });

      if (!contract) {
        missing.push(
          responsibility,
        );

        blockedReasons.push(
          `CONTRACT_NOT_FOUND:${responsibility}:${entity.id}`,
        );

        continue;
      }

      if (
        contract.status !==
        "ACTIVE"
      ) {
        blockedReasons.push(
          `CONTRACT_NOT_ACTIVE:${contract.id}`,
        );

        continue;
      }

      authorities.push({
        responsibility,

        entityId:
          entity.id,

        legalName:
          entity.legalName,

        entityType:
          entity.entityType,

        countryCode:
          entity.countryCode,

        contractId:
          contract.id,

        contractStatus:
          contract.status,

        regulatoryReference:
          entity.regulatoryReference,

        operationalStatus:
          entity.operationalStatus,
      });
    }

    const down =
      authorities.filter(
        (a) =>
          a.operationalStatus ===
            "DOWN" ||
          a.operationalStatus ===
            "UNKNOWN",
      );

    for (
      const entity of down
    ) {
      blockedReasons.push(
        `PARTNER_OPERATIONALLY_UNAVAILABLE:${entity.responsibility}:${entity.entityId}`,
      );
    }

    const complete =
      missing.length === 0 &&
      blockedReasons.length === 0;

    return {
      responsibilityMapId:
        map.id,

      policyVersion:
        map.policyVersion,

      sourceCountry:
        args.sourceCountry,

      destinationCountry:
        args.destinationCountry,

      transferClass:
        args.transferClass,

      custodyModel:
        map.custodyModel,

      authorities,

      legalApprovalReference:
        map.legalApprovalReference,

      partnerApprovalReference:
        map.partnerApprovalReference,

      effectiveAt:
        at.toISOString(),

      complete,

      missingResponsibilities:
        missing,

      blockedReasons,
    };
  }
}

import {
  ForbiddenException,
  Injectable,
} from "@nestjs/common";

@Injectable()
export class ResponsibilityValidatorService {
  assertExecutable(
    resolution:
      RegulatedEntityResolution,
  ) {
    if (
      !resolution.complete
    ) {
      throw new ForbiddenException({
        code:
          "REAL_MONEY_RESPONSIBILITY_NOT_COMPLETE",

        responsibilityMapId:
          resolution.responsibilityMapId,

        missingResponsibilities:
          resolution.missingResponsibilities,

        blockedReasons:
          resolution.blockedReasons,
      });
    }

    if (
      !resolution.legalApprovalReference
    ) {
      throw new ForbiddenException(
        "LEGAL_APPROVAL_REFERENCE_MISSING",
      );
    }

    if (
      !resolution.partnerApprovalReference
    ) {
      throw new ForbiddenException(
        "PARTNER_APPROVAL_REFERENCE_MISSING",
      );
    }
  }
}

        "LINQA_BALANCE" &&
      args.custodyModel ===
        "NO_CUSTOMER_FUNDS"
    ) {
      throw new Error(
        "LINQA_BALANCE_REQUIRES_PARTNER_ACCOUNT_OR_STORED_VALUE_MODEL",
      );
    }

    if (
      args.custodyModel ===
        "PARTNER_ISSUED_STORED_VALUE"
    ) {
      const issuer =
        args.resolution.authorities.find(
          (x) =>
            x.responsibility ===
            "STORED_VALUE_ISSUER",
        );

      if (!issuer) {
        throw new Error(
          "STORED_VALUE_ISSUER_MISSING",
        );
      }
    }

    if (
      [
        "PARTNER_CUSTODY",
        "FBO_OR_EQUIVALENT_PARTNER_ACCOUNT",
      ].includes(
        args.custodyModel,
      )
    ) {
      const custody =
        args.resolution.authorities.find(
          (x) =>
            x.responsibility ===
            "CUSTOMER_FUNDS_CUSTODY",
        );

      const safeguard =
        args.resolution.authorities.find(
          (x) =>
            x.responsibility ===
            "SAFEGUARDING",
        );

      if (
        !custody ||
        !safeguard
      ) {
        throw new Error(
          "CUSTOMER_FUNDS_CUSTODY_OR_SAFEGUARDING_MISSING",
        );
      }
    }
  }
}

import {
  Injectable,
} from "@nestjs/common";

@Injectable()
export class ExecutionAuthorityService {
  getAuthority(
    snapshot:
      FinancialResponsibilitySnapshot,

    responsibility:
      FinancialResponsibility,
  ) {
    const authority =
      snapshot.authorities.find(
        (x) =>
          x.responsibility ===
          responsibility,
      );

    if (!authority) {
      throw new Error(
        `EXECUTION_AUTHORITY_MISSING:${responsibility}`,
      );
    }

    return authority;
  }

  getFundingAuthority(
    snapshot:
      FinancialResponsibilitySnapshot,

    fundingMode:
      string,
  ) {
    switch (
      fundingMode
    ) {
      case "DEBIT_CARD":
      case "CREDIT_CARD":
        return this.getAuthority(
          snapshot,
          "CARD_ACQUIRING",
        );

      case "BANK":
        return this.getAuthority(
          snapshot,
          "BANK_DEBIT",
        );

      case "LINQA_BALANCE":
        return this.getAuthority(
          snapshot,
          "CUSTOMER_FUNDS_CUSTODY",
        );

      default:
        return this.getAuthority(
          snapshot,
          "ACCEPT_CUSTOMER_FUNDS",
        );
    }
  }

  getTransmissionAuthority(
    snapshot:
      FinancialResponsibilitySnapshot,
  ) {
    const crossBorder =
      snapshot.transferClass ===
        "INTERNAL_CROSS_BORDER" ||
      snapshot.transferClass ===
        "EXTERNAL_CROSS_BORDER";

    return this.getAuthority(
      snapshot,
      crossBorder
        ? "CROSS_BORDER_TRANSMISSION"
        : "DOMESTIC_TRANSMISSION",
    );
  }

  getPayoutAuthority(
    snapshot:
      FinancialResponsibilitySnapshot,
    receiveMode:
      string,
  ) {
    switch (
      receiveMode
    ) {
      case "BANK":
        return this.getAuthority(
          snapshot,
          "LOCAL_BANK_PAYOUT",
        );

      case "MOBILE_MONEY":
        return this.getAuthority(
          snapshot,
          "MOBILE_MONEY_PAYOUT",
        );

      case "CARD":
        return this.getAuthority(
          snapshot,
          "CARD_PUSH_PAYOUT",
        );

      case "LINQA_BALANCE":
        return this.getAuthority(
          snapshot,
          "CUSTOMER_ACCOUNT_ISSUER",
        );

      default:
        throw new Error(
          "UNSUPPORTED_RECEIVE_MODE",
        );
    }
  }
}

import {

} from "@nestjs/common";

@Injectable()
export class RealMoneyReadinessService {
  constructor(
    private readonly resolver:
      RegulatedEntityResolverService,

    private readonly validator:
      ResponsibilityValidatorService,

    private readonly custody:
      CustodyModelService,

    private readonly compliance:
      ComplianceResponsibilityService,

      transferClass:
        args.transferClass,

      fundingMode:
        args.fundingMode,

      receiveMode:
        args.receiveMode,

      resolution,
    });

    return {
      ready:
        true,

      resolution,

      snapshot:
        this.snapshots.create(
          resolution,
        ),
    };
  }
}

export class RealMoneyExecutionGateV7 {
  constructor(
    private readonly readiness:
      RealMoneyReadinessService,

    private readonly authority:
      ExecutionAuthorityService,

    private readonly v4:
      SecureMoneyMovementGateV4,
  ) {}

  async execute(args: {
    transfer:
      SecureTransferView;

    sender:
      SenderContext;

    recipient:
      ResolvedRecipient;

    fundingSource:
      FundingSource;

    quote:
      PersistedQuote;

    installationId:
      string;

    deviceId:
      string;

    authorizationToken:
      string;
  }) {
    const financial =
      await this.readiness.evaluate({
        sourceCountry:
          args.transfer.sourceCountry,

        destinationCountry:
          args.transfer.destinationCountry,

        transferClass:
          args.transfer.transferClass,

        sourceCurrency:
          args.quote.sourceCurrency,

        destinationCurrency:
          args.quote.destinationCurrency,

        fundingMode:
          args.fundingSource.mode,

        receiveMode:
          args.recipient.receiveMode,
      });

    const fundingAuthority =
      this.authority.getFundingAuthority(
        financial.snapshot,
        args.fundingSource.mode,
      );

    const transmissionAuthority =
      this.authority.getTransmissionAuthority(
        financial.snapshot,
      );

    const payoutAuthority =
      this.authority.getPayoutAuthority(
        financial.snapshot,
        args.recipient.receiveMode,
      );

    const fxAuthority =
      args.quote.sourceCurrency !==
      args.quote.destinationCurrency
        ? this.authority.getAuthority(
            financial.snapshot,
            "FX_EXECUTION",
          )
        : null;

    return this.v4.execute({
      ...args,

      regulatedExecution: {
        financialResponsibilitySnapshot:
          financial.snapshot,

        fundingAuthority,

        transmissionAuthority,

        payoutAuthority,

        fxAuthority,
      },
    } as any);
  }
}

import {
  Injectable,
} from "@nestjs/common";

@Injectable()
export class ContractExpiryWorker {
  constructor(
    private readonly contracts:
      PartnerContractRepository,

    private readonly maps:
      FinancialResponsibilityRepository,

    private readonly alerts:
      OperationsAlertService,
  ) {}

  async run() {
    const expiring =
      await this.contracts.listExpiringWithinDays(
        30,
      );

    for (
      const contract
      of expiring
    ) {
      await this.alerts.raise({
        type:
          "LINQASEND_PARTNER_CONTRACT_EXPIRING",

        contractId:
          contract.id,

        entityId:
          contract.regulatedEntityId,

        effectiveTo:
          contract.effectiveTo,
      });
    }

    const expired =
      await this.contracts.listExpired();

    for (
      const contract
      of expired
    ) {
      await this.maps.suspendMapsUsingContract(
        contract.id,
      );
    }
  }
}

import {
  Injectable,
} from "@nestjs/common";

@Injectable()
export class PartnerHealthWorker {
  constructor(
    private readonly entities:
      EntityRegistryRepository,

    private readonly checks:
      PartnerHealthCheckRegistry,

    private readonly health:
      PartnerHealthRepository,
  ) {}

  async run() {
    const entities =
      await this.entities.listActive();

    for (
      const entity
      of entities
    ) {
      const checker =
        this.checks.get(
          entity.id,
        );

      if (!checker) {
        await this.health.record({
          entityId:
            entity.id,

          capability:
            "UNKNOWN",

          status:
            "UNKNOWN",

          evidence: {
            reason:
              "HEALTH_CHECK_NOT_CONFIGURED",
          },
        });

        continue;
      }

      const result =
        await checker.check();

      await this.health.record({
        entityId:
          entity.id,

        capability:
          result.capability,

        status:
          result.status,

        evidence:
          result.evidence,
      });
    }
  }
}

export class RealMoneyReleaseGateV7 {
  constructor(
    private readonly maps:
      FinancialResponsibilityRepository,

    private readonly readiness:
      RealMoneyReadinessService,

    private readonly certifications:
      CorridorCertificationRepository,
  ) {}

  async assertCorridorReady(
    corridorId: string,
  ) {
    const cert =
      await this.certifications.findById(
        corridorId,
      );

    if (
      !cert ||
      cert.status !==
        "CERTIFIED"
    ) {
      throw new Error(
        "CORRIDOR_NOT_CERTIFIED",
      );
    }

    const map =
      await this.maps.findByCorridorId(
        corridorId,
      );

    if (!map) {
      throw new Error(
        "FINANCIAL_RESPONSIBILITY_MAP_MISSING",
      );
    }

    const result =
      await this.readiness.evaluate({
        sourceCountry:
          map.sourceCountry,

        destinationCountry:
          map.destinationCountry,

        transferClass:
          map.transferClass,

        sourceCurrency:
          map.sourceCurrency,

        destinationCurrency:
          map.destinationCurrency,

        fundingMode:
          map.fundingMode,

        receiveMode:
          map.receiveMode,
      });

    if (!result.ready) {
      throw new Error(
        "REAL_MONEY_READINESS_NOT_GREEN",
      );
    }

    return true;
  }
}
