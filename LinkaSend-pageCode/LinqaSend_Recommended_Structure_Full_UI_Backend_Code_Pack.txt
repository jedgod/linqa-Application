@Injectable()
export class RecipientResolverService {
  constructor(
    private readonly usersRepo: UsersRepository,
    private readonly externalRepo: ExternalRecipientsRepository,
  ) {}

  async resolveLinqaUser(senderUserId: string, query: string): Promise<ResolvedRecipient | null> {
    const user = await this.usersRepo.findVerifiedByHandlePhoneOrEmail(query);
    if (!user || user.id === senderUserId) return null;
    return {
      recipientId: user.id,
      type: 'LINQA_USER',
      displayName: user.displayName,
      countryCode: user.countryCode,
      currency: user.preferredCurrency,
      deliveryMethods: ['LINQA_PROFILE'],
      preferredDeliveryMethod: 'LINQA_PROFILE',
    };
  }

  async resolveExternal(senderUserId: string, externalRecipientId: string) {
    const recipient = await this.externalRepo.findOwnedBy(senderUserId, externalRecipientId);
    if (!recipient) return null;
    const endpoints = await this.externalRepo.listActiveEndpoints(recipient.id);
    return {
      recipientId: recipient.id,
      type: recipient.recipientKind,
      displayName: recipient.displayName,
      countryCode: recipient.countryCode,
      currency: recipient.currency,
      deliveryMethods: endpoints.map((e) => e.deliveryMethod),
      preferredDeliveryMethod: endpoints[0]?.deliveryMethod,
    };
  }
}

@Injectable()
export class FxEngineService {
  constructor(
    private readonly rateProvider: MarketRateProvider,
    private readonly pricingRepo: FxPricingRepository,
  ) {}

  async quote(input: {
    sourceCountry: string;
    destinationCountry: string;
    sendCurrency: string;
    receiveCurrency: string;
    sendAmount: number;
  }) {
    if (input.sendAmount <= 0) throw new BadRequestException('INVALID_AMOUNT');

    if (input.sendCurrency === input.receiveCurrency) {
      return {
        marketReferenceRate: 1,
        providerExecutionRate: 1,
        customerRate: 1,
        marginBps: 0,
        receiveAmount: input.sendAmount,
        senderFee: 0,
        recipientFee: 0,
        expiresAt: new Date(Date.now() + 30000).toISOString(),
      };
    }

    const market = await this.rateProvider.getRate(input.sendCurrency, input.receiveCurrency);
    const rule = await this.pricingRepo.getActiveRule(input);
    if (!rule) throw new BadRequestException('UNSUPPORTED_CORRIDOR');

    const rawMargin =
      rule.baseMarginBps +
      rule.volatilityBufferBps +
      rule.liquidityBufferBps +
      rule.settlementBufferBps -
      rule.promotionAdjustmentBps;

    const marginBps = Math.min(rule.maxMarginBps, Math.max(rule.minMarginBps, rawMargin));
    const customerRate = Number((market.providerRate * (1 - marginBps / 10000)).toFixed(8));

    return {
      marketReferenceRate: market.marketRate,
      providerExecutionRate: market.providerRate,
      customerRate,
      marginBps,
      receiveAmount: Number((input.sendAmount * customerRate).toFixed(2)),
      senderFee: 0,
      recipientFee: 0,
      expiresAt: new Date(Date.now() + 30000).toISOString(),
    };
  }
}

@Injectable()
export class RiskEngineService {
  async assess(input: {
    recipientIsNew: boolean;
    paymentMethodAgeDays: number;
    accountAgeDays: number;
    recentPasswordReset: boolean;
    recentDeviceChange: boolean;
    velocity24h: number;
  }) {
    let score = 0;
    const reasons: string[] = [];

    if (input.recipientIsNew) { score += 15; reasons.push('NEW_RECIPIENT'); }
    if (input.paymentMethodAgeDays < 1) { score += 20; reasons.push('NEW_PAYMENT_METHOD'); }
    if (input.accountAgeDays < 7) { score += 15; reasons.push('YOUNG_ACCOUNT'); }
    if (input.recentPasswordReset) { score += 20; reasons.push('RECENT_PASSWORD_RESET'); }
    if (input.recentDeviceChange) { score += 20; reasons.push('RECENT_DEVICE_CHANGE'); }
    if (input.velocity24h >= 5) { score += 20; reasons.push('HIGH_TRANSFER_VELOCITY'); }

    let decision: RiskDecision = 'ALLOW';
    if (score >= 80) decision = 'DECLINE';
    else if (score >= 60) decision = 'TEMPORARY_HOLD';
    else if (score >= 40) decision = 'REVIEW';
    else if (score >= 20) decision = 'ALLOW_STEP_UP';

    return { score, decision, reasons };
  }
}

@Injectable()
export class ComplianceEngineService {
  constructor(
    private readonly kycProvider: KycProvider,
    private readonly sanctionsProvider: SanctionsProvider,
    private readonly amlProvider: AmlProvider,
  ) {}

  async screenTransfer(input: any) {
    const [senderKyc, sanctions, aml] = await Promise.all([
      this.kycProvider.checkUser(input.senderUserId),
      this.sanctionsProvider.screenRecipient(input.recipientId),
      this.amlProvider.evaluateTransfer(input),
    ]);

    if (!senderKyc.passed) return { status: 'FAILED', reason: 'KYC_REQUIRED' };
    if (sanctions.match) return { status: 'REVIEW', reason: 'SANCTIONS_REVIEW' };
    if (aml.reviewRequired) return { status: 'REVIEW', reason: 'AML_REVIEW' };
    return { status: 'PASSED', reason: null };
  }
}

@Injectable()
export class LedgerService {
  constructor(private readonly db: DatabaseService) {}

  async postTransfer(input: {
    transferId: string;
    currency: string;
    amount: number;
    fundingClearingAccountId: string;
    transferPayableAccountId: string;
  }) {
    const postings = [
      { accountId: input.fundingClearingAccountId, direction: 'DEBIT', amount: input.amount, currency: input.currency },
      { accountId: input.transferPayableAccountId, direction: 'CREDIT', amount: input.amount, currency: input.currency },
    ] as const;

    const debit = postings.filter(x => x.direction === 'DEBIT').reduce((a, x) => a + x.amount, 0);
    const credit = postings.filter(x => x.direction === 'CREDIT').reduce((a, x) => a + x.amount, 0);
    if (Math.abs(debit - credit) > 0.000001) throw new Error('UNBALANCED_LEDGER');

    await this.db.transaction(async (tx: any) => {
      const ledgerTx = await tx.ledgerTransactions.create({
        referenceType: 'LINQASEND_TRANSFER',
        referenceId: input.transferId,
        description: 'LinqaSend funding authorization',
      });
      for (const posting of postings) {
        await tx.ledgerEntries.create({ transactionId: ledgerTx.id, ...posting });
      }
    });
  }
}

@Injectable()
export class TransferOrchestratorService {
  constructor(
    private readonly transfersRepo: TransfersRepository,
    private readonly quotesRepo: QuotesRepository,
    private readonly risk: RiskEngineService,
    private readonly compliance: ComplianceEngineService,
    private readonly payments: PaymentProviderRegistry,
    private readonly rails: RailRouterService,
    private readonly ledger: LedgerService,
    private readonly notifications: NotificationService,
  ) {}

  async create(input: any) {
    const existing = await this.transfersRepo.findByIdempotency(input.userId, input.idempotencyKey);
    if (existing) return existing;

    const quote = await this.quotesRepo.findValidOwned(input.userId, input.quoteId);
    if (!quote) throw new BadRequestException('QUOTE_EXPIRED_OR_INVALID');

    return this.transfersRepo.create({
      ...input,
      sendAmount: quote.sendAmount,
      sendCurrency: quote.sendCurrency,
      receiveAmount: quote.receiveAmount,
      receiveCurrency: quote.receiveCurrency,
      customerRate: quote.customerRate,
      status: 'AWAITING_CONFIRMATION',
    });
  }

  async confirm(input: any) {
    const transfer = await this.transfersRepo.findOwned(input.userId, input.transferId);
    if (!transfer) throw new ForbiddenException();

    await verifySingleUseTransferAuthorizationToken({
      token: input.authorizationToken,
      userId: input.userId,
      transferId: input.transferId,
      amount: transfer.sendAmount,
      currency: transfer.sendCurrency,
      recipientId: transfer.recipientId,
    });

    await this.transfersRepo.updateStatus(transfer.id, 'COMPLIANCE_SCREENING');
    const complianceResult = await this.compliance.screenTransfer({
      senderUserId: input.userId,
      recipientId: transfer.recipientId,
      amount: transfer.sendAmount,
      currency: transfer.sendCurrency,
      destinationCountry: transfer.destinationCountry,
    });

    if (complianceResult.status === 'FAILED') {
      await this.transfersRepo.updateStatus(transfer.id, 'DECLINED');
      return this.transfersRepo.findById(transfer.id);
    }
    if (complianceResult.status === 'REVIEW') {
      await this.transfersRepo.updateStatus(transfer.id, 'MANUAL_REVIEW');
      return this.transfersRepo.findById(transfer.id);
    }

    await this.transfersRepo.updateStatus(transfer.id, 'RISK_REVIEW');
    const riskResult = await this.risk.assess(transfer.riskInput);
    if (riskResult.decision === 'DECLINE') {
      await this.transfersRepo.updateStatus(transfer.id, 'DECLINED');
      return this.transfersRepo.findById(transfer.id);
    }
    if (riskResult.decision === 'REVIEW' || riskResult.decision === 'TEMPORARY_HOLD') {
      await this.transfersRepo.updateStatus(transfer.id, 'ON_HOLD');
      return this.transfersRepo.findById(transfer.id);
    }

    await this.transfersRepo.updateStatus(transfer.id, 'FUNDING_AUTHORIZATION');
    const paymentProvider = this.payments.forPaymentMethod(transfer.paymentMethod);
    const funding = await paymentProvider.authorizeFunding({
      transferId: transfer.id,
      amount: transfer.sendAmount,
      currency: transfer.sendCurrency,
      providerPaymentToken: transfer.paymentMethod.providerToken,
      customerId: input.userId,
      idempotencyKey: `${transfer.id}:fund`,
    });

    if (funding.status === 'REQUIRES_ACTION') {
      await this.transfersRepo.updateStatus(transfer.id, 'REQUIRES_ACTION');
      return { ...(await this.transfersRepo.findById(transfer.id)), clientActionToken: funding.clientActionToken };
    }
    if (funding.status === 'DECLINED') {
      await this.transfersRepo.updateStatus(transfer.id, 'DECLINED');
      return this.transfersRepo.findById(transfer.id);
    }

    await this.transfersRepo.updateStatus(transfer.id, 'FUNDS_AUTHORIZED');
    await this.ledger.postTransfer({
      transferId: transfer.id,
      currency: transfer.sendCurrency,
      amount: transfer.sendAmount,
      fundingClearingAccountId: transfer.fundingClearingAccountId,
      transferPayableAccountId: transfer.transferPayableAccountId,
    });

    const rail = this.rails.choose({
      countryCode: transfer.destinationCountry,
      currency: transfer.receiveCurrency,
      deliveryMethod: transfer.deliveryMethod,
    });

    await this.transfersRepo.updateStatus(transfer.id, 'PAYOUT_SUBMITTED');
    const payout = await rail.payout({
      transferId: transfer.id,
      amount: transfer.receiveAmount,
      currency: transfer.receiveCurrency,
      countryCode: transfer.destinationCountry,
      deliveryMethod: transfer.deliveryMethod,
      providerRecipientToken: transfer.providerRecipientToken,
      idempotencyKey: `${transfer.id}:payout`,
    });

    await this.transfersRepo.savePayoutReference(transfer.id, rail.name, payout.providerPayoutId);
    await this.transfersRepo.updateStatus(transfer.id, payout.status === 'COMPLETED' ? 'COMPLETED' : 'PAYOUT_PROCESSING');
    await this.notifications.transferStatusChanged(transfer.id);
    return this.transfersRepo.findById(transfer.id);
  }
}

@Injectable()
export class ReconciliationService {
  constructor(
    private readonly ledgerRepo: LedgerRepository,
    private readonly providers: ReconciliationProviderRegistry,
    private readonly reconciliationRepo: ReconciliationRepository,
  ) {}

  async reconcileProvider(providerName: string, date: string) {
    const provider = this.providers.get(providerName);
    const [expected, providerReport] = await Promise.all([
      this.ledgerRepo.sumSettlementExpected(providerName, date),
      provider.fetchSettlementReport(date),
    ]);
    const delta = Number((providerReport.totalAmount - expected.totalAmount).toFixed(2));
    const status = Math.abs(delta) < 0.01 ? 'MATCHED' : 'MISMATCH';
    return this.reconciliationRepo.create({
      provider: providerName,
      reconciliationDate: date,
      expectedAmount: expected.totalAmount,
      providerAmount: providerReport.totalAmount,
      deltaAmount: delta,
      status,
      details: { expectedCount: expected.count, providerCount: providerReport.count },
    });
  }
}

    private readonly transferRepo: TransfersRepository,
    private readonly push: PushNotificationProvider,
  ) {}

  async transferStatusChanged(transferId: string) {
    const transfer = await this.transferRepo.findById(transferId);
    if (!transfer) return;
    const template = {
      COMPLETED: { title: 'Transfer completed', body: 'Your LinqaSend transfer was successfully delivered.' },
      PAYOUT_PROCESSING: { title: 'Transfer is on the way', body: 'Your recipient payout is processing.' },
      ON_HOLD: { title: 'Transfer under review', body: 'Your transfer needs additional review.' },
      FAILED: { title: 'Transfer failed', body: 'Your transfer could not be completed.' },
    }[transfer.status];
    if (template) await this.push.sendToUser(transfer.userId, template);
  }
}

export async function authenticateAndSend() {
  const result = await LocalAuthentication.authenticateAsync({
    promptMessage: 'Authorize LinqaSend transfer',
    fallbackLabel: 'Use device passcode',
  });
  if (!result.success) return;
  router.replace('/(linqasend)/send/tracking');
}

================================================================================
SECURITY RULES
================================================================================

- Never store raw PAN/CVV in Linqa database or logs.
- Use provider-hosted tokenization.
- Require idempotency keys for create/confirm/refund/payout retry.
- Server is authoritative for FX quote, fees, risk, compliance, payment state, payout state and receipt.
- Require step-up authentication for transfers.
- Bind authorization token to user, transfer, recipient, amount, currency, quote and expiration.
- Verify all provider webhook signatures and prevent replay.
- Ledger is immutable; corrections use reversals.
- Social/chat domains may initiate payment intent but may not access financial credentials or compliance results.
- External recipient payout details should be stored as provider tokens and masked metadata.
