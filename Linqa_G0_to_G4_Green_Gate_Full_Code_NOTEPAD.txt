package com.linqa.security

import android.security.keystore.KeyGenParameterSpec
import android.security.keystore.KeyProperties
import java.security.KeyPairGenerator
import java.security.KeyStore
import java.security.Signature

class AndroidKeyCustody {
    private val keyStore = KeyStore.getInstance("AndroidKeyStore").apply { load(null) }

    fun createSigningKey(alias: String) {
        if (keyStore.containsAlias(alias)) return
        val generator = KeyPairGenerator.getInstance(
            KeyProperties.KEY_ALGORITHM_EC,
            "AndroidKeyStore"
        )
        val spec = KeyGenParameterSpec.Builder(
            alias,
            KeyProperties.PURPOSE_SIGN or KeyProperties.PURPOSE_VERIFY
        )
            .setDigests(KeyProperties.DIGEST_SHA256)
            .setUserAuthenticationRequired(false)
            .build()
        generator.initialize(spec)
        generator.generateKeyPair()
    }

    fun sign(alias: String, payload: ByteArray): ByteArray {
        val entry = keyStore.getEntry(alias, null) as KeyStore.PrivateKeyEntry
        val signature = Signature.getInstance("SHA256withECDSA")
        signature.initSign(entry.privateKey)
        signature.update(payload)
        return signature.sign()
    }

    fun publicKey(alias: String): ByteArray {
        val entry = keyStore.getEntry(alias, null) as KeyStore.PrivateKeyEntry
        return entry.certificate.publicKey.encoded
    }

    fun delete(alias: String) {
        keyStore.deleteEntry(alias)
    }
}

import Foundation
import Security

final class IOSKeyCustody {
    func createSigningKey(tag: String) throws -> SecKey {
        let tagData = Data(tag.utf8)
        let attrs: [String: Any] = [
            kSecAttrKeyType as String: kSecAttrKeyTypeECSECPrimeRandom,
            kSecAttrKeySizeInBits as String: 256,
            kSecAttrTokenID as String: kSecAttrTokenIDSecureEnclave,
            kSecPrivateKeyAttrs as String: [
                kSecAttrIsPermanent as String: true,
                kSecAttrApplicationTag as String: tagData
            ]
        ]

        var error: Unmanaged<CFError>?
        guard let key = SecKeyCreateRandomKey(attrs as CFDictionary, &error) else {
            throw error!.takeRetainedValue() as Error
        }
        return key
    }

    func sign(privateKey: SecKey, payload: Data) throws -> Data {
        var error: Unmanaged<CFError>?
        let algorithm = SecKeyAlgorithm.ecdsaSignatureMessageX962SHA256
        guard SecKeyIsAlgorithmSupported(privateKey, .sign, algorithm) else {
            throw NSError(domain: "Linqa", code: 1)
        }
        guard let sig = SecKeyCreateSignature(
            privateKey,
            algorithm,
            payload as CFData,
            &error
        ) else {
            throw error!.takeRetainedValue() as Error
        }
        return sig as Data
    }
}

export interface CoreBridge {
  isReady(): Promise<boolean>;
  encryptMessage(input: {
    conversationId: string;
    recipientDeviceId: string;
    plaintext: Uint8Array;
    aad: Uint8Array;
  }): Promise<Uint8Array>;
  decryptMessage(input: {
    envelope: Uint8Array;
    aad: Uint8Array;
  }): Promise<Uint8Array>;
  verifyTransparency(input: unknown): Promise<boolean>;
}

export class ProductionCoreBridge {
  constructor(private readonly wasm: any) {}

  async isReady() {
    return !!this.wasm?.encrypt_message && !!this.wasm?.decrypt_message;
  }

  async encryptMessage(input: any) {
    if (!(await this.isReady())) throw new Error("SECURE_CORE_UNAVAILABLE");
    return this.wasm.encrypt_message(
      input.conversationId,
      input.recipientDeviceId,
      input.plaintext,
      input.aad,
    );
  }

  async decryptMessage(input: any) {
    if (!(await this.isReady())) throw new Error("SECURE_CORE_UNAVAILABLE");
    return this.wasm.decrypt_message(input.envelope, input.aad);
  }

  async verifyTransparency(input: any) {
    if (!this.wasm?.verify_transparency) throw new Error("TRANSPARENCY_UNAVAILABLE");
    return this.wasm.verify_transparency(input);
  }
}

export function createProductionCoreBridge(env: {
  mode: "development" | "staging" | "production";
  allowDemoCore: boolean;
}, wasm: any) {
  if (env.mode === "production" && env.allowDemoCore) {
    throw new Error("DEMO_CORE_FORBIDDEN_IN_PRODUCTION");
  }
  return new ProductionCoreBridge(wasm);
}

export type OutboxState =
  | "CREATED"
  | "ENCRYPTED"
  | "QUEUED"
  | "SENDING"
  | "RELAY_ACCEPTED"
  | "ACKNOWLEDGED"
  | "FAILED_RETRYABLE"
  | "FAILED_PERMANENT";

export interface OutboxItem {
  id: string;
  conversationId: string;
  recipientDeviceId: string;
  ciphertext: Uint8Array;
  aad: Uint8Array;
  state: OutboxState;
  retryCount: number;
  nextAttemptAt: number;
  createdAt: number;
}

export interface OutboxRepo {
  put(item: OutboxItem): Promise<void>;
  update(id: string, patch: Partial<OutboxItem>): Promise<void>;
  due(now: number, limit: number): Promise<OutboxItem[]>;
  get(id: string): Promise<OutboxItem | null>;
}

export class DurableOutbox {
  constructor(private readonly repo: OutboxRepo) {}

  async enqueue(item: OutboxItem) {
    await this.repo.put(item);
  }

  async markRetry(item: OutboxItem, reason: string) {
    const retryCount = item.retryCount + 1;
    if (retryCount >= 10) {
      await this.repo.update(item.id, { state: "FAILED_PERMANENT", retryCount });
      return;
    }
    const backoff = Math.min(60_000, 1000 * 2 ** retryCount);
    await this.repo.update(item.id, {
      state: "FAILED_RETRYABLE",
      retryCount,
      nextAttemptAt: Date.now() + backoff + Math.floor(Math.random() * 500),
    });
  }
}

export class ReceiveWorker {
  private running = false;

  constructor(
    private readonly relay: {
      lease(): Promise<Array<any>>;
      ack(input: { messageId: string; leaseId: string }): Promise<void>;
    },
    private readonly core: any,
    private readonly inbox: {
      has(envelopeId: string): Promise<boolean>;
      put(x: any): Promise<void>;
    },
  ) {}

  async runOnce() {
    const leased = await this.relay.lease();
    for (const msg of leased) {
      if (await this.inbox.has(msg.envelopeId)) {
        await this.relay.ack({ messageId: msg.messageId, leaseId: msg.leaseId });
        continue;
      }

      const plaintext = await this.core.decryptMessage({
        envelope: Buffer.from(msg.ciphertext, "base64"),
        aad: Buffer.from(msg.aad, "base64"),
      });

      await this.inbox.put({
        envelopeId: msg.envelopeId,
        plaintext: Buffer.from(plaintext).toString("utf8"),
        receivedAt: Date.now(),
      });

      await this.relay.ack({ messageId: msg.messageId, leaseId: msg.leaseId });
    }
  }

  async start(intervalMs = 3000) {
    if (this.running) return;
    this.running = true;
    while (this.running) {
      try { await this.runOnce(); } catch {}
      await new Promise(r => setTimeout(r, intervalMs));
    }
  }

  stop() { this.running = false; }
}

import crypto from "node:crypto";

export class SendCoordinator {
  constructor(
    private readonly core: any,
    private readonly outbox: any,
    private readonly relay: { send(x: any): Promise<{ accepted: boolean }> },
  ) {}

  async send(input: {
    conversationId: string;
    recipientDeviceId: string;
    plaintext: string;
    transparencyProof: unknown;
  }) {
    if (!(await this.core.verifyTransparency(input.transparencyProof))) {
      throw new Error("TRANSPARENCY_VERIFICATION_FAILED");
    }

    const id = `env_${crypto.randomBytes(18).toString("base64url")}`;
    const aad = Buffer.from(JSON.stringify({
      envelopeId: id,
      conversationId: input.conversationId,
      recipientDeviceId: input.recipientDeviceId,
    }));

    const ciphertext = await this.core.encryptMessage({
      conversationId: input.conversationId,
      recipientDeviceId: input.recipientDeviceId,
      plaintext: Buffer.from(input.plaintext, "utf8"),
      aad,
    });

    const item = {
      id,
      conversationId: input.conversationId,
      recipientDeviceId: input.recipientDeviceId,
      ciphertext,
      aad,
      state: "QUEUED",
      retryCount: 0,
      nextAttemptAt: Date.now(),
      createdAt: Date.now(),
    };

    await this.outbox.enqueue(item);
    const accepted = await this.relay.send({
      envelopeId: id,
      recipientDeviceId: input.recipientDeviceId,
      ciphertext: Buffer.from(ciphertext).toString("base64"),
      aad: Buffer.from(aad).toString("base64"),
    });

    if (!accepted.accepted) throw new Error("RELAY_REJECTED");
    return { envelopeId: id };
  }
}

import fs from "node:fs/promises";
import path from "node:path";
import crypto from "node:crypto";
import { GateEvidence, GateName, GateCheck } from "./types";

export class GateEvidenceWriter {
  constructor(
    private readonly rootDir: string,
    private readonly revision: string,
    private readonly environment: GateEvidence["environment"],
  ) {}

  async write(gate: GateName, checks: GateCheck[]) {
    const required = checks.filter(x => x.required);
    const passed = required.filter(x => x.passed);
    const failed = required.length - passed.length;

    const evidence: GateEvidence = {
      gate,
      status: failed === 0 ? "GREEN" : passed.length ? "AMBER" : "RED",
      generatedAt: new Date().toISOString(),
      revision: this.revision,
      environment: this.environment,
      checks,
      summary: {
        required: required.length,
        passed: passed.length,
        failed,
      },
    };

    const dir = path.join(this.rootDir, gate);
    await fs.mkdir(dir, { recursive: true });
    const file = path.join(dir, "gate-evidence.json");
    await fs.writeFile(file, JSON.stringify(evidence, null, 2));

    const digest = crypto
      .createHash("sha256")
      .update(JSON.stringify(evidence))
      .digest("hex");

    await fs.writeFile(
      path.join(dir, "gate-evidence.sha256"),
      `${digest}  gate-evidence.json\n`,
    );

    return evidence;
  }
}

import fs from "node:fs/promises";
import path from "node:path";
import { GateEvidence, GateName } from "./types";

const REQUIRED: GateName[] = ["G0", "G1", "G2", "G3", "G4"];

export async function assertG0ToG4Green(rootDir: string) {
  const failures: string[] = [];

  for (const gate of REQUIRED) {
    const file = path.join(rootDir, gate, "gate-evidence.json");
    let evidence: GateEvidence;
    try {
      evidence = JSON.parse(await fs.readFile(file, "utf8"));
    } catch {
      failures.push(`${gate}:MISSING_EVIDENCE`);
      continue;
    }

    if (evidence.status !== "GREEN") {
      failures.push(`${gate}:${evidence.status}`);
    }

    if (!evidence.revision) {
      failures.push(`${gate}:MISSING_REVISION`);
    }

    if (evidence.environment === "staging" && gate === "G3") {
      failures.push(`${gate}:NATIVE_REQUIRES_PRODUCTION_EQUIVALENT_OR_PRODUCTION`);
    }
  }

  if (failures.length) {
    throw new Error(`LINQA_RELEASE_BLOCKED:${failures.join(",")}`);
  }
}

import crypto from "node:crypto";

export interface EnrollmentRepository {
  createChallenge(input: {
    id: string;
    userId: string;
    deviceId: string;
    nonce: string;
    expiresAt: Date;
  }): Promise<void>;
  getChallenge(id: string): Promise<any>;
  consumeChallenge(id: string): Promise<boolean>;
  saveEnrolledDevice(input: {
    userId: string;
    deviceId: string;
    identityPublicKeyPem: string;
    signingPublicKeyPem: string;
    enrolledAt: Date;
  }): Promise<void>;
}

export interface EnrollmentVerifier {
  verifySignature(input: {
    publicKeyPem: string;
    payload: Buffer;
    signature: Buffer;
  }): boolean;
}

export class DeviceEnrollmentService {
  constructor(
    private readonly repo: EnrollmentRepository,
    private readonly verifier: EnrollmentVerifier,
  ) {}

  async begin(userId: string, deviceId: string) {
    const id = crypto.randomUUID();
    const nonce = crypto.randomBytes(32).toString("base64url");
    await this.repo.createChallenge({
      id, userId, deviceId, nonce,
      expiresAt: new Date(Date.now() + 5 * 60_000),
    });
    return { challengeId: id, nonce };
  }

  async complete(input: {
    challengeId: string;
    userId: string;
    deviceId: string;
    identityPublicKeyPem: string;
    signingPublicKeyPem: string;
    signatureBase64: string;
  }) {
    const c = await this.repo.getChallenge(input.challengeId);
    if (!c) throw new Error("ENROLLMENT_CHALLENGE_NOT_FOUND");
    if (c.userId !== input.userId || c.deviceId !== input.deviceId) {
      throw new Error("ENROLLMENT_BINDING_MISMATCH");
    }
    if (new Date(c.expiresAt) <= new Date()) throw new Error("ENROLLMENT_CHALLENGE_EXPIRED");

    const payload = Buffer.from(
      JSON.stringify({
        userId: input.userId,
        deviceId: input.deviceId,
        nonce: c.nonce,
        identityPublicKeyPem: input.identityPublicKeyPem,
        signingPublicKeyPem: input.signingPublicKeyPem,
      }),
      "utf8",
    );

    const ok = this.verifier.verifySignature({
      publicKeyPem: input.signingPublicKeyPem,
      payload,
      signature: Buffer.from(input.signatureBase64, "base64"),
    });

    if (!ok) throw new Error("ENROLLMENT_SIGNATURE_INVALID");
    if (!(await this.repo.consumeChallenge(input.challengeId))) {
      throw new Error("ENROLLMENT_CHALLENGE_REPLAY");
    }

    await this.repo.saveEnrolledDevice({
      userId: input.userId,
      deviceId: input.deviceId,
      identityPublicKeyPem: input.identityPublicKeyPem,
      signingPublicKeyPem: input.signingPublicKeyPem,
      enrolledAt: new Date(),
    });

    return { enrolled: true };
  }
}

import crypto from "node:crypto";

export interface CanonicalKeyPackageV1 {
  version: 1;
  userId: string;
  deviceId: string;
  algorithm: "X25519_ED25519_MLKEM768_V1";
  identityKey: string;
  preKey: string;
  pqPreKey: string;
  createdAt: string;
  expiresAt: string;
  nonce: string;
  signature: string;
}

export interface EnrolledDeviceRepo {
  get(userId: string, deviceId: string): Promise<{
    signingPublicKeyPem: string;
    revokedAt?: Date;
  } | null>;
}

export interface KeyPackageRepo {
  insert(pkg: CanonicalKeyPackageV1, digest: string): Promise<void>;
}

function canonicalPayload(pkg: CanonicalKeyPackageV1) {
  const { signature, ...unsigned } = pkg;
  return Buffer.from(JSON.stringify(unsigned), "utf8");
}

export class KeyPackageService {
  constructor(
    private readonly devices: EnrolledDeviceRepo,
    private readonly repo: KeyPackageRepo,
  ) {}

  async verifyAndStore(pkg: CanonicalKeyPackageV1) {
    if (pkg.version !== 1) throw new Error("KEY_PACKAGE_VERSION_UNSUPPORTED");
    if (pkg.algorithm !== "X25519_ED25519_MLKEM768_V1") {
      throw new Error("KEY_PACKAGE_ALGORITHM_UNSUPPORTED");
    }

    const device = await this.devices.get(pkg.userId, pkg.deviceId);
    if (!device) throw new Error("DEVICE_NOT_ENROLLED");
    if (device.revokedAt) throw new Error("DEVICE_REVOKED");

    const now = Date.now();
    const created = Date.parse(pkg.createdAt);
    const expires = Date.parse(pkg.expiresAt);
    if (!Number.isFinite(created) || !Number.isFinite(expires)) throw new Error("KEY_PACKAGE_TIME_INVALID");
    if (expires <= now) throw new Error("KEY_PACKAGE_EXPIRED");
    if (created > now + 60_000) throw new Error("KEY_PACKAGE_CREATED_IN_FUTURE");
    if (expires - created > 7 * 24 * 3600_000) throw new Error("KEY_PACKAGE_LIFETIME_TOO_LONG");

    const ok = crypto.verify(
      null,
      canonicalPayload(pkg),
      device.signingPublicKeyPem,
      Buffer.from(pkg.signature, "base64"),
    );
    if (!ok) throw new Error("KEY_PACKAGE_SIGNATURE_INVALID");

    const digest = crypto.createHash("sha256").update(canonicalPayload(pkg)).digest("hex");
    await this.repo.insert(pkg, digest);
    return { accepted: true, digest };
  }
}

export interface RelayRepo {
  enqueue(input: {
    envelopeId: string;
    senderUserId: string;
    senderDeviceId: string;
    recipientUserId: string;
    recipientDeviceId: string;
    ciphertext: Buffer;
    aad: Buffer;
  }): Promise<{ messageId: string }>;

  lease(input: {
    recipientUserId: string;
    recipientDeviceId: string;
    leaseOwner: string;
    leaseUntil: Date;
    limit: number;
  }): Promise<any[]>;

  ack(input: {
    messageId: string;
    leaseId: string;
    recipientUserId: string;
    recipientDeviceId: string;
  }): Promise<boolean>;
}

export class RelayService {
  constructor(private readonly repo: RelayRepo) {}

  async enqueue(auth: { userId: string; deviceId: string }, body: any) {
    for (const forbidden of ["plaintext", "text", "body", "message"]) {
      if (forbidden in body) throw new Error("PLAINTEXT_MESSAGES_FORBIDDEN");
    }
    if (!body.ciphertext || !body.aad) throw new Error("CIPHERTEXT_REQUIRED");

    return this.repo.enqueue({
      envelopeId: body.envelopeId,
      senderUserId: auth.userId,
      senderDeviceId: auth.deviceId,
      recipientUserId: body.recipientUserId,
      recipientDeviceId: body.recipientDeviceId,
      ciphertext: Buffer.from(body.ciphertext, "base64"),
      aad: Buffer.from(body.aad, "base64"),
    });
  }

  async lease(auth: { userId: string; deviceId: string }) {
    return this.repo.lease({
      recipientUserId: auth.userId,
      recipientDeviceId: auth.deviceId,
      leaseOwner: `${auth.userId}:${auth.deviceId}`,
      leaseUntil: new Date(Date.now() + 60_000),
      limit: 50,
    });
  }

  async ack(auth: { userId: string; deviceId: string }, input: any) {
    const ok = await this.repo.ack({
      messageId: input.messageId,
      leaseId: input.leaseId,
      recipientUserId: auth.userId,
      recipientDeviceId: auth.deviceId,
    });
    if (!ok) throw new Error("ACK_FORBIDDEN_OR_STALE");
  }
}

import crypto from "node:crypto";
import fs from "node:fs/promises";

export interface Db {
  query(sql: string, params?: any[]): Promise<{ rows: any[] }>;
}

export class MigrationRunner {
  constructor(private readonly db: Db) {}

  async ensureLedger() {
    await this.db.query(`
      CREATE TABLE IF NOT EXISTS schema_migrations (
        version TEXT PRIMARY KEY,
        checksum TEXT NOT NULL,
        applied_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
        execution_ms INTEGER NOT NULL,
        application_version TEXT NOT NULL,
        dirty BOOLEAN NOT NULL DEFAULT FALSE
      )
    `);
  }

  async apply(file: string, version: string, appVersion: string) {
    await this.ensureLedger();
    const sql = await fs.readFile(file, "utf8");
    const checksum = crypto.createHash("sha256").update(sql).digest("hex");

    await this.db.query(`SELECT pg_advisory_lock(hashtext('linqa_schema_migrations'))`);
    try {
      const existing = await this.db.query(
        `SELECT checksum, dirty FROM schema_migrations WHERE version=$1`,
        [version],
      );

      if (existing.rows.length) {
        const row = existing.rows[0];
        if (row.dirty) throw new Error(`MIGRATION_DIRTY:${version}`);
        if (row.checksum !== checksum) throw new Error(`MIGRATION_CHECKSUM_MISMATCH:${version}`);
        return;
      }

      const started = Date.now();
      await this.db.query("BEGIN");
      try {
        await this.db.query(
          `INSERT INTO schema_migrations(version,checksum,execution_ms,application_version,dirty)
           VALUES($1,$2,0,$3,true)`,
          [version, checksum, appVersion],
        );
        await this.db.query(sql);
        await this.db.query(
          `UPDATE schema_migrations
           SET dirty=false, execution_ms=$2
           WHERE version=$1`,
          [version, Date.now() - started],
        );
        await this.db.query("COMMIT");
      } catch (e) {
        await this.db.query("ROLLBACK");
        throw e;
      }
    } finally {
      await this.db.query(`SELECT pg_advisory_unlock(hashtext('linqa_schema_migrations'))`);
    }
  }
}

import { CapabilityProbeResult, CapabilityState } from "./CapabilityRegistry";

const DEPENDENCIES: Record<string, string[]> = {
  SECURE_MESSAGING: [
    "DATABASE",
    "SECURE_CORE",
    "DEVICE_ENROLLMENT",
    "KEY_PACKAGE_VERIFICATION",
    "TRANSPARENCY",
    "MESSAGE_RELAY",
    "REALTIME",
  ],
  PHONE_AUTH: [
    "DATABASE",
    "SMS_DELIVERY",
    "RATE_LIMITING",
  ],
  RECOVERY_EMAIL: [
    "DATABASE",
    "EMAIL_DELIVERY",
    "RATE_LIMITING",
  ],
};

export function deriveCapability(
  key: string,
  snapshot: Record<string, CapabilityProbeResult>,
): CapabilityProbeResult {
  const deps = DEPENDENCIES[key] ?? [];
  const states: Record<string, CapabilityState> = {};
  let state: CapabilityState = "READY";
  let reason: string | undefined;

  for (const dep of deps) {
    const s = snapshot[dep]?.state ?? "NOT_CONFIGURED";
    states[dep] = s;
    if (s === "NOT_CONFIGURED") {
      state = "NOT_CONFIGURED";
      reason = `DEPENDENCY_NOT_CONFIGURED:${dep}`;
      break;
    }
    if (s === "DEGRADED" && state === "READY") {
      state = "DEGRADED";
      reason = `DEPENDENCY_DEGRADED:${dep}`;
    }
  }

  return {
    key,
    state,
    checkedAt: new Date().toISOString(),
    reason,
    dependencies: states,
  };
}

export function assertProductionBootSafe(
  snapshot: Record<string, CapabilityProbeResult>,
  config: {
    env: string;
    demoCoreEnabled: boolean;
    devOtpEnabled: boolean;
  },
) {

  const violations: string[] = [];

  if (config.devOtpEnabled) violations.push("DEV_OTP_ENABLED");

  const mandatory = [
    "DATABASE",
    "SECURE_CORE",
    "PHONE_AUTH",
    "DEVICE_ENROLLMENT",
    "KEY_PACKAGE_VERIFICATION",
  ];

  for (const key of mandatory) {
    if (snapshot[key]?.state !== "READY") {
      violations.push(`${key}:${snapshot[key]?.state ?? "MISSING"}`);
    }
  }

  if (violations.length) {
    throw new Error(`PRODUCTION_BOOT_BLOCKED:${violations.join(",")}`);
  }
}

import crypto from "node:crypto";

export interface TransparencyLeaf {
  userId: string;
  deviceId: string;
  keyPackageDigest: string;
  sequence: number;
}

export interface TransparencyLogRepo {
  append(leafHash: Buffer, leaf: TransparencyLeaf): Promise<{
    leafIndex: number;
    treeSize: number;
    rootHash: Buffer;
    inclusionProof: Buffer[];
  }>;
}

export class TransparencyService {
  constructor(
    private readonly repo: TransparencyLogRepo,
    private readonly checkpointSigningKeyPem: string,
  ) {}

  leafHash(leaf: TransparencyLeaf): Buffer {
    return crypto.createHash("sha256")
      .update(Buffer.from(JSON.stringify(leaf), "utf8"))
      .digest();
  }

  async append(leaf: TransparencyLeaf) {
    const leafHash = this.leafHash(leaf);
    const result = await this.repo.append(leafHash, leaf);

    const checkpointPayload = Buffer.from(JSON.stringify({
      treeSize: result.treeSize,
      rootHash: result.rootHash.toString("base64"),
    }), "utf8");

    const signature = crypto.sign(null, checkpointPayload, this.checkpointSigningKeyPem);

    return {
      leaf,
      leafHash: leafHash.toString("base64"),
      leafIndex: result.leafIndex,
      treeSize: result.treeSize,
      rootHash: result.rootHash.toString("base64"),
      inclusionProof: result.inclusionProof.map(x => x.toString("base64")),
      checkpointSignature: signature.toString("base64"),
    };
  }
}
