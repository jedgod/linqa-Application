# Linqa Admin Monitoring

Production-oriented implementation of the Admin Monitoring screenshot.

## Stack
- Next.js / React
- Tailwind CSS
- shadcn-style UI primitives
- SWR live refresh
- Recharts
- Express API
- PostgreSQL
- Zod validation
- RBAC
- Audit logging

## Quick start

1. Create PostgreSQL database.
2. Copy `.env.example` to `.env`.
3. Install packages:
   npm install
4. Run migration:
   npm run db:migrate
5. Start API:
   npm run dev:api
6. Start web:
   npm run dev:web

Mount `AdminMonitoringPage` as the Admin route, for example:
`app/admin/monitoring/page.tsx`.

## Recommended production hardening
- Replace header-based sample principal resolution with your existing Linqa staff SSO/JWT middleware.
- Enforce MFA for all staff.
- Use HttpOnly Secure SameSite cookies or short-lived access tokens.
- Add Redis for presence, queues, caching and WebSocket fanout.
- Feed `ops_system_services` from real health checks.
- Materialize daily metrics using a scheduled job.
- Add row-level tenancy constraints if Linqa enterprise tenants share operations infrastructure.
- Stream audit logs to immutable/WORM storage or SIEM.
- Never expose E2EE message plaintext, Vault encryption keys, passwords, recovery secrets, CVVs or full PANs to this dashboard.

import { Pool, PoolClient, QueryResultRow } from "pg";

export const pool = new Pool({
  connectionString: process.env.DATABASE_URL,
  max: Number(process.env.PG_POOL_MAX ?? 20),
  idleTimeoutMillis: 30_000,
  connectionTimeoutMillis: 5_000,
  ssl: process.env.NODE_ENV === "production"
    ? { rejectUnauthorized: true }
    : false,
});

export async function query<T extends QueryResultRow = any>(text: string, params: unknown[] = []) {
  return pool.query<T>(text, params);
}

export async function tx<T>(fn: (client: PoolClient) => Promise<T>) {
  const client = await pool.connect();
  try {
    await client.query("BEGIN");
    const out = await fn(client);
    await client.query("COMMIT");
    return out;
  } catch (error) {
    await client.query("ROLLBACK");
    throw error;
  } finally {
    client.release();
  }
}

import type { Request, Response, NextFunction } from "express";
import { hasPermission, type AdminPermission } from "../shared/permissions";

export interface StaffPrincipal {
  userId: string;
  staffId?: string;
  role: string;
  permissions?: string[];
}

declare global {
  namespace Express {
    interface Request {
      principal?: StaffPrincipal;
      requestId?: string;
    }
  }
}

export function requireStaff(req: Request, res: Response, next: NextFunction) {
  const userId = String(req.headers["x-linqa-user-id"] ?? "");
  const role = String(req.headers["x-linqa-role"] ?? "");
  const staffId = String(req.headers["x-linqa-staff-id"] ?? "");

  if (!userId || !role) return res.status(401).json({ error: "unauthorized" });

  req.principal = { userId, role, staffId: staffId || undefined };
  next();
}

export function requirePermission(permission: AdminPermission) {
  return (req: Request, res: Response, next: NextFunction) => {
    const p = req.principal;
    if (!p) return res.status(401).json({ error: "unauthorized" });
    if (!hasPermission(p.role, permission)) {
      return res.status(403).json({ error: "forbidden", permission });
    }
    next();
  };
}

import type { Request } from "express";
import { query } from "./db";
import crypto from "node:crypto";

export async function writeAudit(
  req: Request,
  input: {
    action: string;
    resourceType: string;
    resourceId?: string;
    reasonCode?: string;
    before?: unknown;
    after?: unknown;
  }
) {
  const ip = req.ip ?? "";
  const ipHash = ip ? crypto.createHash("sha256").update(ip).digest("hex") : null;

  await query(
    `INSERT INTO ops_audit_log
      (actor_user_id, actor_role, action, resource_type, resource_id, reason_code, request_id, ip_hash, before_json, after_json)
     VALUES ($1::uuid,$2,$3,$4,$5,$6,$7,$8,$9::jsonb,$10::jsonb)`,
    [
      req.principal?.userId ?? null,
      req.principal?.role ?? null,
      input.action,
      input.resourceType,
      input.resourceId ?? null,
      input.reasonCode ?? null,
      req.requestId ?? null,
      ipHash,
      JSON.stringify(input.before ?? null),
      JSON.stringify(input.after ?? null),
    ]
  );
}

const categoryLabels: Record<string, string> = {
  ACCOUNT_SUPPORT: "Account Support",
  LINQASEND: "LinqaSend",
  ACCOUNT_RECOVERY: "Account Recovery",
  VAULT_SUPPORT: "Vault Support",

export async function getDashboard(): Promise<AdminDashboardPayload> {
  const [
    summaryR, trendR, categoriesR, staffR, queuesR, priorityR, alertsR, activityR, servicesR
  ] = await Promise.all([
    query(`
      WITH current AS (
        SELECT
          count(*) FILTER (WHERE status IN ('OPEN','PENDING'))::int open_cases,
          count(*) FILTER (WHERE status IN ('OPEN','PENDING') AND assigned_staff_id IS NULL)::int unassigned_cases,
          count(*) FILTER (WHERE status='RESOLVED' AND resolved_at::date=current_date)::int resolved_today,
          avg(extract(epoch from (first_response_at-created_at))) FILTER (WHERE first_response_at IS NOT NULL) avg_response
        FROM ops_cases
      ),
      sla AS (
        SELECT COALESCE(
          100.0 * count(*) FILTER (
            WHERE first_response_at IS NOT NULL
              AND extract(epoch from (first_response_at-created_at)) <= q.sla_target_seconds
          ) / NULLIF(count(*) FILTER (WHERE first_response_at IS NOT NULL),0),
          100
        ) sla_pct
        FROM ops_cases c
        LEFT JOIN ops_queues q ON q.id=c.queue_id
        WHERE c.created_at >= now()-interval '7 days'
      )
      SELECT current.*, sla.sla_pct FROM current CROSS JOIN sla
    `),
    query(`
      SELECT metric_date::text date,
             opened_cases "newCases",
             resolved_cases "resolvedCases",
             backlog_end backlog
      FROM ops_daily_metrics
      WHERE metric_date >= current_date - 6
      ORDER BY metric_date
    `),
    query(`
      SELECT category::text category, count(*)::int count
      FROM ops_cases
      WHERE status IN ('OPEN','PENDING')
      GROUP BY category ORDER BY count DESC
    `),
    query(`
      SELECT id::text, display_name "displayName", role_name "roleName",
             status::text, active_case_count "activeCases"
      FROM ops_staff
      WHERE status <> 'OFFLINE'
      ORDER BY status='ONLINE' DESC, active_case_count DESC
      LIMIT 8
    `),
    query(`
      SELECT q.id::text, q.name,
             count(c.id) FILTER (WHERE c.status IN ('OPEN','PENDING'))::int "openCases",
             COALESCE(max(extract(epoch from (now()-c.created_at))) FILTER (WHERE c.status IN ('OPEN','PENDING')),0)::int "longestWaitSeconds",
             COALESCE(
               100.0 * count(c.id) FILTER (
                 WHERE c.first_response_at IS NOT NULL
                   AND extract(epoch from(c.first_response_at-c.created_at)) <= q.sla_target_seconds
               )
               / NULLIF(count(c.id) FILTER (WHERE c.first_response_at IS NOT NULL),0),
               100
             )::float8 "slaCompliancePct"
      FROM ops_queues q
      LEFT JOIN ops_cases c ON c.queue_id=q.id
      WHERE q.is_active=true
      GROUP BY q.id,q.name,q.sla_target_seconds
      ORDER BY "openCases" DESC
      LIMIT 8
    `),
    query(`
      SELECT priority::text priority,
             count(*)::int count,
             count(*) FILTER (
               WHERE first_response_at IS NULL
                 AND now()-created_at > interval '30 minutes'
             )::int "slaBreached"
      FROM ops_cases
      WHERE status IN ('OPEN','PENDING')
      GROUP BY priority
      ORDER BY priority
    `),
    query(`
      SELECT id::text, severity, title, description, created_at "createdAt"
      FROM ops_alerts WHERE resolved=false
      ORDER BY created_at DESC LIMIT 6
    `),
    query(`
      SELECT id::text, event_type type, summary text, created_at "createdAt"
      FROM ops_activity ORDER BY created_at DESC LIMIT 8
    `),
    query(`
      SELECT id::text, display_name name, status
      FROM ops_system_services
      ORDER BY display_name
    `),
  ]);

  const s = summaryR.rows[0] ?? {};
  const total = categoriesR.rows.reduce((n:number, r:any) => n + Number(r.count), 0);

  const categories: CaseCategoryPoint[] = categoriesR.rows.map((r:any) => ({
    category: r.category,
    label: categoryLabels[r.category] ?? r.category,
    count: Number(r.count),
    percentage: total ? Math.round((Number(r.count)/total)*1000)/10 : 0
  }));

  const priorities: PriorityRow[] = priorityR.rows.map((r:any) => ({
    priority: r.priority,
    label: ({P1:"P1 Critical",P2:"P2 High",P3:"P3 Medium",P4:"P4 Low"} as any)[r.priority],
    count: Number(r.count),
    slaBreached: Number(r.slaBreached),
  }));

  return {
    summary: {
      openCases: Number(s.open_cases ?? 0),
      unassignedCases: Number(s.unassigned_cases ?? 0),
      slaCompliancePct: Number(s.sla_pct ?? 100),
      avgResponseSeconds: Math.round(Number(s.avg_response ?? 0)),
      resolvedToday: Number(s.resolved_today ?? 0),
      deltas: {
        openCasesPct: 0,
        unassignedPct: 0,
        slaPct: 0,
        avgResponseSeconds: 0,
        resolvedTodayPct: 0,
      },
    },
    trend: trendR.rows as any,
    categories,
    liveStaff: staffR.rows as LiveStaffRow[],
    queues: queuesR.rows as QueueRow[],
    priorities,
    alerts: alertsR.rows as any,
    activity: activityR.rows as any,
    systemServices: servicesR.rows as any,
  };
}

adminRouter.post("/cases/:id/assign", requirePermission("cases.assign"), async (req, res, next) => {
  const schema = z.object({
    staffId: z.string().uuid(),
    reasonCode: z.string().min(3).max(100)
  });
  try {
    const body = schema.parse(req.body);
    const result = await tx(async client => {
      const staff = await client.query(`
        SELECT * FROM ops_staff WHERE id=$1::uuid FOR UPDATE
      `, [body.staffId]);
      if (!staff.rowCount) throw Object.assign(new Error("staff_not_found"), {status:404});
      const s = staff.rows[0];
      if (s.status === "OFFLINE") throw Object.assign(new Error("staff_offline"), {status:409});
      if (s.active_case_count >= s.max_concurrent_cases)
        throw Object.assign(new Error("staff_capacity_reached"), {status:409});

      const c = await client.query(`
        UPDATE ops_cases SET assigned_staff_id=$2::uuid, updated_at=now()
        WHERE id=$1::uuid AND status IN ('OPEN','PENDING')
        RETURNING *
      `, [req.params.id, body.staffId]);
      if (!c.rowCount) throw Object.assign(new Error("case_not_assignable"), {status:409});
      await client.query(`
        UPDATE ops_staff SET active_case_count=active_case_count+1, updated_at=now()
        WHERE id=$1::uuid
      `, [body.staffId]);
      return c.rows[0];
    });
    await writeAudit(req, {
      action:"case.assign", resourceType:"case", resourceId:req.params.id,
      reasonCode:body.reasonCode, after:result
    });
    res.json(result);
  } catch (e) { next(e); }
});
