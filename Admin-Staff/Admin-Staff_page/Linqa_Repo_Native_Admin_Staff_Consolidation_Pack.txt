# Linqa Repo-Native Admin/Staff Consolidation Pack

Targets the real architecture:
Fastify + existing `jwt.ts` scopes + `support_cases_v2` + `support-operations.service.ts` + `clients/web`.

This pack intentionally does not introduce Express, a second role-authority, `ops_cases`,
or a second support case table.

Flow:
Admin/Supervisor -> `/api/ops/cases/:id/assign`
-> existing JWT scope guard
-> transaction/lock `support_cases_v2`
-> update assignment
-> `ops_case_assignments_v2`
-> `support_case_events_v2`
-> `ops_outbox`
-> existing realtime transport
-> Staff My Work + Admin refresh.

CREATE TABLE IF NOT EXISTS ops_staff_profiles (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id uuid UNIQUE NOT NULL,
  display_name text NOT NULL,
  staff_role text NOT NULL,
  team_name text,
  presence ops_staff_presence NOT NULL DEFAULT 'OFFLINE',
  active_case_count integer NOT NULL DEFAULT 0 CHECK (active_case_count >= 0),
  max_concurrent_cases integer NOT NULL DEFAULT 8 CHECK (max_concurrent_cases > 0),
  suspended boolean NOT NULL DEFAULT false,
  last_seen_at timestamptz,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS ops_case_assignments_v2 (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  case_id uuid NOT NULL,
  staff_id uuid NOT NULL REFERENCES ops_staff_profiles(id),
  assigned_by_user_id uuid NOT NULL,
  assignment_type text NOT NULL CHECK (assignment_type IN ('AUTO','MANUAL','REASSIGN','ESCALATION')),
  reason_code text NOT NULL,
  assigned_at timestamptz NOT NULL DEFAULT now(),
  released_at timestamptz,
  release_reason text
);

CREATE UNIQUE INDEX IF NOT EXISTS uq_ops_case_assignments_v2_active

  );

import type { DbLike, TxLike } from "./db-types";
import { enqueueOpsEvent } from "./outbox.service";
import { writeOpsAudit } from "./audit.service";

const WORKABLE = new Set(["NEW","QUEUED","OPEN","ASSIGNED","IN_PROGRESS","PENDING","PENDING_CLIENT","ESCALATED"]);

export class CaseAssignmentService {
  constructor(private db:DbLike) {}

  async assign(input:{
    caseId:string; targetStaffId:string; actorUserId:string; actorScopes?:string[];
    reasonCode:string; assignmentType:"AUTO"|"MANUAL"|"REASSIGN"|"ESCALATION";
    requestId?:string|null;
  }) {
    return this.db.tx(async tx=>{
      const c = await this.caseForUpdate(tx,input.caseId);
      if(!WORKABLE.has(c.status)) throw Object.assign(new Error("CASE_NOT_ASSIGNABLE"),{statusCode:409});

      const staff = await this.staffForUpdate(tx,input.targetStaffId);
      if(staff.suspended) throw Object.assign(new Error("STAFF_SUSPENDED"),{statusCode:409});
      if(staff.presence==="OFFLINE") throw Object.assign(new Error("STAFF_OFFLINE"),{statusCode:409});
      if(Number(staff.active_case_count)>=Number(staff.max_concurrent_cases))
        throw Object.assign(new Error("STAFF_CAPACITY_REACHED"),{statusCode:409});

      const previousOwner = c.assigned_staff_id ?? null;

      if(previousOwner && previousOwner!==input.targetStaffId) {
        await tx.query(
          `UPDATE ops_case_assignments_v2
           SET released_at=now(),release_reason='REASSIGNED'
           WHERE case_id=$1::uuid AND released_at IS NULL`,
          [input.caseId]
        );
        await tx.query(
          `UPDATE ops_staff_profiles
           SET active_case_count=GREATEST(active_case_count-1,0),updated_at=now()
           WHERE id=$1::uuid`,
          [previousOwner]
        );
      }

      if(previousOwner!==input.targetStaffId) {
        await tx.query(
          `INSERT INTO ops_case_assignments_v2(
             case_id,staff_id,assigned_by_user_id,assignment_type,reason_code
           ) VALUES($1::uuid,$2::uuid,$3::uuid,$4,$5)`,
          [input.caseId,input.targetStaffId,input.actorUserId,input.assignmentType,input.reasonCode]
        );
        await tx.query(
          `UPDATE ops_staff_profiles
           SET active_case_count=active_case_count+1,updated_at=now()
           WHERE id=$1::uuid`,
          [input.targetStaffId]
        );
      }

      const updated = await tx.query<any>(
        `UPDATE support_cases_v2
         SET assigned_staff_id=$2::uuid,
             status=CASE WHEN status IN ('NEW','QUEUED','OPEN') THEN 'ASSIGNED' ELSE status END,
             updated_at=now()
         WHERE id=$1::uuid
         RETURNING *`,
        [input.caseId,input.targetStaffId]
      );

      await tx.query(
        `INSERT INTO support_case_events_v2(case_id,event_type,actor_id,metadata,created_at)
         VALUES($1::uuid,'ASSIGNED',$2::uuid,$3::jsonb,now())`,
        [input.caseId,input.actorUserId,JSON.stringify({
          previousOwner,targetStaffId:input.targetStaffId,
          assignmentType:input.assignmentType,reasonCode:input.reasonCode
        })]
      );

      await enqueueOpsEvent(tx,{
        aggregateType:"support_case",
        aggregateId:input.caseId,
        eventType:"ops.case.assigned",
        payload:{caseId:input.caseId,previousOwner,targetStaffId:input.targetStaffId}
      });

      await writeOpsAudit(tx,{
        actorUserId:input.actorUserId,actorScopes:input.actorScopes,
        action:"ops.case.assign",resourceType:"support_case_v2",resourceId:input.caseId,
        reasonCode:input.reasonCode,requestId:input.requestId,before:c,after:updated.rows[0]
      });

      return updated.rows[0];
    });
  }

  async release(input:{
    caseId:string; actorUserId:string; actorScopes?:string[];
    reasonCode:string; requestId?:string|null;
  }) {
    return this.db.tx(async tx=>{
      const c=await this.caseForUpdate(tx,input.caseId);
      const owner=c.assigned_staff_id??null;
      if(!owner) return c;

      await tx.query(
        `UPDATE ops_case_assignments_v2
         SET released_at=now(),release_reason=$2
         WHERE case_id=$1::uuid AND released_at IS NULL`,
        [input.caseId,input.reasonCode]
      );
      await tx.query(
        `UPDATE ops_staff_profiles
         SET active_case_count=GREATEST(active_case_count-1,0),updated_at=now()
         WHERE id=$1::uuid`,
        [owner]
      );
      const updated=await tx.query<any>(
        `UPDATE support_cases_v2
         SET assigned_staff_id=NULL,status='QUEUED',updated_at=now()
         WHERE id=$1::uuid RETURNING *`,
        [input.caseId]
      );
      await enqueueOpsEvent(tx,{
        aggregateType:"support_case",aggregateId:input.caseId,eventType:"ops.case.released",
        payload:{caseId:input.caseId,previousOwner:owner}
      });
      return updated.rows[0];
    });
  }

  private async caseForUpdate(tx:TxLike,caseId:string){
    const r=await tx.query<any>(`SELECT * FROM support_cases_v2 WHERE id=$1::uuid FOR UPDATE`,[caseId]);
    if(!r.rowCount) throw Object.assign(new Error("CASE_NOT_FOUND"),{statusCode:404});
    return r.rows[0];
  }

  private async staffForUpdate(tx:TxLike,staffId:string){
    const r=await tx.query<any>(`SELECT * FROM ops_staff_profiles WHERE id=$1::uuid FOR UPDATE`,[staffId]);
    if(!r.rowCount) throw Object.assign(new Error("STAFF_NOT_FOUND"),{statusCode:404});
    return r.rows[0];
  }
}

import type { DbLike } from "./db-types";

export class StaffOperationsService {
  constructor(private db:DbLike){}

  async listLiveStaff(){
    const r=await this.db.query<any>(
      `SELECT id::text,user_id::text "userId",display_name "displayName",
              staff_role "staffRole",team_name "teamName",presence::text,
              active_case_count "activeCases",max_concurrent_cases "maxConcurrentCases",
              last_seen_at "lastSeenAt"
       FROM ops_staff_profiles
       WHERE suspended=false
       ORDER BY CASE presence WHEN 'AVAILABLE' THEN 0 WHEN 'BUSY' THEN 1 WHEN 'AWAY' THEN 2 ELSE 3 END,
                display_name`
    );
    return r.rows;
  }

  async getMyWorkByUserId(userId:string){
    const p=await this.db.query<any>(`SELECT id::text FROM ops_staff_profiles WHERE user_id=$1::uuid`,[userId]);
    if(!p.rowCount) throw Object.assign(new Error("STAFF_PROFILE_NOT_FOUND"),{statusCode:404});

    const r=await this.db.query<any>(
      `SELECT id::text,public_id "publicId",status,priority,category,updated_at "updatedAt"
       FROM support_cases_v2
       WHERE assigned_staff_id=$1::uuid AND status NOT IN ('RESOLVED','CLOSED')
       ORDER BY CASE priority WHEN 'P1' THEN 1 WHEN 'P2' THEN 2 WHEN 'P3' THEN 3 ELSE 4 END,
                updated_at DESC`,
      [p.rows[0].id]
    );
    return r.rows;
  }
}

import type { DbLike } from "./db-types";

export class OperationsDashboardService {
  constructor(private db:DbLike){}

  async overview(){
    const [cases,staff,sla]=await Promise.all([
      this.db.query<any>(
        `SELECT
           count(*) FILTER(WHERE status NOT IN ('RESOLVED','CLOSED'))::int "openCases",
           count(*) FILTER(WHERE status NOT IN ('RESOLVED','CLOSED') AND assigned_staff_id IS NULL)::int "unassignedCases",
           count(*) FILTER(WHERE status='RESOLVED' AND updated_at::date=current_date)::int "resolvedToday"
         FROM support_cases_v2`
      ),
      this.db.query<any>(
        `SELECT
           count(*) FILTER(WHERE presence='AVAILABLE')::int "availableStaff",
           count(*) FILTER(WHERE presence='BUSY')::int "busyStaff"
         FROM ops_staff_profiles WHERE suspended=false`
      ),
      this.db.query<any>(
        `SELECT count(*) FILTER(WHERE first_response_breached_at IS NULL)::int good,
                count(*)::int total
         FROM ops_case_sla_state_v2`
      )
    ]);

    const c=cases.rows[0]??{}, s=staff.rows[0]??{}, sl=sla.rows[0]??{};
    return {
      openCases:Number(c.openCases??0),
      unassignedCases:Number(c.unassignedCases??0),
      resolvedToday:Number(c.resolvedToday??0),
      availableStaff:Number(s.availableStaff??0),
      busyStaff:Number(s.busyStaff??0),
      slaCompliancePct:Number(sl.total)>0
        ? Math.round((Number(sl.good)/Number(sl.total))*1000)/10
        : 100
    };
  }
}

import type { FastifyPluginAsync } from "fastify";
import { z } from "zod";
import { assertScope,scopesOf } from "./auth";
import { CaseAssignmentService } from "../../services/operations/case-assignment.service";
import { StaffOperationsService } from "../../services/operations/staff-operations.service";
import { OperationsDashboardService } from "../../services/operations/dashboard.service";
import { repoNativeSupportOps } from "../../services/operations/support-operations.adapter";

const CaseParams=z.object({caseId:z.string().uuid()});
const AssignBody=z.object({staffId:z.string().uuid(),reasonCode:z.string().min(3).max(100)});
const ReleaseBody=z.object({reasonCode:z.string().min(3).max(100)});

export const operationsRoutes:FastifyPluginAsync=async(app)=>{
  const assignment=new CaseAssignmentService((app as any).db);
  const staff=new StaffOperationsService((app as any).db);
  const dashboard=new OperationsDashboardService((app as any).db);

  app.get("/dashboard",async(request,reply)=>{
    const p=await (app as any).authenticate(request,reply);
    assertScope(p,"ops:dashboard:read");
    return dashboard.overview();
  });

  app.get("/staff/live",async(request,reply)=>{
    const p=await (app as any).authenticate(request,reply);
    assertScope(p,"ops:staff:read");
    return staff.listLiveStaff();
  });

  app.get("/staff/my-work",async(request,reply)=>{
    const p=await (app as any).authenticate(request,reply);
    assertScope(p,"ops:case:read");
    return staff.getMyWorkByUserId(p.sub);
  });

  app.get("/cases/:caseId",async(request,reply)=>{
    const p=await (app as any).authenticate(request,reply);
    assertScope(p,"ops:case:read");
    const {caseId}=CaseParams.parse(request.params);
    const result=await repoNativeSupportOps.getCaseWorkspace(caseId);
    if(!result) return reply.code(404).send({error:"CASE_NOT_FOUND"});
    return result;
  });

  app.post("/cases/:caseId/assign",async(request,reply)=>{
    const p=await (app as any).authenticate(request,reply);
    assertScope(p,"ops:case:assign");
    const {caseId}=CaseParams.parse(request.params);
    const body=AssignBody.parse(request.body);
    return assignment.assign({
      caseId,targetStaffId:body.staffId,actorUserId:p.sub,actorScopes:scopesOf(p),
      reasonCode:body.reasonCode,assignmentType:"MANUAL",requestId:request.id
    });
  });

  app.post("/cases/:caseId/reassign",async(request,reply)=>{
    const p=await (app as any).authenticate(request,reply);
    assertScope(p,"ops:case:reassign");
    const {caseId}=CaseParams.parse(request.params);
    const body=AssignBody.parse(request.body);
    return assignment.assign({
      caseId,targetStaffId:body.staffId,actorUserId:p.sub,actorScopes:scopesOf(p),
      reasonCode:body.reasonCode,assignmentType:"REASSIGN",requestId:request.id
    });
  });

  app.post("/cases/:caseId/release",async(request,reply)=>{
    const p=await (app as any).authenticate(request,reply);
    assertScope(p,"ops:case:release");
    const {caseId}=CaseParams.parse(request.params);
    const body=ReleaseBody.parse(request.body);
    return assignment.release({
      caseId,actorUserId:p.sub,actorScopes:scopesOf(p),
      reasonCode:body.reasonCode,requestId:request.id
    });
  });
};

export interface OpsEventTransport {
  publish(topic:string,payload:unknown):Promise<void>;
}

/**
 * Inject the repository's existing websocket/event transport here.
 * Do not create a second realtime framework.
 */
export async function publishPendingOpsOutbox(db:any,transport:OpsEventTransport,batchSize=100){
  return db.tx(async(tx:any)=>{
    const r=await tx.query(
      `SELECT id::text,event_type,payload
       FROM ops_outbox
       WHERE published_at IS NULL
       ORDER BY created_at
       FOR UPDATE SKIP LOCKED
       LIMIT $1`,
      [batchSize]
    );

    for(const row of r.rows){
      try{
        await transport.publish(row.event_type,row.payload);
        await tx.query(
          `UPDATE ops_outbox
           SET published_at=now(),publish_attempts=publish_attempts+1,last_error=NULL
           WHERE id=$1::uuid`,
          [row.id]
        );
      }catch(e:any){
        await tx.query(
          `UPDATE ops_outbox
           SET publish_attempts=publish_attempts+1,last_error=$2
           WHERE id=$1::uuid`,
          [row.id,String(e?.message??e).slice(0,1000)]
        );
      }
    }
    return r.rowCount;
  });
}

async function api<T>(path:string,init?:RequestInit):Promise<T>{
  const r=await fetch(path,{
    credentials:"include",
    ...init,
    headers:{"content-type":"application/json",...(init?.headers??{})}
  });
  const body=await r.json().catch(()=>({}));
  if(!r.ok) throw new Error(body.error??`HTTP_${r.status}`);
  return body as T;
}

export const operationsApi={
  overview:()=>api<Overview>("/api/ops/dashboard"),
  liveStaff:()=>api<LiveStaff[]>("/api/ops/staff/live"),
  myWork:()=>api<any[]>("/api/ops/staff/my-work"),
  case:(id:string)=>api<any>(`/api/ops/cases/${id}`),
  assign:(id:string,staffId:string,reasonCode:string)=>api<any>(`/api/ops/cases/${id}/assign`,{
    method:"POST",body:JSON.stringify({staffId,reasonCode})
  }),
  reassign:(id:string,staffId:string,reasonCode:string)=>api<any>(`/api/ops/cases/${id}/reassign`,{
    method:"POST",body:JSON.stringify({staffId,reasonCode})
  }),
  release:(id:string,reasonCode:string)=>api<any>(`/api/ops/cases/${id}/release`,{
    method:"POST",body:JSON.stringify({reasonCode})
  })

import { useEffect } from "react";
import { refreshOperations } from "./useOperations";

/**
 * Attach this to the existing Linqa realtime client.
 * Do not create a second websocket connection if clients/web already has one.
 */
export function useOperationsRealtime(){
  useEffect(()=>{
    const rt=(window as any).linqaRealtime;
    if(!rt?.subscribe) return;
    return rt.subscribe("ops.*",(event:any)=>{
      if(
        event?.type==="ops.case.assigned" ||
        event?.type==="ops.case.released" ||
        String(event?.type??"").startsWith("ops.sla.")
      ){
        refreshOperations();
      }
    });
  },[]);
}
