# Linqa Repo-Native Client 360 Account & Services Enrichment Pack

Read-only enrichment only.

Uses:
- Fastify
- principal.scopes Set
- raw pg.Pool
- support_cases_v2.user_id
- existing domain services
- existing ops:client360:read scope
- clients/web accessToken + polling

No new tables, no duplicated business logic, no specialist actions, no SLA, no new realtime.

import type { Pool } from "pg";
import type { Client360DomainAdapter } from "./client360-domain.adapter";

function iso(v:string|Date|null|undefined):string|null {
  if(!v) return null;
  const d=v instanceof Date?v:new Date(v);
  return Number.isNaN(d.getTime())?null:d.toISOString();
}

export class Client360EnrichmentService {
  constructor(private pool:Pool,private domains:Client360DomainAdapter){}

  async getForCase(caseId:string){
    const caseR=await this.pool.query(
      `SELECT id::text,case_number AS "caseNumber",user_id::text AS "userId",
              state::text,priority::text
       FROM support_cases_v2
       WHERE id=$1::uuid
       LIMIT 1`,
      [caseId]
    );

    if(!caseR.rowCount) return null;
    const supportCase=caseR.rows[0];
    const userId=supportCase.userId as string;

    const results=await Promise.allSettled([
      this.domains.getAccountSummary(userId),
      this.domains.getDeviceSummary(userId),
      this.domains.getLinqaSendSummary(userId),
      this.domains.getVaultSummary(userId),
      this.domains.getRecoverySummary(userId),
      this.domains.getRiskSummary(userId)
    ]);

    const [accountR,devicesR,sendR,vaultR,recoveryR,riskR]=results;
    const account=accountR.status==="fulfilled"?accountR.value:null;
    const devices=devicesR.status==="fulfilled"?devicesR.value:null;
    const linqaSend=sendR.status==="fulfilled"?sendR.value:null;
    const vault=vaultR.status==="fulfilled"?vaultR.value:null;
    const recovery=recoveryR.status==="fulfilled"?recoveryR.value:null;
    const risk=riskR.status==="fulfilled"?riskR.value:null;

    return {
      caseId:supportCase.id,
      caseNumber:supportCase.caseNumber,
      userId,
      identity:account?{
        displayName:account.displayName??null,
        maskedEmail:account.maskedEmail??null,
        maskedPhone:account.maskedPhone??null,
        accountState:account.accountState,
        countryCode:account.countryCode??null,
        emailVerified:account.emailVerified??null,
        phoneVerified:account.phoneVerified??null,
        memberSince:iso(account.createdAt)
      }:null,
      devices:devices?{
        trustedDeviceCount:devices.trustedDeviceCount,
        activeDeviceCount:devices.activeDeviceCount,
        lastTrustedDeviceSeenAt:iso(devices.lastTrustedDeviceSeenAt),
        platforms:devices.platforms??[]
      }:null,
      linqaSend:linqaSend?{
        enabled:linqaSend.enabled,
        accountState:linqaSend.accountState??null,
        verificationState:linqaSend.verificationState??null,
        activeHoldCount:linqaSend.activeHoldCount??null,
        lastTransactionAt:iso(linqaSend.lastTransactionAt)
      }:null,
      vault:vault?{
        enabled:vault.enabled,
        planName:vault.planName??null,
        usedBytes:vault.usedBytes??null,
        quotaBytes:vault.quotaBytes??null,
        quarantineCount:vault.quarantineCount??null
      }:null,
      recovery:recovery?{
        recoveryEmailConfigured:recovery.recoveryEmailConfigured,
        trustedRecoveryDeviceCount:recovery.trustedRecoveryDeviceCount??null,
        twoFactorEnabled:recovery.twoFactorEnabled??null,
        passkeyCount:recovery.passkeyCount??null,
        lastRecoveryAt:iso(recovery.lastRecoveryAt)
      }:null,
      risk:risk?{
        level:risk.level,
        score:risk.score??null,
        openRiskCaseCount:risk.openRiskCaseCount??null,
        safetyRestrictionActive:risk.safetyRestrictionActive??null,
        lastReviewedAt:iso(risk.lastReviewedAt)
      }:null,
      availability:{
        account:accountR.status==="fulfilled",
        devices:devicesR.status==="fulfilled",
        linqaSend:sendR.status==="fulfilled",
        vault:vaultR.status==="fulfilled",
        recovery:recoveryR.status==="fulfilled",
        risk:riskR.status==="fulfilled"
      }
    };
  }
}

    getRecoverySummary:(userId)=>deps.recoveryService.getSupportSummary(userId),
    getRiskSummary:(userId)=>deps.riskService.getSupportSummary(userId)
  };
}

import type { FastifyPluginAsync } from "fastify";
import { z } from "zod";
import { requireScope } from "../../auth/jwt";
import type { Client360DomainAdapter } from "../../services/operations/client360-domain.adapter";
import { Client360EnrichmentService } from "../../services/operations/client360-enrichment.service";

const Params=z.object({caseId:z.string().uuid()});

export function buildClient360EnrichmentRoutes(
  pool:any,
  domains:Client360DomainAdapter
):FastifyPluginAsync {
  return async function client360EnrichmentRoutes(app){
    const service=new Client360EnrichmentService(pool,domains);

    app.get("/cases/:caseId/client360",async(request,reply)=>{
      const p=await (app as any).authenticate(request,reply);
      requireScope(p,"ops:client360:read");

      const {caseId}=Params.parse(request.params);
      const result=await service.getForCase(caseId);

      if(!result) return reply.code(404).send({error:"CASE_NOT_FOUND"});
      return result;
    });
  };
}

async function api<T>(token:string,path:string):Promise<T>{
  const r=await fetch(path,{headers:{authorization:`Bearer ${token}`}});
  const body=await r.json().catch(()=>({}));
  if(!r.ok) throw new Error(body.error??`HTTP_${r.status}`);
  return body as T;
}

export const client360Api={
  getForCase:(token:string,caseId:string)=>
    api<Client360Projection>(token,`/v1/admin/operations/cases/${caseId}/client360`)
};

function bytes(n?:number|null){
  if(n==null) return "—";
  const u=["B","KB","MB","GB","TB"];
  let v=n,i=0;
  while(v>=1024&&i<u.length-1){v/=1024;i++;}
  return `${v.toFixed(i?1:0)} ${u[i]}`;
}

# Integration notes

Server:
- No migration.
- Add GET /cases/:caseId/client360 under existing /v1/admin/operations.
- Use createAuthenticator/requireScope/withErrors.
- Require existing ops:client360:read.
- Resolve only support_cases_v2.user_id in operations.
- Inject existing domain services.
- If a domain lacks a support-safe summary, add it to that owning service.

Client:
- Fold Client360EnrichmentPanel into the existing CaseWorkspacePanel.
- Use existing usePolling at ~15 seconds.
- Keep it read-only.

Never expose:
password hashes, OTP/TOTP secrets, recovery secrets, passkey private material,
E2EE keys, Vault encryption keys, CVV, full PAN, or unrelated private content.
