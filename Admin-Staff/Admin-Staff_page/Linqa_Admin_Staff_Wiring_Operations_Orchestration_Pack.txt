# Linqa Admin ↔ Staff Wiring & Operations Orchestration Pack

This pack turns the Admin Monitoring and Staff Support interfaces into one shared operational system.

## Included

- Shared PostgreSQL schema migration
- Unified staff identity/RBAC
- Transaction-safe case ownership and reassignment
- Workload/capacity-aware assignment engine
- Transactional outbox
- Redis Streams publisher
- NATS publisher
- WebSocket fanout gateway
- SLA warning/breach worker
- Supervisor reassignment / release / priority controls
- Client 360 role-filtered projection gateway
- Sensitive-action approval engine
- Step-up / dual-approval policy hooks
- REST API contracts
- Audit events
- End-to-end integration tests

## Core event flow

Admin/Supervisor action
→ PostgreSQL transaction
→ case assignment / case state update
→ audit event
→ ops_outbox row
→ outbox dispatcher
→ Redis Streams and/or NATS
→ WebSocket gateway
→ Admin dashboard + Staff My Work update

## Important boundaries

This operations layer must not create a master bypass into client privacy.

Do not expose:
- E2EE keys
- arbitrary decrypted client conversations
- Vault encryption keys / unrestricted Vault plaintext
- passwords
- recovery secrets
- CVV
- full payment card PAN
- unrelated personal data

Sensitive operational actions must be authorized and audited.
High-impact actions may require dual approval and/or step-up authentication.

## Suggested run order

1. Apply DB migration.
2. Start PostgreSQL.
3. Start Redis and/or NATS.
4. Start API.
5. Start outbox dispatcher.
6. Start SLA worker.
7. Point Admin and Staff web clients to `/api/ops`.
8. Subscribe both UIs to `/ws/ops`.
9. Run integration tests.

## Local environment

npm install
cp .env.example .env
npm run db:migrate
npm run dev:api
npm run dev:outbox
npm run dev:sla

import { Pool, PoolClient, QueryResultRow } from "pg";

export const pool = new Pool({
  connectionString: process.env.DATABASE_URL,
  max: 30,
  idleTimeoutMillis: 30_000,
  connectionTimeoutMillis: 5_000,
  ssl: process.env.NODE_ENV === "production" ? { rejectUnauthorized: true } : false,
});

export async function query<T extends QueryResultRow = any>(
  sql: string,
  params: unknown[] = []
) {
  return pool.query<T>(sql, params);
}

export async function tx<T>(fn: (client: PoolClient) => Promise<T>) {
  const client = await pool.connect();
  try {
    await client.query("BEGIN");
    const result = await fn(client);
    await client.query("COMMIT");
    return result;
  } catch (error) {
    await client.query("ROLLBACK");
    throw error;
  } finally {
    client.release();
  }
}

import crypto from "node:crypto";
import type { Request } from "express";
import type { PoolClient } from "pg";

export async function writeAudit(
  client: PoolClient,
  req: Request,
  input: {
    action: string;
    resourceType: string;
    resourceId?: string;
    reasonCode?: string;
    before?: unknown;
    after?: unknown;
  }
) {
  const ipHash = req.ip
    ? crypto.createHash("sha256").update(req.ip).digest("hex")
    : null;

  await client.query(`
    INSERT INTO ops_audit_log(
      actor_user_id,actor_staff_id,actor_role,action,resource_type,
      resource_id,reason_code,request_id,ip_hash,before_json,after_json
    )
    VALUES ($1::uuid,$2::uuid,$3,$4,$5,$6,$7,$8,$9,$10::jsonb,$11::jsonb)
  `, [
    req.principal?.userId ?? null,
    req.principal?.staffId ?? null,
    req.principal?.role ?? null,
    input.action,
    input.resourceType,
    input.resourceId ?? null,
    input.reasonCode ?? null,
    req.requestId ?? null,
    ipHash,
    JSON.stringify(input.before ?? null),
    JSON.stringify(input.after ?? null),
  ]);
}

import type { Request, Response, NextFunction } from "express";
import crypto from "node:crypto";
import { hasPermission, type Permission } from "./rbac.js";

export interface Principal {
  userId: string;
  staffId: string;
  role: string;
  displayName: string;
  stepUpVerifiedAt?: string | null;
}

declare global {
  namespace Express {
    interface Request {
      principal?: Principal;
      requestId?: string;
    }
  }
}

export function requestContext(req: Request, res: Response, next: NextFunction) {
  req.requestId = String(req.headers["x-request-id"] ?? crypto.randomUUID());
  res.setHeader("x-request-id", req.requestId);
  next();
}

export function requireStaff(req: Request, res: Response, next: NextFunction) {
  // Production: replace with Linqa Staff SSO/JWT/session validation.
  const userId = String(req.headers["x-linqa-user-id"] ?? "");
  const staffId = String(req.headers["x-linqa-staff-id"] ?? "");
  const role = String(req.headers["x-linqa-role"] ?? "");
  const displayName = String(req.headers["x-linqa-staff-name"] ?? "");
  const stepUpVerifiedAt = req.headers["x-linqa-step-up-at"]
    ? String(req.headers["x-linqa-step-up-at"])
    : null;

  if (!userId || !staffId || !role) {
    return res.status(401).json({ error: "unauthorized" });
  }

  req.principal = {
    userId,
    staffId,
    role,
    displayName: displayName || "Staff",
    stepUpVerifiedAt
  };
  next();
}

export function requirePermission(permission: Permission) {
  return (req: Request, res: Response, next: NextFunction) => {
    if (!req.principal) return res.status(401).json({ error: "unauthorized" });
    if (!hasPermission(req.principal.role, permission)) {
      return res.status(403).json({ error: "forbidden", permission });
    }
    next();
  };
}

import Redis from "ioredis";
import { connect, StringCodec } from "nats";

const codec = StringCodec();

export interface EventPublisher {
  publish(eventType: string, eventKey: string, payload: unknown): Promise<void>;
  close(): Promise<void>;
}

class RedisPublisher implements EventPublisher {
  private redis = new Redis(process.env.REDIS_URL ?? "redis://localhost:6379");

  async publish(eventType: string, eventKey: string, payload: unknown) {
    await this.redis.xadd(
      "linqa:ops:events",
      "*",
      "eventType", eventType,
      "eventKey", eventKey,
      "payload", JSON.stringify(payload)
    );
  }

  async close() {
    await this.redis.quit();
  }
}

class NatsPublisher implements EventPublisher {
  private ncPromise = connect({ servers: process.env.NATS_URL ?? "nats://localhost:4222" });

  async publish(eventType: string, eventKey: string, payload: unknown) {
    const nc = await this.ncPromise;
    nc.publish(
      `linqa.ops.${eventType.replaceAll(".", "_")}`,
      codec.encode(JSON.stringify({ eventType, eventKey, payload }))
    );
    await nc.flush();
  }

  async close() {
    const nc = await this.ncPromise;
    await nc.drain();
  }
}

class BothPublisher implements EventPublisher {
  constructor(private a: EventPublisher, private b: EventPublisher) {}
  async publish(eventType: string, eventKey: string, payload: unknown) {
    await Promise.all([
      this.a.publish(eventType,eventKey,payload),
      this.b.publish(eventType,eventKey,payload)
    ]);
  }
  async close() { await Promise.all([this.a.close(),this.b.close()]); }
}

export function makePublisher(): EventPublisher {
  const mode = process.env.EVENT_TRANSPORT ?? "both";
  if (mode === "redis") return new RedisPublisher();
  if (mode === "nats") return new NatsPublisher();
  return new BothPublisher(new RedisPublisher(), new NatsPublisher());
}

import "dotenv/config";
import { pool } from "../common/db.js";
import { makePublisher } from "./publishers.js";

const publisher = makePublisher();
const batchSize = Number(process.env.OUTBOX_BATCH_SIZE ?? 100);
const pollMs = Number(process.env.OUTBOX_POLL_MS ?? 1000);

async function dispatchOnce() {
  const client = await pool.connect();
  try {
    await client.query("BEGIN");

    const r = await client.query(`
      SELECT id::text, event_type "eventType", event_key "eventKey", payload
      FROM ops_outbox
      WHERE published_at IS NULL
      ORDER BY created_at
      FOR UPDATE SKIP LOCKED
      LIMIT $1
    `, [batchSize]);

    for (const row of r.rows) {
      try {
        await publisher.publish(row.eventType,row.eventKey,row.payload);
        await client.query(`
          UPDATE ops_outbox
          SET published_at=now(),publish_attempts=publish_attempts+1,last_error=NULL
          WHERE id=$1::uuid
        `,[row.id]);
      } catch (e:any) {
        await client.query(`
          UPDATE ops_outbox
          SET publish_attempts=publish_attempts+1,last_error=$2
          WHERE id=$1::uuid
        `,[row.id,String(e?.message ?? e).slice(0,1000)]);
      }
    }

    await client.query("COMMIT");
  } catch (e) {
    await client.query("ROLLBACK");
    console.error("outbox dispatch failed", e);
  } finally {
    client.release();
  }
}

const timer = setInterval(dispatchOnce,pollMs);
dispatchOnce().catch(console.error);

process.on("SIGTERM", async () => {
  clearInterval(timer);
  await publisher.close();
  await pool.end();
  process.exit(0);
});

function validateAssignee(staff:any, requiredRole?:string|null) {
  if (staff.suspended) throw new HttpError(409,"staff_suspended");
  if (staff.presence === "OFFLINE") throw new HttpError(409,"staff_offline");
  if (staff.active_case_count >= staff.max_concurrent_cases) {
    throw new HttpError(409,"staff_capacity_reached");
  }
  if (requiredRole && staff.role_name !== requiredRole && !["SUPER_ADMIN","OPS_ADMIN","SUPERVISOR"].includes(staff.role_name)) {
    throw new HttpError(409,"staff_role_not_eligible");
  }
}

export async function assignCase(
  client: PoolClient,
  input: {
    caseId: string;
    targetStaffId: string;
    actorStaffId: string;
    reasonCode: string;
    assignmentType: "AUTO"|"MANUAL"|"REASSIGN"|"ESCALATION";
  }
) {
  const c = await loadCaseForUpdate(client,input.caseId);
  if (!WORKABLE.includes(c.status)) throw new HttpError(409,"case_not_assignable");

  const target = await loadStaffForUpdate(client,input.targetStaffId);
  validateAssignee(target,c.required_role);

  const currentOwnerId = c.owner_staff_id as string | null;

  if (currentOwnerId === input.targetStaffId) {
    return c;
  }

  if (currentOwnerId) {
    await client.query(`
      UPDATE case_assignments
      SET released_at=now(),release_reason='REASSIGNED'
      WHERE case_id=$1::uuid AND released_at IS NULL
    `,[input.caseId]);

    await client.query(`
      UPDATE staff_users
      SET active_case_count=GREATEST(active_case_count-1,0),updated_at=now()
      WHERE id=$1::uuid
    `,[currentOwnerId]);
  }

  await client.query(`
    INSERT INTO case_assignments(
      case_id,staff_id,assigned_by_staff_id,reason_code,assignment_type
    )
    VALUES ($1::uuid,$2::uuid,$3::uuid,$4,$5)
  `,[input.caseId,input.targetStaffId,input.actorStaffId,input.reasonCode,input.assignmentType]);

  await client.query(`
    UPDATE staff_users
    SET active_case_count=active_case_count+1,updated_at=now()
    WHERE id=$1::uuid
  `,[input.targetStaffId]);

  const updatedR = await client.query(`
    UPDATE support_cases
    SET owner_staff_id=$2::uuid,
        status=CASE WHEN status IN ('NEW','QUEUED') THEN 'ASSIGNED'::ops_case_status ELSE status END,
        version=version+1,
        updated_at=now()
    WHERE id=$1::uuid
    RETURNING *
  `,[input.caseId,input.targetStaffId]);

  await client.query(`
    INSERT INTO case_events(case_id,actor_staff_id,event_type,payload)
    VALUES ($1::uuid,$2::uuid,'CASE_ASSIGNED',$3::jsonb)
  `,[
    input.caseId,input.actorStaffId,
    JSON.stringify({fromStaffId:currentOwnerId,toStaffId:input.targetStaffId,reasonCode:input.reasonCode})
  ]);

  await enqueueEvent(client,{
    aggregateType:"case",
    aggregateId:input.caseId,
    eventType:"case.assigned",
    eventKey:input.caseId,
    payload:{
      caseId:input.caseId,
      publicId:c.public_id,
      fromStaffId:currentOwnerId,
      toStaffId:input.targetStaffId,
      assignmentType:input.assignmentType
    }
  });

  return updatedR.rows[0];
}

import type { PoolClient } from "pg";
import { HttpError } from "../common/errors.js";
import { ACTION_POLICIES } from "./policy.js";
import { enqueueEvent } from "../events/outbox.js";

function stepUpFresh(value?:string|null) {
  if (!value) return false;
  const t = new Date(value).getTime();
  return Number.isFinite(t) && Date.now()-t <= 10*60*1000;
}

export async function createSensitiveActionRequest(
  client:PoolClient,
  input:{
    caseId?:string|null;
    clientId:string;
    actionType:string;
    requestedByStaffId:string;
    requesterRole:string;
    reasonCode:string;
    justification:string;
    payload:unknown;
    stepUpVerifiedAt?:string|null;
  }
) {
  const p = ACTION_POLICIES[input.actionType];
  if (!p) throw new HttpError(400,"unknown_sensitive_action");
  if (!p.allowedRequesterRoles.includes(input.requesterRole)) {
    throw new HttpError(403,"requester_role_not_allowed");
  }
  if (p.requiresStepUp && !stepUpFresh(input.stepUpVerifiedAt)) {
    throw new HttpError(403,"step_up_required");
  }

  const r = await client.query(`
    INSERT INTO sensitive_action_requests(
      case_id,client_id,action_type,requested_by_staff_id,reason_code,
      justification,payload,required_approvals,requires_step_up,expires_at
    )
    VALUES ($1::uuid,$2::uuid,$3,$4::uuid,$5,$6,$7::jsonb,$8,$9,now()+interval '30 minutes')
    RETURNING *
  `,[
    input.caseId ?? null,input.clientId,input.actionType,input.requestedByStaffId,
    input.reasonCode,input.justification,JSON.stringify(input.payload),
    p.requiredApprovals,p.requiresStepUp
  ]);

  await enqueueEvent(client,{
    aggregateType:"approval",aggregateId:r.rows[0].id,eventType:"approval.requested",
    eventKey:r.rows[0].id,payload:{
      requestId:r.rows[0].id,actionType:input.actionType,clientId:input.clientId,
      caseId:input.caseId ?? null
    }
  });

  return r.rows[0];
}

export async function decideSensitiveAction(
  client:PoolClient,
  input:{
    requestId:string;
    approverStaffId:string;
    approverRole:string;
    decision:"APPROVE"|"REJECT";
    note?:string|null;
    stepUpVerifiedAt?:string|null;
  }
) {
  const reqR = await client.query(`
    SELECT * FROM sensitive_action_requests
    WHERE id=$1::uuid
    FOR UPDATE
  `,[input.requestId]);
  if (!reqR.rowCount) throw new HttpError(404,"approval_request_not_found");

  const req = reqR.rows[0];
  if (req.status !== "PENDING") throw new HttpError(409,"approval_request_not_pending");
  if (new Date(req.expires_at).getTime() <= Date.now()) {
    await client.query(`UPDATE sensitive_action_requests SET status='EXPIRED',updated_at=now() WHERE id=$1::uuid`,[input.requestId]);
    throw new HttpError(409,"approval_request_expired");
  }

  const p = ACTION_POLICIES[req.action_type];
  if (!p.allowedApproverRoles.includes(input.approverRole)) {
    throw new HttpError(403,"approver_role_not_allowed");
  }
  if (req.requires_step_up && !stepUpFresh(input.stepUpVerifiedAt)) {
    throw new HttpError(403,"step_up_required");
  }
  if (req.requested_by_staff_id === input.approverStaffId && req.required_approvals > 1) {
    throw new HttpError(409,"requester_cannot_self_approve_dual_control");
  }

  await client.query(`
    INSERT INTO sensitive_action_approvals(request_id,approver_staff_id,decision,note)
    VALUES ($1::uuid,$2::uuid,$3,$4)
  `,[input.requestId,input.approverStaffId,input.decision,input.note ?? null]);

  if (input.decision === "REJECT") {
    await client.query(`
      UPDATE sensitive_action_requests
      SET status='REJECTED',updated_at=now()
      WHERE id=$1::uuid
    `,[input.requestId]);

    await enqueueEvent(client,{
      aggregateType:"approval",aggregateId:input.requestId,eventType:"approval.rejected",
      eventKey:input.requestId,payload:{requestId:input.requestId}
    });

    return {status:"REJECTED"};
  }

  const countR = await client.query(`
    SELECT count(*)::int approved
    FROM sensitive_action_approvals
    WHERE request_id=$1::uuid AND decision='APPROVE'
  `,[input.requestId]);

  const approved = Number(countR.rows[0].approved);
  if (approved >= Number(req.required_approvals)) {
    await client.query(`
      UPDATE sensitive_action_requests
      SET status='APPROVED',updated_at=now()
      WHERE id=$1::uuid
    `,[input.requestId]);

    await enqueueEvent(client,{
      aggregateType:"approval",aggregateId:input.requestId,eventType:"approval.approved",
      eventKey:input.requestId,payload:{
        requestId:input.requestId,actionType:req.action_type,clientId:req.client_id,caseId:req.case_id
      }
    });

    return {status:"APPROVED"};
  }

  return {status:"PENDING",approved,required:Number(req.required_approvals)};
}

export async function processSlaBatch(client:PoolClient) {
  const r = await client.query(`
    SELECT s.*,c.public_id,c.owner_staff_id,c.status
    FROM case_sla_state s
    JOIN support_cases c ON c.id=s.case_id
    WHERE c.status NOT IN ('RESOLVED','CLOSED')
    FOR UPDATE OF s SKIP LOCKED
    LIMIT 200
  `);

  for (const row of r.rows) {
    const now = Date.now();

    if (row.first_response_due_at && !row.first_response_breached_at) {
      const due = new Date(row.first_response_due_at).getTime();
      if (!row.first_response_warned_at && due-now <= 5*60*1000 && due>now) {
        await client.query(`
          UPDATE case_sla_state SET first_response_warned_at=now(),updated_at=now()
          WHERE case_id=$1::uuid
        `,[row.case_id]);

        await enqueueEvent(client,{
          aggregateType:"case",aggregateId:row.case_id,eventType:"sla.first_response.warning",
          eventKey:row.case_id,payload:{caseId:row.case_id,publicId:row.public_id,ownerStaffId:row.owner_staff_id}
        });
      }
      if (due <= now) {
        await client.query(`
          UPDATE case_sla_state SET first_response_breached_at=now(),updated_at=now()
          WHERE case_id=$1::uuid
        `,[row.case_id]);

        await enqueueEvent(client,{
          aggregateType:"case",aggregateId:row.case_id,eventType:"sla.first_response.breached",
          eventKey:row.case_id,payload:{caseId:row.case_id,publicId:row.public_id,ownerStaffId:row.owner_staff_id}
        });
      }
    }

    if (row.resolution_due_at && !row.resolution_breached_at) {
      const due = new Date(row.resolution_due_at).getTime();
      if (!row.resolution_warned_at && due-now <= 30*60*1000 && due>now) {
        await client.query(`
          UPDATE case_sla_state SET resolution_warned_at=now(),updated_at=now()
          WHERE case_id=$1::uuid
        `,[row.case_id]);

        await enqueueEvent(client,{
          aggregateType:"case",aggregateId:row.case_id,eventType:"sla.resolution.warning",
          eventKey:row.case_id,payload:{caseId:row.case_id,publicId:row.public_id,ownerStaffId:row.owner_staff_id}
        });
      }
      if (due <= now) {
        await client.query(`
          UPDATE case_sla_state SET resolution_breached_at=now(),updated_at=now()
          WHERE case_id=$1::uuid
        `,[row.case_id]);

        await enqueueEvent(client,{

          eventKey:row.case_id,payload:{caseId:row.case_id,publicId:row.public_id,ownerStaffId:row.owner_staff_id}
        });
      }
    }
  }
}

import { WebSocketServer, WebSocket } from "ws";
import type { Server } from "node:http";
import Redis from "ioredis";

type Client = {
  socket: WebSocket;
  staffId: string;
  role: string;
  topics: Set<string>;
};

const clients = new Set<Client>();

export function attachOpsWebSocket(server:Server) {
  const wss = new WebSocketServer({server,path:"/ws/ops"});

  wss.on("connection",(socket,request)=>{
    // Production: validate the authenticated staff session/cookie/JWT here.
    const url = new URL(request.url ?? "", "http://localhost");
    const staffId = url.searchParams.get("staffId");
    const role = url.searchParams.get("role");

    if (!staffId || !role) {
      socket.close(4401,"Unauthorized");
      return;
    }

    const c:Client = {socket,staffId,role,topics:new Set([`staff:${staffId}`])};
    if (["SUPER_ADMIN","OPS_ADMIN","SUPERVISOR"].includes(role)) {
      c.topics.add("admin");
    }
    clients.add(c);

    socket.on("message",raw=>{
      try {
        const msg = JSON.parse(String(raw));
        if (msg.type==="subscribe.case" && typeof msg.caseId==="string") {
          c.topics.add(`case:${msg.caseId}`);
        }
        if (msg.type==="unsubscribe.case" && typeof msg.caseId==="string") {
          c.topics.delete(`case:${msg.caseId}`);
        }
      } catch {}
    });

    socket.on("close",()=>clients.delete(c));
  });

  return wss;
}

function fanout(event:any) {
  const payload = JSON.stringify(event);
  const eventPayload = event.payload ?? {};
  const topics = new Set<string>(["admin"]);

  if (eventPayload.caseId) topics.add(`case:${eventPayload.caseId}`);
  if (eventPayload.toStaffId) topics.add(`staff:${eventPayload.toStaffId}`);
  if (eventPayload.ownerStaffId) topics.add(`staff:${eventPayload.ownerStaffId}`);

  for (const c of clients) {
    if (c.socket.readyState !== WebSocket.OPEN) continue;
    if ([...topics].some(t=>c.topics.has(t))) c.socket.send(payload);
  }
}

export function startRedisRealtimeBridge() {
  const redis = new Redis(process.env.REDIS_URL ?? "redis://localhost:6379");
  let lastId = "$";

  async function loop() {
    while (true) {
      const result = await redis.xread(
        "BLOCK",5000,
        "COUNT",100,
        "STREAMS","linqa:ops:events",lastId
      ) as any;

      if (!result) continue;

      for (const [,entries] of result) {
        for (const [id,fields] of entries) {
          lastId = id;
          const obj:any = {};
          for (let i=0;i<fields.length;i+=2) obj[fields[i]]=fields[i+1];
          fanout({
            type:obj.eventType,
            eventKey:obj.eventKey,
            payload:JSON.parse(obj.payload)
          });
        }
      }
    }
  }

  loop().catch(e=>console.error("realtime redis bridge failed",e));
  return redis;
}

import "dotenv/config";
import http from "node:http";
import express from "express";
import helmet from "helmet";
import cors from "cors";
import compression from "compression";
import { requestContext,requireStaff } from "./auth/middleware.js";
import { opsRouter } from "./api/routes.js";
import { attachOpsWebSocket,startRedisRealtimeBridge } from "./realtime/ws-gateway.js";

export const app = express();

app.disable("x-powered-by");
app.use(helmet());
app.use(cors({
  origin:(process.env.OPS_WEB_ORIGINS ?? "http://localhost:3000").split(","),
  credentials:true
}));
app.use(compression());
app.use(express.json({limit:"1mb"}));
app.use(requestContext);

app.get("/health",(_req,res)=>res.json({ok:true}));
app.use("/api/ops",requireStaff,opsRouter);

app.use((err:any,_req:any,res:any,_next:any)=>{
  const status = err.status ?? (err.name==="ZodError" ? 400 : 500);
  if (status>=500) console.error(err);
  res.status(status).json({
    error:status===500?"internal_server_error":err.message,
    details:err.name==="ZodError"?err.issues:err.details
  });
});

const server = http.createServer(app);
attachOpsWebSocket(server);
startRedisRealtimeBridge();

if (process.env.NODE_ENV !== "test") {
  const port = Number(process.env.PORT ?? 8080);
  server.listen(port,()=>console.log(`Linqa Ops Orchestration listening on :${port}`));
}
