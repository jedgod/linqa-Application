# Linqa Staff Support Workspace

This pack implements the staff-support interface shown in the supplied design.

## Included
- My Work / active case workspace
- Case header with status, priority, SLA, created/updated timestamps
- Case details and tags
- Client 360 summary with masked identity data
- Account & services summary
- Staff/client support chat
- Internal notes
- Case timeline/history
- Quick actions
- Recovery-verification API
- 2FA-reset authorization API
- Secure recovery links
- Escalation and resolution APIs
- Presence API
- WebSocket case-event contract
- RBAC
- PostgreSQL persistence
- Audit logging

## Security model

Ordinary support agents can communicate with the client, add notes, escalate,
and issue limited secure recovery links. Sensitive identity/recovery actions
are restricted to specialist roles.

The staff workspace must NOT expose:
- E2EE private keys or decrypted unrelated private messages
- Vault encryption keys or arbitrary Vault plaintext
- Passwords
- recovery secrets
- CVVs
- full payment-card PANs
- unrelated personal data

## Production wiring

1. Replace sample x-linqa-* header principal resolution with Linqa staff SSO/JWT.
2. Require MFA/passkey for staff.
3. Add CSRF protection if using cookie sessions.
4. Put sensitive recovery actions behind step-up authentication.
5. Deliver secure-link tokens through an approved user notification channel.
6. Connect outbound client replies to the Linqa support messaging gateway.
7. Publish case updates through Redis/NATS/Kafka for multi-instance WebSocket fanout.
8. Send immutable audit events to SIEM/WORM storage.
9. Add rate limits and per-case action throttles.
10. Bind account recovery to Linqa's actual identity service—this pack authorizes the operation; it does not directly mutate authentication secrets.

## Quick start

npm install
cp .env.example .env
npm run db:migrate
npm run dev:api
npm run dev:web

Mount `StaffSupportPage` under a route such as:
`app/staff/cases/[id]/page.tsx`.

"use client";
import { useMemo, useState } from "react";
import { Send, Smile, Paperclip, ShieldCheck, FileText } from "lucide-react";
import type { ChatMessage } from "../shared/types";
import { Button, Textarea } from "./ui";

export function SupportChat({
  messages, onSend, onAddNote, sending
}:{
  messages:ChatMessage[];
  onSend:(body:string)=>Promise<void>;
  onAddNote:(body:string)=>Promise<void>;
  sending:boolean;
}) {
  const [mode,setMode] = useState<"reply"|"note">("reply");
  const [text,setText] = useState("");

  async function submit() {
    const body = text.trim();
    if (!body) return;
    if (mode==="reply") await onSend(body);
    else await onAddNote(body);
    setText("");
  }

  return (
    <div className="flex min-h-[560px] flex-col">
      <div className="flex-1 space-y-4 overflow-y-auto p-4">
        {messages.map(m=>(
          <div key={m.id} className={`flex ${m.senderType==="STAFF"?"justify-end":"justify-start"}`}>
            <div className={`max-w-[72%] rounded-xl px-4 py-3 ${m.senderType==="STAFF"?"bg-emerald-800/80":"bg-slate-800"}`}>
              <div className="mb-1 text-xs font-medium text-slate-300">{m.senderName}</div>
              <div className="whitespace-pre-wrap text-sm text-slate-100">{m.body}</div>
              <div className="mt-2 text-right text-[10px] text-slate-400">
                {new Date(m.createdAt).toLocaleTimeString([], {hour:"2-digit",minute:"2-digit"})}
              </div>
            </div>
          </div>
        ))}
      </div>

      <div className="border-t border-slate-800 p-3">
        <div className="mb-2 flex gap-5 text-sm">
          <button className={mode==="reply"?"text-violet-400":"text-slate-400"} onClick={()=>setMode("reply")}>Reply</button>
          <button className={mode==="note"?"text-violet-400":"text-slate-400"} onClick={()=>setMode("note")}>Internal Note</button>
        </div>
        <Textarea value={text} onChange={e=>setText(e.target.value)} placeholder={mode==="reply"?"Type your message…":"Add a private internal note…"} />
        <div className="mt-2 flex items-center justify-between">
          <div className="flex gap-3 text-slate-400"><Smile className="h-4 w-4"/><Paperclip className="h-4 w-4"/><FileText className="h-4 w-4"/><ShieldCheck className="h-4 w-4"/></div>
          <Button disabled={sending||!text.trim()} onClick={submit} className="bg-violet-600 text-white hover:bg-violet-500"><Send className="mr-2 h-4 w-4"/>Send</Button>
        </div>
      </div>
    </div>
  );
}

"use client";

import { useMemo, useState } from "react";
import useSWR from "swr";
import { Bell, Clock3, MoreVertical, CheckCircle2 } from "lucide-react";
import type { StaffCaseWorkspacePayload } from "../shared/types";
import { StaffSidebar } from "../components/StaffSidebar";
import { SupportChat } from "../components/SupportChat";
import { QuickActions } from "../components/QuickActions";
import { Card, Badge, Button } from "../components/ui";

const fetcher = async (url:string) => {
  const r = await fetch(url,{credentials:"include"});
  if (!r.ok) throw new Error("request_failed");
  return r.json();
};

function fmtDate(v:string) {
  return new Date(v).toLocaleString([], {month:"short",day:"numeric",year:"numeric",hour:"numeric",minute:"2-digit"});
}

export default function StaffSupportPage({caseId}:{caseId:string}) {
  const {data,error,isLoading,mutate} = useSWR<StaffCaseWorkspacePayload>(`/api/staff/cases/${caseId}/workspace`,fetcher,{
    refreshInterval:15000
  });
  const [sending,setSending] = useState(false);
  const [busy,setBusy] = useState(false);

  async function api(path:string,body:unknown) {
    const r = await fetch(path,{
      method:"POST",
      headers:{"content-type":"application/json"},
      credentials:"include",
      body:JSON.stringify(body)
    });
    const json = await r.json().catch(()=>({}));
    if (!r.ok) throw new Error(json.error ?? "request_failed");
    return json;
  }

  async function sendReply(body:string) {
    setSending(true);
    try {
      await api(`/api/staff/cases/${caseId}/reply`,{body});
      await mutate();
    } finally { setSending(false); }
  }

  async function addNote(body:string) {
    setSending(true);
    try {
      await api(`/api/staff/cases/${caseId}/notes`,{body});
      await mutate();
    } finally { setSending(false); }
  }

  async function action(a:string) {
    if (!data) return;
    setBusy(true);
    try {
      if (a==="escalate") {
        await api(`/api/staff/cases/${caseId}/escalate`,{
          reasonCode:"AGENT_ESCALATION",
          note:"Escalated by support agent for specialist review."
        });
      } else if (a==="secure") {
        await api(`/api/staff/cases/${caseId}/secure-link`,{
          purpose:"RECOVERY_PROOF",reasonCode:"CLIENT_RECOVERY"
        });
      } else if (a==="verify") {
        await api(`/api/staff/cases/${caseId}/recovery/verify`,{
          smsVerified:true,reasonCode:"SMS_OTP_VERIFIED"
        });
      } else if (a==="reset2fa") {
        await api(`/api/staff/cases/${caseId}/recovery/reset-2fa`,{
          reasonCode:"RECOVERY_VERIFIED"
        });
      } else {
        alert("This action should open a specialist workflow modal.");
      }
      await mutate();
    } catch(e:any) {
      alert(e.message);
    } finally { setBusy(false); }
  }

  if (isLoading) return <div className="min-h-screen bg-[#050d16] p-10 text-white">Loading case workspace…</div>;
  if (error || !data) return <div className="min-h-screen bg-[#050d16] p-10 text-white">Unable to load staff workspace.</div>;

  const c = data.supportCase;
  const client = data.client;
  const session = data.session;
  const slaTarget = c.slaTargetAt ? new Date(c.slaTargetAt).getTime() : null;
  const msLeft = slaTarget ? Math.max(0,slaTarget-Date.now()) : 0;
  const mins = Math.floor(msLeft/60000);
  const secs = Math.floor((msLeft%60000)/1000);

  return (
    <div className="min-h-screen bg-[#050d16] text-slate-100">
      <div className="mx-auto flex max-w-[1920px]">
        <StaffSidebar name={session.displayName} role={session.role}/>

        <main className="min-w-0 flex-1">
          <header className="sticky top-0 z-30 flex h-16 items-center justify-between border-b border-slate-800 bg-[#050d16]/95 px-5 backdrop-blur">
            <div className="flex items-center gap-3 text-sm">
              <button className="text-slate-400">☰</button>
              <span>My Work</span><span className="text-slate-600">/</span><span>{c.publicId}</span>
            </div>
            <div className="flex items-center gap-3">
              <div className="hidden rounded-lg border border-slate-800 px-3 py-2 text-xs md:block"><Clock3 className="mr-2 inline h-4 w-4 text-emerald-400"/>Active Session</div>
              <button className="relative rounded-lg p-2 hover:bg-slate-800"><Bell className="h-5 w-5"/><span className="absolute -right-1 -top-1 rounded-full bg-rose-500 px-1.5 text-[10px]">5</span></button>
              <select className="rounded-lg border border-slate-800 bg-slate-900 px-3 py-2 text-xs"><option>Available</option><option>Busy</option><option>Away</option></select>
            </div>
          </header>

          <div className="space-y-3 p-4">
            <Card className="flex flex-wrap items-center gap-6 p-4">
              <div className="min-w-[260px] flex-1">
                <Badge className="bg-emerald-500/15 text-emerald-300">{c.status}</Badge>
                <div className="mt-1 text-xl font-bold">{c.publicId}</div>
                <div className="text-sm text-slate-400">{c.type} <span className="mx-2">›</span> {c.category}</div>
              </div>
              <div><div className="text-xs text-slate-500">Priority</div><Badge className="mt-2 border border-amber-600 bg-amber-950/40 text-amber-300">{c.priority} Medium</Badge></div>
              <div className="border-l border-slate-800 pl-6"><div className="text-xs text-slate-500">SLA</div><div className="mt-2 text-sm font-semibold text-emerald-400">{String(mins).padStart(2,"0")}m {String(secs).padStart(2,"0")}s left</div></div>
              <div className="border-l border-slate-800 pl-6"><div className="text-xs text-slate-500">Created</div><div className="mt-2 text-xs">{fmtDate(c.createdAt)}</div></div>
              <div className="border-l border-slate-800 pl-6"><div className="text-xs text-slate-500">Updated</div><div className="mt-2 text-xs">{fmtDate(c.updatedAt)}</div></div>
              <MoreVertical className="h-5 w-5 text-slate-400"/>

                <Card className="p-4">

import { Pool, PoolClient, QueryResultRow } from "pg";

export const pool = new Pool({
  connectionString: process.env.DATABASE_URL,
  max: Number(process.env.PG_POOL_MAX ?? 20),
  idleTimeoutMillis: 30_000,
  connectionTimeoutMillis: 5_000,
  ssl: process.env.NODE_ENV === "production" ? { rejectUnauthorized: true } : false,
});

export async function query<T extends QueryResultRow = any>(sql: string, params: unknown[] = []) {
  return pool.query<T>(sql, params);
}

export async function tx<T>(fn: (client: PoolClient) => Promise<T>) {
  const client = await pool.connect();
  try {
    await client.query("BEGIN");
    const result = await fn(client);
    await client.query("COMMIT");
    return result;
  } catch (e) {
    await client.query("ROLLBACK");
    throw e;
  } finally {
    client.release();
  }
}

import type { Request, Response, NextFunction } from "express";
import { can, type StaffPermission } from "../shared/permissions";

export interface StaffPrincipal {
  userId: string;
  staffId: string;
  role: string;
  displayName: string;
}

declare global {
  namespace Express {
    interface Request {
      principal?: StaffPrincipal;
      requestId?: string;
    }
  }
}

export function requireStaff(req: Request, res: Response, next: NextFunction) {
  // Replace with Linqa staff SSO/JWT session validation in production.
  const userId = String(req.headers["x-linqa-user-id"] ?? "");
  const staffId = String(req.headers["x-linqa-staff-id"] ?? "");
  const role = String(req.headers["x-linqa-role"] ?? "");
  const displayName = String(req.headers["x-linqa-staff-name"] ?? "");

  if (!userId || !staffId || !role) {
    return res.status(401).json({ error: "unauthorized" });
  }

  req.principal = { userId, staffId, role, displayName: displayName || "Staff" };
  next();
}

export function requirePermission(permission: StaffPermission) {
  return (req: Request, res: Response, next: NextFunction) => {
    if (!req.principal) return res.status(401).json({ error: "unauthorized" });
    if (!can(req.principal.role, permission)) {
      return res.status(403).json({ error: "forbidden", permission });
    }
    next();
  };
}

import crypto from "node:crypto";
import type { Request } from "express";
import { query } from "./db";

export async function audit(
  req: Request,
  input: {
    action: string;
    resourceType: string;
    resourceId?: string;
    reasonCode?: string;
    before?: unknown;
    after?: unknown;
  }
) {
  const ipHash = req.ip
    ? crypto.createHash("sha256").update(req.ip).digest("hex")
    : null;

  await query(`
    INSERT INTO support_audit_log (
      actor_user_id, actor_staff_id, actor_role, action, resource_type,
      resource_id, reason_code, request_id, ip_hash, before_json, after_json
    )
    VALUES ($1::uuid,$2::uuid,$3,$4,$5,$6,$7,$8,$9,$10::jsonb,$11::jsonb)
  `, [
    req.principal?.userId ?? null,
    req.principal?.staffId ?? null,
    req.principal?.role ?? null,
    input.action,
    input.resourceType,
    input.resourceId ?? null,
    input.reasonCode ?? null,
    req.requestId ?? null,
    ipHash,
    JSON.stringify(input.before ?? null),
    JSON.stringify(input.after ?? null),
  ]);
}

export async function getCaseWorkspace(caseId: string, staffId: string): Promise<StaffCaseWorkspacePayload> {
  const [caseR, clientR, messagesR, notesR, timelineR, recoveryR, staffR, tagsR] = await Promise.all([
    query(`
      SELECT c.id::text, c.public_id "publicId", c.client_id::text "clientId",
        c.case_type "type", c.category, c.channel, c.status::text, c.priority::text,
        q.name "queueName", c.assigned_staff_id::text "assignedStaffId",
        s.display_name "assignedStaffName", c.source, c.reference_id "referenceId",
        c.description, c.created_at "createdAt", c.updated_at "updatedAt",
        (c.created_at + make_interval(secs => COALESCE(q.sla_seconds,1800))) "slaTargetAt"
      FROM support_cases c
      LEFT JOIN support_queues q ON q.id=c.queue_id
      LEFT JOIN support_staff s ON s.id=c.assigned_staff_id
      WHERE c.id=$1::uuid
    `, [caseId]),
    query(`
      SELECT client_id::text id, display_name "displayName", masked_email "maskedEmail",
        masked_phone "maskedPhone", account_status "accountStatus", member_since "memberSince",
        risk_level "riskLevel", country, verified,
        jsonb_build_object(
          'linqaAccount',linqa_account_active,
          'linqaSend',linqasend_active,
          'vaultPlan',vault_plan,
          'twoFactorEnabled',two_factor_enabled,
          'trustedDevices',trusted_devices
        ) services
      FROM support_client_snapshot
      WHERE client_id=(SELECT client_id FROM support_cases WHERE id=$1::uuid)
    `, [caseId]),
    query(`
      SELECT id::text, case_id::text "caseId", sender_type::text "senderType",
        sender_name "senderName", body, created_at "createdAt",
        delivery_status "deliveryStatus", is_internal "isInternal"
      FROM support_messages
      WHERE case_id=$1::uuid AND is_internal=false
      ORDER BY created_at ASC
    `, [caseId]),
    query(`
      SELECT n.id::text, n.case_id::text "caseId", s.display_name "authorName",
        n.body, n.created_at "createdAt"
      FROM support_internal_notes n
      JOIN support_staff s ON s.id=n.author_staff_id
      WHERE n.case_id=$1::uuid
      ORDER BY n.created_at DESC
    `, [caseId]),
    query(`
      SELECT t.id::text, t.event_type type, t.text, t.created_at "createdAt",
        s.display_name "actorName"
      FROM support_case_timeline t
      LEFT JOIN support_staff s ON s.id=t.actor_staff_id
      WHERE t.case_id=$1::uuid
      ORDER BY t.created_at ASC
    `, [caseId]),
    query(`
      SELECT sms_verified "smsVerified", email_verified "emailVerified",
        trusted_device_matched "trustedDeviceMatched", carrier_risk "carrierRisk",
        risk_score::float8 "riskScore", decision
      FROM support_recovery_reviews WHERE case_id=$1::uuid
    `, [caseId]),
    query(`
      SELECT id::text "staffId", user_id::text "userId", display_name "displayName",
        role_name role, presence::text, session_started_at "sessionStartedAt"
      FROM support_staff WHERE id=$1::uuid
    `, [staffId]),
    query(`SELECT tag FROM support_case_tags WHERE case_id=$1::uuid ORDER BY tag`, [caseId])
  ]);

  if (!caseR.rowCount) throw Object.assign(new Error("case_not_found"), { status: 404 });
  if (!staffR.rowCount) throw Object.assign(new Error("staff_not_found"), { status: 404 });

  const supportCase = {
    ...caseR.rows[0],
    tags: tagsR.rows.map((r:any)=>r.tag)
  } as any;

  return {
    session: staffR.rows[0] as any,
    supportCase,
    client: clientR.rows[0] as any,
    messages: messagesR.rows as any,
    notes: notesR.rows as any,
    timeline: timelineR.rows as any,
    recovery: recoveryR.rows[0] ?? null
  };
}

import { WebSocketServer, WebSocket } from "ws";
import type { Server } from "node:http";

type Connection = {
  socket: WebSocket;
  staffId: string;
  caseIds: Set<string>;
};

const connections = new Set<Connection>();

export function attachSupportRealtime(server: Server) {
  const wss = new WebSocketServer({ server, path: "/ws/support" });

  wss.on("connection", (socket, request) => {
    // Replace query-string auth with signed staff session/cookie validation.
    const url = new URL(request.url ?? "", "http://localhost");
    const staffId = url.searchParams.get("staffId");
    if (!staffId) return socket.close(4401, "Unauthorized");

    const conn: Connection = { socket, staffId, caseIds: new Set() };
    connections.add(conn);

    socket.on("message", raw => {
      try {
        const msg = JSON.parse(String(raw));
        if (msg.type === "subscribe.case" && typeof msg.caseId === "string") {
          conn.caseIds.add(msg.caseId);
        }
        if (msg.type === "unsubscribe.case" && typeof msg.caseId === "string") {
          conn.caseIds.delete(msg.caseId);
        }
      } catch {}
    });

    socket.on("close", () => connections.delete(conn));
  });
}

export function publishCaseEvent(caseId: string, event: unknown) {
  const payload = JSON.stringify({ type: "case.event", caseId, event });
  for (const c of connections) {
    if (c.caseIds.has(caseId) && c.socket.readyState === WebSocket.OPEN) {
      c.socket.send(payload);
    }
  }
}
