ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/../.." && pwd)"
EVIDENCE_DIR="${LINQA_EVIDENCE_DIR:-$ROOT/certification/evidence}"
mkdir -p "$EVIDENCE_DIR"

fail() { echo "FAIL: $*" >&2; exit 1; }
pass() { echo "PASS: $*"; }
require_cmd() { command -v "$1" >/dev/null 2>&1 || fail "missing required command: $1"; }

json_escape() {
  python3 -c 'import json,sys; print(json.dumps(sys.stdin.read()))'
}

sha256_file() {
  if command -v sha256sum >/dev/null 2>&1; then sha256sum "$1" | awk '{print $1}'
  else shasum -a 256 "$1" | awk '{print $1}'
  fi
}

git_sha() {
  git -C "$ROOT" rev-parse HEAD
}

git_dirty() {
  if [[ -n "$(git -C "$ROOT" status --porcelain)" ]]; then echo true; else echo false; fi
}

write_gate() {
  local gate="$1" status="$2" detail="$3"
  python3 - "$EVIDENCE_DIR/gates.json" "$gate" "$status" "$detail" <<'PY'
import json,sys,os,datetime
path,gate,status,detail=sys.argv[1:]
data={}
if os.path.exists(path):
    with open(path,encoding="utf-8") as f: data=json.load(f)
data[gate]={
    "status":status,
    "detail":detail,
    "recordedAt":datetime.datetime.now(datetime.timezone.utc).isoformat()
}
tmp=path+".tmp"
with open(tmp,"w",encoding="utf-8") as f:
    json.dump(data,f,indent=2,sort_keys=True)
os.replace(tmp,path)
PY
}

const FORBIDDEN_KEYS = new Set([
  "plaintext",
  "plaintextUtf8",
  "message",
  "messageBody",
  "bodyText",
  "decrypted",
  "decryptedContent",
  "contentText",
  "cleartext",
]);

export function assertNoPlaintextFields(input: unknown, path = "$"): void {
  if (input === null || input === undefined) return;

  if (Array.isArray(input)) {
    input.forEach((v, i) => assertNoPlaintextFields(v, `${path}[${i}]`));
    return;
  }

  if (typeof input !== "object") return;

  for (const [key, value] of Object.entries(input as Record<string, unknown>)) {
    if (FORBIDDEN_KEYS.has(key)) {
      throw new Error(`forbidden_plaintext_field:${path}.${key}`);
    }
    assertNoPlaintextFields(value, `${path}.${key}`);
  }
}

api() {
  local method="$1" token="$2" path="$3" body="${4:-}"
  local args=(-sS -D /tmp/linqa_headers -o /tmp/linqa_body -w "%{http_code}"
    -X "$method"
    -H "authorization: Bearer $token"
    -H "content-type: application/json"
    "${LINQA_BASE_URL}${path}")
  if [[ -n "$body" ]]; then args+=("--data-binary" "$body"); fi
  local code
  code="$(curl "${args[@]}")"
  cat /tmp/linqa_body
  echo
  [[ "$code" =~ ^2 ]] || return "$code"
}

package com.linqa.certification

import org.json.JSONObject
import java.time.Instant

data class CertificationEvent(
    val runId: String,
    val gitSha: String,
    val deviceRole: String,
    val deviceId: String,
    val userId: String,
    val event: String,
    val status: String,
    val evidence: Map<String, Any?> = emptyMap()
) {
    fun toJson(): String {
        val ev = JSONObject()
        evidence.forEach { (k, v) -> ev.put(k, v) }

        return JSONObject()
            .put("schemaVersion", 1)
            .put("runId", runId)
            .put("gitSha", gitSha)
            .put("deviceRole", deviceRole)
            .put("deviceId", deviceId)
            .put("userId", userId)
            .put("event", event)
            .put("status", status)
            .put("timestamp", Instant.now().toString())
            .put("evidence", ev)
            .toString()
    }
}

package com.linqa.certification

import android.content.Context
import java.io.File

class CertificationRecorder(private val context: Context) {
    private val file: File
        get() = File(context.filesDir, "linqa-certification-events.jsonl")

    @Synchronized
    fun record(event: CertificationEvent) {
        // Evidence must contain IDs/hashes/state only. Never plaintext or key material.
        file.appendText(event.toJson() + "\n", Charsets.UTF_8)
    }

    fun filePath(): String = file.absolutePath
}

package com.linqa.certification

/**
 * Wire these calls into the REAL send/receive/key lifecycle paths.
 * They do not make a test pass themselves; they only capture device-side evidence.
 */
class CertificationHooks(
    private val recorder: CertificationRecorder,
    private val runId: String,
    private val gitSha: String,
    private val role: String,
    private val userId: String,
    private val deviceId: String,
) {
    fun sendAccepted(envelopeId: String, clientMessageId: String) =
        rec("SEND_ACCEPTED", mapOf(
            "envelopeId" to envelopeId,
            "clientMessageId" to clientMessageId
        ))

    fun receiveDecrypted(envelopeId: String, clientMessageId: String, plaintextSha256: String) =
        rec("RECEIVE_DECRYPTED", mapOf(
            "envelopeId" to envelopeId,
            "clientMessageId" to clientMessageId,
            "plaintextSha256" to plaintextSha256
        ))

    fun offlineEnter() = rec("OFFLINE_ENTER")
    fun offlineExit() = rec("OFFLINE_EXIT")

    fun duplicateRetry(originalEnvelopeId: String, retryEnvelopeId: String, idempotent: Boolean) =
        rec("DUPLICATE_RETRY", mapOf(
            "originalEnvelopeId" to originalEnvelopeId,
            "retryEnvelopeId" to retryEnvelopeId,
            "idempotent" to idempotent
        ), if (idempotent) "PASS" else "FAIL")

    fun tamperRejected(failureCode: String) =
        rec("TAMPER_REJECTED", mapOf("failureCode" to failureCode),
            if (failureCode in setOf("authentication_failed","invalid_mac","aead_open_failed"))
                "PASS" else "FAIL")

    fun keyRotated(oldPackageId: String, newPackageId: String) =
        rec("KEY_ROTATED", mapOf(
            "oldPackageId" to oldPackageId,
            "newPackageId" to newPackageId
        ))

    fun revokedDeviceBlocked(revokedDeviceId: String) =
        rec("REVOKED_DEVICE_BLOCKED", mapOf("revokedDeviceId" to revokedDeviceId))

    private fun rec(
        event: String,
        evidence: Map<String, Any?> = emptyMap(),
        status: String = "PASS"
    ) {
        recorder.record(
            CertificationEvent(
                runId = runId,
                gitSha = gitSha,
                deviceRole = role,
                deviceId = deviceId,
                userId = userId,
                event = event,
                status = status,
                evidence = evidence
            )
        )
    }
}

#!/usr/bin/env python3
import json, glob, os, sys, hashlib, subprocess
from collections import defaultdict

root = os.path.abspath(os.path.join(os.path.dirname(__file__), ".."))
evdir = os.environ.get("LINQA_EVIDENCE_DIR", os.path.join(root, "certification", "evidence"))
sha = subprocess.check_output(["git","-C",root,"rev-parse","HEAD"], text=True).strip()

events=[]
for path in glob.glob(os.path.join(evdir,"device-*.jsonl")):
    with open(path,encoding="utf-8") as f:
        for line in f:
            line=line.strip()
            if line: events.append(json.loads(line))

if not events:
    raise SystemExit("FAIL: no physical-device evidence JSONL files")

for e in events:
    if e.get("gitSha") != sha:
        raise SystemExit(f"FAIL: evidence SHA mismatch {e.get('gitSha')} != {sha}")
    if e.get("status") != "PASS":
        raise SystemExit(f"FAIL: device event failed: {e}")

roles={e["deviceRole"] for e in events}
if not {"alice","bob"} <= roles:
    raise SystemExit("FAIL: both alice and bob device evidence required")

by_role=defaultdict(list)
for e in events: by_role[e["deviceRole"]].append(e)

def find(role,name):
    return [e for e in by_role[role] if e["event"]==name]

# Bidirectional proof: each sends and each receives.
for role in ("alice","bob"):
    if not find(role,"SEND_ACCEPTED"):
        raise SystemExit(f"FAIL: {role} has no SEND_ACCEPTED")
    if not find(role,"RECEIVE_DECRYPTED"):
        raise SystemExit(f"FAIL: {role} has no RECEIVE_DECRYPTED")

# Offline proof must be on a device with enter -> exit -> receive.
offline=[]
for role in ("alice","bob"):
    seq=by_role[role]
    names=[e["event"] for e in seq]
    if "OFFLINE_ENTER" in names and "OFFLINE_EXIT" in names and "RECEIVE_DECRYPTED" in names:
        i=names.index("OFFLINE_ENTER")
        j=names.index("OFFLINE_EXIT",i+1)
        if any(x=="RECEIVE_DECRYPTED" for x in names[j+1:]):
            offline.append(role)
if not offline:
    raise SystemExit("FAIL: no offline -> reconnect -> decrypt sequence")

if not any(e["event"]=="DUPLICATE_RETRY" and e["evidence"].get("idempotent") is True for e in events):
    raise SystemExit("FAIL: duplicate retry idempotency evidence missing")

if not any(e["event"]=="TAMPER_REJECTED" for e in events):
    raise SystemExit("FAIL: tamper rejection evidence missing")

if not any(e["event"]=="KEY_ROTATED" for e in events):
    raise SystemExit("FAIL: key rotation evidence missing")

if not any(e["event"]=="REVOKED_DEVICE_BLOCKED" for e in events):
    raise SystemExit("FAIL: revoked-device block evidence missing")

summary={
    "gitSha": sha,
    "eventCount": len(events),
    "roles": sorted(roles),
    "offlineProvenOn": offline,
    "requirements": {
        "aliceToBob": True,
        "bobToAlice": True,
        "offlineDelivery": True,
        "duplicateRetry": True,
        "tamperRejected": True,
        "keyRotation": True,
        "revokedDeviceIsolation": True
    }
}
path=os.path.join(evdir,"device-certification-summary.json")
with open(path,"w",encoding="utf-8") as f: json.dump(summary,f,indent=2,sort_keys=True)
print("PASS:",path)

#!/usr/bin/env bash
set -euo pipefail
source "$(dirname "$0")/lib/common.sh"
require_cmd psql

DB="${LINQA_DATABASE_URL:?LINQA_DATABASE_URL required}"
CANARY="${LINQA_PLAINTEXT_CANARY:?LINQA_PLAINTEXT_CANARY required}"

echo "Checking suspicious schema columns..."
BAD_COLS="$(psql "$DB" -Atc "
SELECT table_schema||'.'||table_name||'.'||column_name
FROM information_schema.columns
WHERE table_schema NOT IN ('pg_catalog','information_schema')
  AND lower(column_name) ~ '(plaintext|cleartext|message_body|decrypted_content|content_text)';
")"

if [[ -n "$BAD_COLS" ]]; then
  printf '%s\n' "$BAD_COLS" > "$EVIDENCE_DIR/db-forbidden-columns.txt"
  write_gate C8 FAIL "forbidden plaintext-like database columns detected"
  fail "forbidden plaintext-like columns detected"
fi

# Search textual columns in message/relay/envelope-related tables only.
python3 - "$DB" "$CANARY" "$EVIDENCE_DIR/db-opacity.json" <<'PY'
import subprocess,sys,json,re
db,canary,out=sys.argv[1:]
q=r"""
SELECT table_schema,table_name,column_name
FROM information_schema.columns
WHERE table_schema='public'
  AND data_type IN ('text','character varying','json','jsonb')
  AND lower(table_name) ~ '(message|relay|envelope|mailbox|delivery)';
"""
rows=subprocess.check_output(["psql",db,"-At","-F","\t","-c",q],text=True).splitlines()
hits=[]
for row in rows:
    if not row.strip(): continue
    schema,table,col=row.split("\t")
    ident=lambda s:'"'+s.replace('"','""')+'"'
    sql=f"SELECT count(*) FROM {ident(schema)}.{ident(table)} WHERE CAST({ident(col)} AS text) LIKE '%' || $${canary}$$ || '%';"
    count=int(subprocess.check_output(["psql",db,"-At","-c",sql],text=True).strip() or 0)
    if count: hits.append({"table":table,"column":col,"count":count})
with open(out,"w",encoding="utf-8") as f:
    json.dump({"canaryFound":bool(hits),"hits":hits,"columnsScanned":len(rows)},f,indent=2)
if hits:
    print(json.dumps(hits,indent=2))
    raise SystemExit(1)
PY

write_gate C8 PASS "database schema/content opacity checks passed"
pass "database opacity"

#!/usr/bin/env bash
set -euo pipefail
source "$(dirname "$0")/lib/common.sh"

# This does not prohibit plaintext inside client UI/core code; it prohibits suspicious server-side message persistence.
PATTERN='(plaintext|messageBody|decryptedContent|contentText|cleartext)'
TARGETS=(server/src server/db server/migrations)

hits="$EVIDENCE_DIR/server-plaintext-static-scan.txt"
: > "$hits"

for t in "${TARGETS[@]}"; do
  [[ -e "$ROOT/$t" ]] || continue
  grep -RniE "$PATTERN" "$ROOT/$t" \
    --exclude='forbidden-plaintext.ts' \
    --exclude='*.test.ts' \
    --exclude='*.spec.ts' \
    >> "$hits" || true
done

# Allow documentation/comments to exist, but fail on likely persistence/property declarations.
if grep -Ei '(plaintext|messageBody|decryptedContent|contentText|cleartext).*(TEXT|VARCHAR|JSONB|: string|z\.string)' "$hits" >/dev/null; then
  write_gate C8 FAIL "static scan found likely server plaintext storage/API field"
  fail "review $hits"
fi

pass "static plaintext boundary scan"

DECLARE
  n bigint;
BEGIN
  /*
   * Adapt table name only if the repository's relay mailbox table differs.
   * The intent is strict: no envelope accepted for the revoked device after revocation.
   */
  IF to_regclass('public.envelopes') IS NOT NULL THEN
    EXECUTE format(
      'SELECT count(*) FROM envelopes WHERE recipient_device_id=%L AND accepted_at > %L',
      :'revoked_device_id', :'revoked_at'
    ) INTO n;
  ELSIF to_regclass('public.encrypted_envelopes') IS NOT NULL THEN
    EXECUTE format(
      'SELECT count(*) FROM encrypted_envelopes WHERE recipient_device_id=%L AND server_received_at > %L',
      :'revoked_device_id', :'revoked_at'
    ) INTO n;
  ELSE
    RAISE EXCEPTION 'No recognized relay envelope table';
  END IF;

  IF n <> 0 THEN
    RAISE EXCEPTION 'Revoked device received % post-revocation envelopes', n;
  END IF;
END $$;

import os,glob,json,hashlib,subprocess,datetime

root=os.path.abspath(os.path.join(os.path.dirname(__file__),".."))

records=[]
for p in sorted(glob.glob(os.path.join(evdir,"*"))):
    if os.path.isdir(p): continue
    if os.path.basename(p) in {"manifest.json","manifest.sha256"}: continue
    h=hashlib.sha256()
    with open(p,"rb") as f:
        for chunk in iter(lambda:f.read(1024*1024),b""): h.update(chunk)
    records.append({
        "path":os.path.relpath(p,root).replace("\\","/"),
        "bytes":os.path.getsize(p),
        "sha256":h.hexdigest()
    })

manifest={
    "schemaVersion":1,
    "gitSha":sha,
    "createdAt":datetime.datetime.now(datetime.timezone.utc).isoformat(),
    "files":records
}
mp=os.path.join(evdir,"manifest.json")
with open(mp,"w",encoding="utf-8") as f:json.dump(manifest,f,indent=2,sort_keys=True)
digest=hashlib.sha256(open(mp,"rb").read()).hexdigest()
open(os.path.join(evdir,"manifest.sha256"),"w",encoding="utf-8").write(digest+"  manifest.json\n")
print("PASS:",digest)

# Two-Device Certification Matrix

| Gate | Procedure | Machine evidence | PASS rule |
|---|---|---|---|
| C0 | Compile Rust/server/Android exact SHA | repository.json + build logs | all commands exit 0; clean git |
| C1 | Alice sends to Bob | Alice SEND_ACCEPTED + Bob RECEIVE_DECRYPTED | same logical message; Bob local decrypt |
| C2 | Bob sends to Alice | Bob SEND_ACCEPTED + Alice RECEIVE_DECRYPTED | same logical message; Alice local decrypt |
| C3 | Bob offline | OFFLINE_ENTER -> Alice send -> OFFLINE_EXIT -> RECEIVE_DECRYPTED | queued ciphertext delivered after reconnect |
| C4 | Retry same clientMessageId | DUPLICATE_RETRY | same envelope or explicit idempotent replay; no duplicate local message |
| C5 | Flip ciphertext/AAD bit | TAMPER_REJECTED + Rust test | authentication failure; zero plaintext |
| C6 | Rotate Bob K1 -> K2 | KEY_ROTATED + package lifecycle DB evidence | new sends use K2; stale K1 unusable |
| C7 | Revoke device | REVOKED_DEVICE_BLOCKED + DB query | zero post-revocation envelopes |
| C8 | DB opacity | db-opacity.json | no canary + no forbidden plaintext columns |
| C9 | Log opacity | server.log scan | no canary + no forbidden plaintext keys |
| C10 | Exact-SHA binding | manifest.json | all evidence hashed under current SHA |
| C11 | README promotion | 12_promote_readme.py | impossible unless verdict GREEN |

The plaintext canary should be a unique full message, e.g.:
`LINQA_CERT_CANARY_<timestamp>_<random>`

It MAY exist:
- transiently in sender UI memory
- inside the trusted cryptographic core before encryption
- in recipient UI/local protected store after authenticated decryption
- as a SHA-256 digest in test evidence

It MUST NOT exist:
- relay HTTP request body as plaintext
- server log
- server DB plaintext field
- queue payload
- push notification payload
- analytics/APM/crash breadcrumb
