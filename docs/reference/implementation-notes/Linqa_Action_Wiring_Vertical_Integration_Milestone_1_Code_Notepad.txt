# Linqa Action Wiring & Vertical Integration — Milestone 1

This milestone follows the 34-module interface pack.

It converts the first seven critical modules from UI/action definitions into wired workflows:

1. Authentication
2. Home
3. Chats
4. Conversation
5. Calls
6. File Vault
7. Privacy + Security

## Core design rule

Every action must resolve to one of four explicit states:

- READY: real client + server + persistence + required provider are connected.
- DEGRADED: workflow is available with a documented limitation.
- UNAVAILABLE: dependency is missing.
- FORBIDDEN: authorization or policy blocks the action.

No sensitive action is allowed to return a false success state.

## Vertical slices

### Authentication
email/phone login, OTP challenge, passkey handoff, session refresh, logout.

### Home
profile, unread counts, pinned chats, recent calls, status, notifications,
privacy score, storage summary, security alerts.

### Chats
list, filters, new conversation, archive, pin, mute, unread, message requests.

### Conversation
send encrypted envelope, receive, reply, edit metadata, delete, bookmark,
attachments, location payloads, call launch.

### Calls
create session, ringing state, accept/reject/end, history, scheduled call,
signaling bootstrap.

### File Vault
list, folder create, encrypted-upload init, finalize, download grant,
favorite, soft delete.

### Privacy + Security
privacy center, trusted devices, active sessions, session revoke,
device revoke, security alerts, key-transparency summary, account export/delete initiation.

export class ApiClient {
  constructor(
    private readonly baseUrl:string,
    private readonly getAccessToken:()=>Promise<string>,
  ) {}

  async request<T>(
    path:string,
    init:RequestInit={},
  ):Promise<T> {
    const token=await this.getAccessToken();

    const response=await fetch(`${this.baseUrl}${path}`,{
      ...init,
      headers:{
        "content-type":"application/json",
        authorization:`Bearer ${token}`,
        ...(init.headers??{}),
      },
    });

    if(!response.ok) {
      const payload=await response.json().catch(()=>null);
      throw new Error(
        payload?.error ??
        `API request failed (${response.status})`,
      );
    }

    if(response.status===204) {
      return undefined as T;
    }

    return response.json();
  }
}

import type { ApiClient } from "./ApiClient";

export interface SecureCoreMessaging {
  encrypt(input:{
    conversationId:string;
    plaintext:string;
    messageType:string;
  }):Promise<{
    messageId:string;
    ciphertext:Uint8Array;
    expiresAt?:string;
  }>;

  decrypt(ciphertext:Uint8Array):Promise<{
    messageId:string;
    plaintext:string;
  }>;
}

export class ConversationService {
  constructor(
    private readonly api:ApiClient,
    private readonly core:SecureCoreMessaging,
  ) {}

  async send(input:{
    conversationId:string;
    plaintext:string;
    messageType?:string;
  }) {
    const encrypted=await this.core.encrypt({
      conversationId:input.conversationId,
      plaintext:input.plaintext,
      messageType:input.messageType??"TEXT",
    });

    return this.api.request(
      `/v2/chats/${input.conversationId}/messages`,
      {
        method:"POST",
        body:JSON.stringify({
          messageId:encrypted.messageId,
          ciphertextBase64:toBase64(encrypted.ciphertext),
          messageType:input.messageType??"TEXT",
          expiresAt:encrypted.expiresAt,
        }),
      },
    );
  }

  async decryptServerRows(rows:any[]) {
    const output=[];

    for(const row of rows) {
      if(row.deleted_at) continue;

      try {
        const decrypted=await this.core.decrypt(
          fromBase64(row.envelope_ciphertext),
        );
        output.push({
          ...row,
          plaintext:decrypted.plaintext,
        });
      } catch {
        output.push({
          ...row,
          decryptError:true,
        });
      }
    }

    return output;
  }
}

function toBase64(bytes:Uint8Array) {
  let binary="";
  for(const b of bytes) binary+=String.fromCharCode(b);
  return btoa(binary);
}

function fromBase64(value:string) {
  return Uint8Array.from(
    atob(value),
    c=>c.charCodeAt(0),
  );
}

# Implementation Sequence

## Gate 1 — Authentication
Do not mount authenticated tabs until real login/session refresh works.

## Gate 2 — Home
Home must load only authenticated, user-scoped data.

## Gate 3 — Chats
Users can create and manage real conversation records.

## Gate 4 — Conversation
A message is not displayed as sent until:
1. encrypted by secure core;
2. persisted in local encrypted outbox;
3. accepted by relay.

## Gate 5 — Calls
Call UI reflects signaling state. History is written from terminal call state.

## Gate 6 — File Vault
Files are encrypted client-side before upload. Server receives ciphertext only.

## Gate 7 — Privacy/Security
Users can inspect and revoke sessions/devices and see runtime security state.

## Exit criteria
Two real accounts on two devices can authenticate, exchange an encrypted
message, make a call, upload/download a Vault file, and revoke a session.

import type { FastifyInstance } from "fastify";
import { z } from "zod";
import type { AppConfig } from "../config.js";
import { createAuthenticator,requireScope } from "../auth/jwt.js";
import type { HomeV2Service } from "../services/home-v2.service.js";
import type { ChatV2Service } from "../services/chat-v2.service.js";
import type { ConversationV2Service } from "../services/conversation-v2.service.js";
import type { CallV2Service } from "../services/call-v2.service.js";
import type { VaultV2Service } from "../services/vault-v2.service.js";
import type { PrivacySecurityV2Service } from "../services/privacy-security-v2.service.js";

export async function registerVertical1Routes(
  app:FastifyInstance,
  config:AppConfig,
  services:{
    home:HomeV2Service;
    chats:ChatV2Service;
    conversations:ConversationV2Service;
    calls:CallV2Service;
    vault:VaultV2Service;
    privacy:PrivacySecurityV2Service;
  },
) {
  const authenticate=createAuthenticator(config);

  app.get("/v2/home",async request=>{
    const p=await authenticate(request);
    requireScope(p,"profile:read");
    return services.home.load(p.userId,p.deviceId);
  });

  app.get("/v2/chats",async request=>{
    const p=await authenticate(request);
    requireScope(p,"messages:read");
    const query=z.object({
      filter:z.enum(["ALL","UNREAD","FAVORITES","ARCHIVED"]).optional(),
    }).parse(request.query);
    return {chats:await services.chats.list({
      userId:p.userId,
      filter:query.filter,
    })};
  });

  app.post("/v2/chats/direct",async(request,reply)=>{
    const p=await authenticate(request);
    requireScope(p,"messages:write");
    const body=z.object({
      peerUserId:z.string().uuid(),
    }).parse(request.body);
    return reply.code(201).send(
      await services.chats.createDirect(p.userId,body.peerUserId),
    );
  });

  app.get("/v2/chats/:conversationId/messages",async request=>{
    const p=await authenticate(request);
    requireScope(p,"messages:read");
    const params=z.object({
      conversationId:z.string().uuid(),
    }).parse(request.params);
    const query=z.object({
      limit:z.coerce.number().int().min(1).max(500).default(100),
    }).parse(request.query);

    return {
      messages:await services.conversations.messages(
        p.userId,
        params.conversationId,
        query.limit,
      ),
    };
  });

  app.post(
    "/v2/chats/:conversationId/messages",
    async(request,reply)=>{
      const p=await authenticate(request);
      requireScope(p,"messages:write");

      const params=z.object({
        conversationId:z.string().uuid(),
      }).parse(request.params);

      const body=z.object({
        messageId:z.string().uuid(),
        ciphertextBase64:z.string().min(44),
        messageType:z.string().min(2).max(40).default("TEXT"),
        expiresAt:z.string().datetime().optional(),
      }).parse(request.body);

      const result=await services.conversations.acceptEncryptedMessage({
        userId:p.userId,
        deviceId:p.deviceId,
        conversationId:params.conversationId,
        messageId:body.messageId,
        ciphertext:Buffer.from(body.ciphertextBase64,"base64"),
        messageType:body.messageType,
        expiresAt:body.expiresAt?new Date(body.expiresAt):undefined,
      });

      return reply.code(result.accepted?201:200).send(result);
    },
  );

  app.post("/v2/calls",async(request,reply)=>{
    const p=await authenticate(request);
    requireScope(p,"calls:write");
    const body=z.object({
      peerUserId:z.string().uuid(),
      callType:z.enum(["AUDIO","VIDEO"]),
      privacyMode:z.enum(["STANDARD","RELAY_ONLY"]).default("STANDARD"),
    }).parse(request.body);

    return reply.code(201).send(
      await services.calls.create({
        userId:p.userId,
        deviceId:p.deviceId,
        ...body,
      }),
    );
  });

  app.post("/v2/calls/:callId/state",async(request,reply)=>{
    const p=await authenticate(request);
    requireScope(p,"calls:write");

    const params=z.object({

      input.state === "ACCEPTED" || input.state === "ACTIVE"

        : input.state === "ENDED"

       WHERE ${clauses.join(" AND ")}

         ) VALUES($1,$2,$3,$4,$5,$6,$7,$8)`,
        [
          objectId,
          input.userId,
          input.folderId ?? null,
          objectKey,
          input.nameCiphertext,
          input.metadataCiphertext,
          input.keyEnvelope,
          input.contentHash,
        ],
      );

      await client.query(
        `INSERT INTO vault_upload_sessions_v2(
           id,user_id,vault_object_id,expected_size,
           expected_hash,status,expires_at
         ) VALUES($1,$2,$3,$4,$5,'CREATED',now()+interval '10 minutes')`,
        [
          uploadId,
          input.userId,
          objectId,
          input.encryptedSize,
          input.contentHash,
        ],
      );

      await client.query("COMMIT");

      const upload = await this.storage.createUpload({
        objectKey,
        contentType: input.contentType,
        contentLength: input.encryptedSize,
      });

      return {
        uploadId,
        objectId,
        ...upload,
      };
    } catch (error) {
      await client.query("ROLLBACK");
      throw error;
    } finally {
      client.release();
    }
  }

  async finalize(userId: string, uploadId: string) {
    const result = await this.db.query(
      `SELECT s.*,v.object_key
       FROM vault_upload_sessions_v2 s
       JOIN vault_objects v ON v.id=s.vault_object_id
       WHERE s.id=$1 AND s.user_id=$2
         AND s.status IN('CREATED','UPLOADING')
         AND s.expires_at>now()`,
      [uploadId, userId],
    );

    const row = result.rows[0];
    if (!row) throw new Error("valid upload session not found");

    const object = await this.storage.head({
      objectKey: row.object_key,
    });

    if (Number(row.expected_size) !== object.size) {
      throw new Error("uploaded object size mismatch");
    }

    await this.db.query(
      `UPDATE vault_upload_sessions_v2
       SET status='FINALIZED',finalized_at=now()
       WHERE id=$1`,
      [uploadId],
    );

    return {
      finalized: true,
      objectId: row.vault_object_id,
    };
  }

  async list(userId: string) {
    const result = await this.db.query(
      `SELECT id,folder_id,name_ciphertext,metadata_ciphertext,
              key_envelope,content_hash,favorite,
              offline_available,created_at
       FROM vault_objects
       WHERE user_id=$1 AND deleted_at IS NULL
       ORDER BY created_at DESC`,
      [userId],
    );

    return result.rows;
  }
}
