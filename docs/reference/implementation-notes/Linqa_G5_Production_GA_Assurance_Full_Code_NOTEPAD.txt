# G5 GREEN Label

G5 is GREEN only if ALL sub-gates G5-A through G5-K are GREEN.

Code can enforce:
- evidence presence/hash binding
- revision/release binding
- scan/test results
- artifact signing/SBOM/provenance
- backup/DR/load/chaos thresholds
- canary/rollback rules
- mobile release checks
- approval completeness

Code cannot substitute for:
- independent cryptographic review
- independent penetration test
- legal/privacy approval
- human incident readiness
- app-store review

These must be represented as verified evidence artifacts and remain release-blocking.

import crypto from "node:crypto";

export interface BackupAdapter {
  createBackup(): Promise<{ id: string; checksum: string }>;
  restoreIntoIsolatedTarget(backupId: string): Promise<{ connection: string }>;
  runIntegrityChecks(connection: string): Promise<{
    schemaOk: boolean;
    rowCountsOk: boolean;
    criticalQueriesOk: boolean;
    encryptionMetadataOk: boolean;
  }>;
}

export class BackupRestoreVerifier {
  constructor(private readonly adapter: BackupAdapter) {}

  async proveRestore() {
    const backup = await this.adapter.createBackup();
    const restored = await this.adapter.restoreIntoIsolatedTarget(backup.id);
    const checks = await this.adapter.runIntegrityChecks(restored.connection);

    const passed =
      checks.schemaOk &&
      checks.rowCountsOk &&
      checks.criticalQueriesOk &&
      checks.encryptionMetadataOk;

    return {
      id: `dr_${crypto.randomUUID()}`,
      backupId: backup.id,
      backupChecksum: backup.checksum,
      passed,
      checks,
      verifiedAt: new Date().toISOString(),
    };
  }
}

export interface LoadResult {
  scenario: string;
  errorRate: number;
  p95Ms: number;
  p99Ms: number;
  sustainedMinutes: number;
}

const LIMITS: Record<string, { errorRate: number; p95Ms: number; minMinutes: number }> = {
  auth: { errorRate: 0.01, p95Ms: 1500, minMinutes: 30 },
  relay: { errorRate: 0.005, p95Ms: 1000, minMinutes: 60 },
  websocket_connect: { errorRate: 0.01, p95Ms: 2000, minMinutes: 30 },
};

export function assertLoadGate(r: LoadResult) {
  const limit = LIMITS[r.scenario];
  if (!limit) throw new Error(`UNKNOWN_LOAD_SCENARIO:${r.scenario}`);
  if (r.errorRate > limit.errorRate) throw new Error(`LOAD_ERROR_RATE_EXCEEDED:${r.scenario}`);
  if (r.p95Ms > limit.p95Ms) throw new Error(`LOAD_P95_EXCEEDED:${r.scenario}`);
  if (r.sustainedMinutes < limit.minMinutes) throw new Error(`LOAD_DURATION_TOO_SHORT:${r.scenario}`);
}

export interface CanaryMetrics {
  errorRate: number;
  authFailureDelta: number;
  relayFailureDelta: number;
  crashFreeSessions: number;
  securityAlertCount: number;
}

export type CanaryDecision = "CONTINUE" | "HOLD" | "ROLLBACK";

export function decideCanary(m: CanaryMetrics): CanaryDecision {
  if (m.securityAlertCount > 0) return "ROLLBACK";
  if (m.errorRate > 0.02) return "ROLLBACK";
  if (m.crashFreeSessions < 0.995) return "ROLLBACK";
  if (m.authFailureDelta > 0.25 || m.relayFailureDelta > 0.25) return "HOLD";
  return "CONTINUE";
}

import fs from "node:fs/promises";
import path from "node:path";
import crypto from "node:crypto";
import { EvidenceArtifact } from "./types";

export class EvidenceRegistry {
  constructor(private readonly root: string) {}

  async register(input: Omit<EvidenceArtifact, "sha256"> & { contentPath: string }) {
    const bytes = await fs.readFile(input.contentPath);
    const sha256 = crypto.createHash("sha256").update(bytes).digest("hex");

    const artifact: EvidenceArtifact = {
      id: input.id,
      type: input.type,
      path: input.path,
      sha256,
      producedAt: input.producedAt,
      producer: input.producer,
      environment: input.environment,
      metadata: input.metadata,
    };

    await fs.mkdir(this.root, { recursive: true });
    await fs.writeFile(
      path.join(this.root, `${artifact.id}.json`),
      JSON.stringify(artifact, null, 2),
    );

    return artifact;
  }

  async get(id: string): Promise<EvidenceArtifact | null> {
    try {
      return JSON.parse(await fs.readFile(path.join(this.root, `${id}.json`), "utf8"));
    } catch {
      return null;
    }
  }
}

import fs from "node:fs/promises";
import crypto from "node:crypto";

export interface ExternalEvidenceManifest {
  reviewType: "CRYPTO_REVIEW" | "PEN_TEST" | "PRIVACY_REVIEW";
  organization: string;
  reportPath: string;
  reportSha256: string;
  completedAt: string;
  criticalOpen: number;
  highOpen: number;
  approved: boolean;
}

export async function verifyExternalEvidence(m: ExternalEvidenceManifest) {
  const bytes = await fs.readFile(m.reportPath);
  const digest = crypto.createHash("sha256").update(bytes).digest("hex");

  if (digest !== m.reportSha256) throw new Error("EXTERNAL_REPORT_HASH_MISMATCH");
  if (!m.approved) throw new Error("EXTERNAL_REVIEW_NOT_APPROVED");
  if (m.criticalOpen > 0) throw new Error("OPEN_CRITICAL_FINDINGS");
  if (m.reviewType === "CRYPTO_REVIEW" && m.highOpen > 0) {
    throw new Error("OPEN_HIGH_CRYPTO_FINDINGS");
  }

  return true;
}

export function evaluateGate(
  gate: G5SubGate,
  revision: string,
  releaseVersion: string,
  checks: G5Check[],
): G5GateEvidence {
  const required = checks.filter(c => c.required);
  const failed = required.filter(c => !c.passed);

  return {
    gate,
    status: failed.length === 0 ? "GREEN" : required.some(c => c.passed) ? "AMBER" : "RED",
    generatedAt: new Date().toISOString(),
    revision,
    releaseVersion,
    checks,
  };
}

import fs from "node:fs/promises";
import path from "node:path";
import { G5GateEvidence, G5SubGate } from "./types";

const GATES: G5SubGate[] = [
  "G5-A","G5-B","G5-C","G5-D","G5-E",
  "G5-F","G5-G","G5-H","G5-I","G5-J","G5-K"
];

export async function assertG5Green(root: string, expectedRevision: string, releaseVersion: string) {
  const failures: string[] = [];

  for (const gate of GATES) {
    const file = path.join(root, gate, "gate-evidence.json");
    let evidence: G5GateEvidence;
    try {
      evidence = JSON.parse(await fs.readFile(file, "utf8"));
    } catch {
      failures.push(`${gate}:MISSING_EVIDENCE`);
      continue;
    }

    if (evidence.status !== "GREEN") failures.push(`${gate}:${evidence.status}`);
    if (evidence.revision !== expectedRevision) failures.push(`${gate}:REVISION_MISMATCH`);
    if (evidence.releaseVersion !== releaseVersion) failures.push(`${gate}:RELEASE_VERSION_MISMATCH`);
  }

  if (failures.length) {
    throw new Error(`LINQA_GA_BLOCKED:${failures.join(",")}`);
  }
}

export function classifyAlert(a: AlertSample): Severity {

  if (a.name === "linqa_refresh_replay_total" && a.value >= 10 && a.windowMinutes <= 5) return "PAGE";

}

export interface RotatableSecretProvider {
  id: string;
  currentVersion(name: string): Promise<string>;
  createNext(name: string): Promise<{ version: string }>;
  validate(name: string, version: string): Promise<boolean>;
  activate(name: string, version: string): Promise<void>;
  retire(name: string, version: string): Promise<void>;
}

export class SecretRotationCoordinator {
  constructor(private readonly provider: RotatableSecretProvider) {}

  async rotate(name: string) {
    const current = await this.provider.currentVersion(name);
    const next = await this.provider.createNext(name);

    if (!(await this.provider.validate(name, next.version))) {
      throw new Error(`SECRET_ROTATION_VALIDATION_FAILED:${name}`);
    }

    await this.provider.activate(name, next.version);

    // Retire only after the application has been observed using the new secret.
    return { name, previousVersion: current, activeVersion: next.version };
  }

  async retirePrevious(name: string, previousVersion: string) {
    await this.provider.retire(name, previousVersion);
  }
}

export interface SecurityEvent {
  id: string;
  type: string;
  occurredAt: string;
  actorUserId?: string;
  actorDeviceId?: string;
  severity: "LOW" | "MEDIUM" | "HIGH" | "CRITICAL";
  metadata?: Record<string,unknown>;
}

export interface SiemSink {
  send(batch: SecurityEvent[]): Promise<void>;
}

export class SecurityEventExporter {
  constructor(private readonly sink: SiemSink) {}

  async export(events: SecurityEvent[]) {
    const safe = events.map(e => ({
      ...e,
      metadata: this.redact(e.metadata ?? {}),
    }));
    await this.sink.send(safe);
  }

  private redact(obj: Record<string,unknown>) {
    const blocked = new Set(["plaintext","otp","token","privateKey","authorization","cvc","cardNumber"]);
    return Object.fromEntries(
      Object.entries(obj).map(([k,v]) => [k, blocked.has(k) ? "[REDACTED]" : v])
    );
  }
}
